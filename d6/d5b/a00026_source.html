14template <int dim, int dimworld, typename PatchGrid>
15requires(dim == 2) && (dimworld == 2 || dimworld == 3) && (dim <= dimworld)
16class IbraReader
17{
18 using GridFamily = typename PatchGrid::GridFamily;
19 using PatchData = NURBSPatchData<dim, dimworld, typename GridFamily::ctype>;
20 using PatchTrimData = typename GridFamily::TrimmerTraits::PatchTrimData;
21 using TrimmingCurve = typename GridFamily::TrimmerTraits::TrimmingCurve;
22
23 using ControlPointType = typename PatchData::ControlPointType;
24 using ControlPointNetType = typename PatchData::ControlPointNetType;
25
26public:
27 static auto read(const std::string& fileName, const bool trim = true,
28 std::array<int, 2> preKnotRefine = {0, 0}) -> std::tuple<PatchData, std::optional<PatchTrimData>> {
29 Ibra::Brep brep = readJson<dimworld>(fileName);
30 // Each surface in a brep is one Patch, as for now only one brep is supported in NURBSGrid
31 assert(brep.surfaces.size() == 1);
32
33 std::cout << "IbraReader successfully read in 1 patch with " << brep.trims.size() << " edges." << std::endl;
34
35 // Reader has done its job, now NURBSGrid can be constructed
36
37 Ibra::Surface<dimworld> _surface = brep.surfaces[0];
38 const std::array<std::vector<double>, dim> knotSpans = _surface.compileKnotVectors();
39 const std::vector<std::vector<ControlPointType>> controlPoints =
40 _surface.template transformControlPoints<ControlPointType>();
41 std::array<int, dim> dimsize = _surface.n_controlPoints;
42
43 auto controlNet = ControlPointNetType(dimsize, controlPoints);
44 PatchData _patchData{knotSpans, controlNet, _surface.degree};
45
46 // Optional preKnot refinement
47 for (const auto i : std::views::iota(0, dim)) {
48 if (preKnotRefine[i] > 0) {
49 auto newKnots = Splines::generateRefinedKnots(knotSpans, i, preKnotRefine[i]);
50 _patchData = Splines::knotRefinement(_patchData, newKnots, i);
51 }
52 }
53
54 if (trim) {
55 PatchTrimData trimData{};
56 constructTrimmingCurves(brep, trimData);
57 return std::make_tuple(_patchData, trimData);
58 }
59 return std::make_tuple(_patchData, std::nullopt);
60 }
61
62private:
63 static void constructTrimmingCurves(const Ibra::Brep<dimworld>& brep, PatchTrimData& trimData) {
64 const std::vector<Ibra::BrepLoop> loops = brep.loops;
65 for (int i = 0; const Ibra::BrepLoop& loop : loops) {
66 trimData.addLoop();
67 for (const Ibra::BrepTrim& trim : loop.trims)
68 trimData.insertTrimCurve(trim.asCurve<TrimmingCurve>(), i);
69 i++;
70 }
71 }
72};
