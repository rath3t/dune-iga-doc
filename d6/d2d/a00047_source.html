58 const IdType& subId(const IdType& elementId, int localSubIndex, int codim) const {

80 int outsideIntersectionIndex(const IdType& insideElementId, const IdType& outsideElementId, int indexInInside) const {
81 const IdType& insideSubId = subId(insideElementId, indexInInside, 1);
82
83 for (const auto i : Dune::range(subIds(outsideElementId, 1))) {
84 const IdType& outsideSubId = subId(outsideElementId, i, 1);
85 if (outsideSubId == insideSubId) {
86 return i;
87 }
88 }
89 DUNE_THROW(GridError, "outsideIntersectionIndex not successfull");
90 }

142 template <int cc>
143 auto subIndexFromId(const IdType& id, int i, int codim, int lvl) const {
144 assert(codim > 0);
145 if constexpr (cc == 0) {
146 if (codim == 1)
147 return idToEdgeInfoMap.at(globalEdgesIdOfElementsMap_.at(id)[i]).indexInLvlStorage;
148 else if (codim == 2)
149 return idToVertexInfoMap[lvl].at(globalVerticesIdOfElementsMap.at(id)[i]).indexInLvlStorage;
150 __builtin_unreachable();
151 } else if constexpr (cc == 1) {
152 if (codim == 1)
153 return idToEdgeInfoMap.at(id).indexInLvlStorage;
154 else if (codim == 2) // get index of vertex from edge id
155 return idToVertexInfoMap[lvl].at(globalVertexIdOfEdgesMap_.at(id)[i]).indexInLvlStorage;
156 __builtin_unreachable();
157 } else {
158 assert(codim == 2 and i == 0);
159 return idToVertexInfoMap[lvl].at(id).indexInLvlStorage;
160 }

174 GeoTypes types(int codim, int level) const {
175 if (codim == 0) {
176 if (numberOfTrimmedElements[level] == 0)

179 if (numberOfUnTrimmedElements[level] == 0)

204 std::size_t size(GeometryType type, int lvl) const {
205 if (type == GeometryTypes::quadrilateral)
206 return numberOfUnTrimmedElements[lvl];
207 if (type == GeometryTypes::none(gridDim))
208 return numberOfTrimmedElements[lvl];
209 if (type == GeometryTypes::line)
210 return std::get<1>(entityImps_[lvl]).size();
211 if (type == GeometryTypes::vertex)
212 return std::get<2>(entityImps_[lvl]).size();
213 return 0;
214 }

216 std::size_t sizeOfInfos(int codim, int lvl) const {
217 if (codim == 2)
218 return idToVertexInfoMap[lvl].size();
219
220 auto count = [=](auto&& range) -> std::size_t {
221 return std::ranges::count_if(range, [=](const auto& entityInfoPair) { return entityInfoPair.second.lvl == lvl; });
222 };
223 if (codim == 0)
224 return count(idToElementInfoMap);
225 if (codim == 1)
