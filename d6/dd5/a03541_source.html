26 template <class GridImp, IntersectionType type_>
27 struct PatchGridIntersectionImpl
28 {
29 constexpr static int dim = GridImp::dimension;
30 constexpr static int mydim = GridImp::dimension - 1;
31 constexpr static int dimworld = GridImp::dimensionworld;
32
33 using Geometry = typename GridImp::template Codim<1>::Geometry;
34 using LocalGeometry = typename GridImp::template Codim<1>::LocalGeometry;
35 using Entity = typename GridImp::template Codim<0>::Entity;
36
37 using ctype = typename GridImp::ctype;
38 using NormalVector = FieldVector<ctype, dimworld>;
39 using LocalCoordinate = FieldVector<ctype, mydim>;
40
41 using ParameterSpaceIntersection =
42 std::conditional_t<type_ == IntersectionType::Leaf,
43 typename GridImp::ParameterSpace::ParameterSpaceTraits::ParameterSpaceLeafIntersection,
44 typename GridImp::ParameterSpace::ParameterSpaceTraits::ParameterSpaceLevelIntersection>;
45
46 PatchGridIntersectionImpl() = default;
47
48 PatchGridIntersectionImpl(const GridImp* parameterSpaceGrid, const ParameterSpaceIntersection& hostIntersection)
49 : patchGrid_(parameterSpaceGrid),
50 parameterSpaceIntersection_(hostIntersection) {}
51
52 PatchGridIntersectionImpl(const GridImp* parameterSpaceGrid, ParameterSpaceIntersection&& hostIntersection)
53 : patchGrid_(parameterSpaceGrid),
54 parameterSpaceIntersection_(std::move(hostIntersection)) {}
55
56 bool equals(const PatchGridIntersectionImpl& other) const {
57 return parameterSpaceIntersection_ == other.parameterSpaceIntersection_;
58 }
59
60 // return Entity on the inside of this intersection
61 // (that is the Entity where we started this Iterator)
62 Entity inside() const {
63 return PatchGridEntity<0, dim, GridImp>(patchGrid_, parameterSpaceIntersection_.inside());
64 }
65
66 // return Entity on the outside of this intersection
67 // (that is the neighboring Entity)
68 Entity outside() const {
69 return PatchGridEntity<0, dim, GridImp>(patchGrid_, parameterSpaceIntersection_.outside());
70 }
71
72 // return true if intersection is with boundary.
73 bool boundary() const {
74 return parameterSpaceIntersection_.boundary();
75 }
76
82 NormalVector centerUnitOuterNormal() const {
83 LocalCoordinate localcenter;
84 localcenter = 0.5;
85 return this->unitOuterNormal(localcenter);
86 }
87
88 // return true if across the edge an neighbor on this level exists
89 bool neighbor() const {
90 return parameterSpaceIntersection_.neighbor();
91 }
92
93 // return the boundary segment index
94 size_t boundarySegmentIndex() const {
95 return parameterSpaceIntersection_.boundarySegmentIndex();
96 }
97
98 // Return true if this is a conforming intersection
99 bool conforming() const {
100 return parameterSpaceIntersection_.conforming();
101 }
102
103 // Geometry type of an intersection
104 GeometryType type() const {
105 return parameterSpaceIntersection_.type();
106 }
107
108 // intersection of codimension 1 of this neighbor with element where
109 // iteration started.
110 // Here returned element is in LOCAL coordinates of the element
111 // where iteration started.
112 LocalGeometry geometryInInside() const {
113 return LocalGeometry(typename LocalGeometry::Implementation(parameterSpaceIntersection_.geometryInInside()));
114 }
115
116 // intersection of codimension 1 of this neighbor with element where iteration started.
117 // Here returned element is in LOCAL coordinates of neighbor
118 LocalGeometry geometryInOutside() const {
119 return LocalGeometry(typename LocalGeometry::Implementation(parameterSpaceIntersection_.geometryInOutside()));
120 }
121
122 // intersection of codimension 1 of this neighbor with element where iteration started.
123 // Here returned element is in GLOBAL coordinates of the element where iteration started.
124 Geometry geometry() const {
125 // @todo trim this will be wrong as soon as the intersection geometry has a special geoemtry
126 auto geo =
127 typename Geometry::Implementation(parameterSpaceIntersection_.geometry(),
128 patchGridGeometry().template localView<1, typename GridImp::ParameterSpace>());
129 return Geometry(geo);
130 }
131
132 // local number of codim 1 entity in self where intersection is contained in
133 int indexInInside() const {
134 return parameterSpaceIntersection_.indexInInside();
135 }
136
137 // local number of codim 1 entity in neighbor where intersection is contained
138 int indexInOutside() const {
139 return parameterSpaceIntersection_.indexInOutside();
140 }
141
142 // return outer normal
143 FieldVector<ctype, GridImp::dimensionworld> outerNormal(const LocalCoordinate& local) const {
144 FieldMatrix<ctype, dimworld, dim> J =
145 inside().geometry().jacobianInverseTransposed(geometryInInside().global(local));
146 FieldVector<ctype, dimworld> res;
147 J.mv(parameterSpaceIntersection_.outerNormal(local), res);
148 return res;
149 }
150
151 // return outer normal multiplied by the integration element
152 FieldVector<ctype, GridImp::dimensionworld> integrationOuterNormal(const LocalCoordinate& local) const {
153 const ctype detJ = this->geometry().integrationElement(local);
154 FieldVector<ctype, dimworld> res = unitOuterNormal(local);
155 res *= detJ;
156 return res;
157 }
158
159 // return unit outer normal
160 FieldVector<ctype, GridImp::dimensionworld> unitOuterNormal(const LocalCoordinate& local) const {
161 auto Jinv = inside().geometry().jacobianInverseTransposed(geometryInInside().global(local));
162 FieldVector<ctype, dimworld> res;
163 Jinv.mv(parameterSpaceIntersection_.unitOuterNormal(local), res);
164 res /= res.two_norm();
165 return res;
166 }
167
168 bool isTrimmed() const {
169 if constexpr (requires { parameterSpaceIntersection_.isTrimmed(); })
170 return parameterSpaceIntersection_.isTrimmed();
171 return false;
172 }
173
174 private:
175 auto& patchGridGeometry() const {
176 if constexpr (type_ == IntersectionType::Leaf)
177 return patchGrid_->patchGeometryAtBack();
178 else
179 return patchGrid_->patchGeometry(inside().level());
180 }
181 const GridImp* patchGrid_{};
182 ParameterSpaceIntersection parameterSpaceIntersection_{};
183 };
