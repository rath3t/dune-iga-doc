25auto findGoodStartingPoint(const auto& curve, const Clipper2Lib::PointD& pt) -> double {
26 int N = std::max(curve.numberOfControlPoints().front() * 30, 200);
27 auto linSpace = Utilities::linspace(curve.domain().front(), N);
28 auto distances = std::ranges::transform_view(linSpace, [&](const auto u) { return distance(pt, curve.global(u)); });
29 auto min_idx = std::ranges::distance(distances.begin(), std::ranges::min_element(distances));
30 return linSpace[min_idx];
31}

51auto callFindIntersection(const auto& curvePatchGeo, int edgeIdx, const auto& ip,
52 const auto& corners) -> std::pair<double, FieldVector<double, 2>> {
53 auto pos = corners[edgeIdx];
54 auto dir = edgeDirections[edgeIdx];
55 double lineGuess = (edgeIdx == 0 or edgeIdx == 2) ? (ip.x - pos[0]) / dir[0] : (ip.y - pos[1]) / dir[1];
56
57 // Catch a trivial but difficult case (algorithm sometimes is not able to converge to endPoint)
58 if (auto endPoint = curvePatchGeo.corner(1);
59 approxSamePoint(ip, endPoint, 1e-8) and not FloatCmp::eq(endPoint, curvePatchGeo.corner(0))) {
60 return std::make_pair(curvePatchGeo.domain()[0][1], endPoint);
61 }
62
63 // Todo use z-Value to get a good starting point, not brute-force
64 auto guessTParam = FieldVector<double, 2>{findGoodStartingPoint(curvePatchGeo, ip), lineGuess};
65
66 auto [success, tParam, curvePoint] = findIntersectionCurveAndLine(curvePatchGeo, pos, dir, guessTParam);
67 if (success == IntersectionCurveAndLine::intersect)
68 return std::make_pair(tParam[0], curvePoint);
69 if (success == IntersectionCurveAndLine::parallel) {
70 DUNE_THROW(Dune::GridError, "Couldn't find intersection Point, lines are parallel");
71 }
72
73 auto domain = curvePatchGeo.domain();
74 if (FloatCmp::eq(curvePoint, curvePatchGeo.global({domain[0].front()})))
75 return std::make_pair(domain[0][0], curvePoint);
76 if (FloatCmp::eq(curvePoint, curvePatchGeo.global({domain[0].back()})))
77 return std::make_pair(domain[0].back(), curvePoint);
78
79 DUNE_THROW(Dune::GridError, "Couldn't find intersection Point");
80};
