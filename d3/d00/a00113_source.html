12template <typename GridImp>
13struct SimplexIntegrationRuleGenerator
14{
15 using PatchElement = typename GridImp::Traits::template Codim<0>::Entity;
16
17 static constexpr int dim = GridImp::dimension;
18
19 struct Parameters
20 {
21 int maxBoundaryDivisions{5};
22 double targetTolerance{1e-4}; // not used atm
23 };
24
25 static auto createIntegrationRule(const PatchElement& element, GridImp* grid, int quadratureOrder,
26 const QuadratureType::Enum qt = QuadratureType::GaussLegendre,
27 const Parameters& parameters = Parameters{}) {
28 auto gv = grid->levelGridView(element.level());
29
30 std::vector<Point> vertices{};
31 std::vector<Element> elements{};
32
33 for (const auto& intersection : intersections(gv, element)) {
34 auto geometryInInside = intersection.geometryInInside();
35
36 std::vector<Point> pointsT = splitBoundary(geometryInInside, parameters);
37 vertices.insert(vertices.end(), pointsT.begin(), pointsT.end());
38 }
39
40 auto indices = triangulate(vertices);
41 for (auto it = indices.begin(); it < indices.end(); it += 3)
42 elements.emplace_back(GeometryTypes::triangle, std::vector<FieldVector<double, dim>>{
43 vertices[*it], vertices[*(it + 1)], vertices[*(it + 2)]});
44
45 return makeQuadratureRule(elements, quadratureOrder, qt);
46 }
47
48private:
49 using Point = FieldVector<double, dim>;
50 using Element = MultiLinearGeometry<double, dim, dim>;
51 using Index = std::uint64_t;
52
53 static std::vector<Point> splitBoundary(const auto& localGeometry, const Parameters& parameters) {
54 std::vector<Point> points;
55 points.reserve(parameters.maxBoundaryDivisions);
56
57 // todo at the moment only boundaryDivisions
58 for (auto local : Utilities::linspace(0.0, 1.0, parameters.maxBoundaryDivisions))
59 points.push_back(localGeometry.global({local}));
60
61 return points;
62 }
63 static auto triangulate(const std::vector<Point>& points) -> std::vector<Index> {
64 std::vector<std::vector<Point>> polygonInput;
65 polygonInput.push_back(points);
66
67 return mapbox::earcut<Index>(polygonInput);
68 }
69 static auto makeQuadratureRule(const std::vector<Element>& elements, int quadratureOrder, const QuadratureType::Enum qt) {
70 assert(qt == QuadratureType::GaussLegendre);
71
72 Dune::QuadratureRule<double, dim> vector{};
73
74 for (auto& subElement : elements) {
75 const auto& rule = Dune::QuadratureRules<double, dim>::rule(subElement.type(), quadratureOrder, qt);
76 for (auto ip : rule) {
77 auto globalInSpan = subElement.global(ip.position());
78 vector.emplace_back(globalInSpan, ip.weight() * subElement.integrationElement(ip.position()));
79 }
80 }
81 return vector;
82 }
83};
