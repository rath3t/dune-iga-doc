16inline int giveEdgeIdx(const std::size_t e1, const std::size_t e2) {
17 if ((e1 == 0 and e2 == 1) or (e1 == 1 and e2 == 0))
18 return 0;
19 if ((e1 == 1 and e2 == 2) or (e1 == 2 and e2 == 1))
20 return 1;
21 if ((e1 == 2 and e2 == 3) or (e1 == 3 and e2 == 2))
22 return 2;
23 if ((e1 == 3 and e2 == 0) or (e1 == 0 and e2 == 3))
24 return 3;
25 DUNE_THROW(Dune::GridError, "No corresponding edge Index for Input");
26}

28inline bool isConsecutive(const std::size_t e1, const std::size_t e2) {
29 assert(e2 < 4);
30 if (e1 == 3)
31 return e2 == 0;
32 if (e1 > e2)
33 return false;
34 return e2 == e1 + 1;
35}

53 auto it = std::ranges::find_if(eleRect, [&](const auto& vertex) {
54 return FloatCmp::eq(vertex.x, pt.x, 1e-8) and FloatCmp::eq(vertex.y, pt.y, 1e-8);

172 void finish(const auto& patchTrimData) {
173 // Sort the points in counter clockwise manner such that the first point is in the lower left
174 const auto minVertex = lowerLeftVertex();
175 std::ranges::sort(vertices_, [&](const Vertex& aV, const Vertex& bV) { return comparePoints(aV, bV, minVertex); });
176
177 // Resort that we always start on a hostVertex (if there is one)
178 if (const auto it = std::ranges::find_if(vertices_, [](const Vertex& vV) { return vV.isHost(); });
179 it != vertices_.end())
180 std::ranges::rotate(vertices_, it);
181
182 // Now resort InsideVertices (sometimes they end up on the wrong position)
183 for (const auto i : std::views::iota(0ul, vertices_.size())) {
184 if (vertices_[i].isInside()) {
185 const auto insideVertex = vertices_[i];
186 auto it = vertices_.begin() + static_cast<long>(i);
187 const size_t indexBeforeIt =
188 it == vertices_.begin() ? vertices_.size() - 1 : std::ranges::distance(vertices_.begin(), it) - 1;
189 const size_t indexAfterIt =
190 it == std::prev(vertices_.end()) ? 0 : std::ranges::distance(vertices_.begin(), it) + 1;
191
192 // Check vertex before if they line up correctly
193 const auto vertexBefore = vertices_[indexBeforeIt];
194 const auto vertexAfter = vertices_[indexAfterIt];
195
196 if (vertexBefore.isNew() and vertexAfter.isNew()) {
197 auto indices = patchTrimData.getIndices(vertexBefore.zValue());
198 if (indices.loop == insideVertex.loop() and indices.curve == insideVertex.formerCurve())
199 continue;
200 }
201 // If we end up here sort the vertex to a correct place
202 auto correctVertexIt = findCorrespondingVertexIt(insideVertex, patchTrimData);
203 assert(correctVertexIt != vertices_.end() && "No corresponding newVertex found");
204
205 // Add the insideVertex at correctVertexIt+1
206 const auto correctItForInsideVertex = correctVertexIt + 1;
207 if (std::distance(vertices_.begin(), correctItForInsideVertex) < i) {
208 vertices_.insert(correctItForInsideVertex, insideVertex);
209 vertices_.erase(vertices_.begin() + static_cast<long>(i + 1));
210 } else {
211 std::ranges::rotate(it, it + 1, correctItForInsideVertex);
212 }
213 }
214 }
215 }

244 // Sort the points based on polar angle with respect to the reference point
245 inline static auto comparePoints = [](const Vertex& aV, const Vertex& bV, const Vertex& referenceV) -> bool {
246 auto polarAngle = [](const auto& p, const auto& reference) -> double {
247 return atan2(p.y - reference.y, p.x - reference.x);
248 };
249
250 const auto a = aV.pt;
251 const auto b = bV.pt;
252 const auto reference = referenceV.pt;
253 const double angleA = polarAngle(a, reference);
254 const double angleB = polarAngle(b, reference);
255
256 if (angleA != angleB)
257 return angleA < angleB;
258
259 // If two points have the same angle, sort based on distance to the reference point
260 return (a.x - reference.x) * (a.x - reference.x) + (a.y - reference.y) * (a.y - reference.y) <
261 (b.x - reference.x) * (b.x - reference.x) + (b.y - reference.y) * (b.y - reference.y);
262 };
263
264 auto findCorrespondingVertexIt(const Vertex& insideVertex, const auto& patchTrimData) {
265 return std::ranges::find_if(vertices_, [&](const Vertex& vV) {
266 if (vV.isNew()) {
267 const auto indices = patchTrimData.getIndices(vV.zValue());
268 return indices.loop == insideVertex.loop() and indices.curve == insideVertex.formerCurve();
269 } else
270 return false;
271 });
272 }

274 [[nodiscard]] const Vertex& lowerLeftVertex() const {
275 return *std::ranges::min_element(vertices_, [&](const Vertex& aV, const Vertex& bV) {
276 const auto a = aV.pt;
277 const auto b = bV.pt;
278 return (a.y < b.y) || (a.y == b.y && a.x < b.x);
279 });
280 }
