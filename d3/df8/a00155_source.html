37 class CurveLoopIndexEncoder
38 {
39 static constexpr int loopBits = 4;
40 static constexpr int curveBits = 8;
41 static constexpr int sampleBits = 12;
42
43 static constexpr u_int64_t loopMask{(1ULL << loopBits) - 1};
44 static constexpr u_int64_t curveMask{(1ULL << curveBits) - 1};
45 static constexpr u_int64_t sampleMask{(1ULL << sampleBits) - 1};
46
47 public:
48 static int64_t encode(int loopIndex, int curveIndex, int sampleIndex) {
49 // the left hand side shifts the curve index value away from loop values and then inserts the curve index
50 // example curveBits = 4 and loopBits = 3 and sampleBits=12;
51 // curveIndex = 3 and loopIndex = 2 = 00010_b , sampleIndex= 200= 11001000_b
52 // curveIndex = 3 = 00011_b
53 // curveIndex<< (loopBits+ sampleBits) = 110,000,000,000,000,000_b // shift the curve index 16 bits to the left
54 // loopIndex<< sampleBits = 10,000,000,000,000_b // shift the loop index 12 bits to the left
55 // sampleIndex = 11,001,000_b // shift the loop index 12 bits to the left
56 // bitwise or of the three values: 110,010,000,011,001,000_b //
57 // (curveIndex<< loopBits) | loopIndex = 11000_b | 00010_b = 11010_b = 205000_d
58 return (static_cast<u_int64_t>(curveIndex) << (loopBits + sampleBits)) |
59 (static_cast<u_int64_t>(loopIndex) << sampleBits) | sampleIndex;
60 }
61
62 struct IndexResult
63 {
64 int loop;
65 int curve;
66 int sample;
67
68 friend std::ostream& operator<<(std::ostream& os, const IndexResult& res) {
69 os << "Loop: " << res.loop << " Curve: " << res.curve << " Sample: " << res.sample;
70 return os;
71 }
72 };
73
74 static IndexResult decode(int64_t singleIndex) {
75 // since the sample index is stored in the right-most bits, we only have to apply the bitwise "and" with
76 // sampleMask, which returns simply the sample index
77 int sampleIndex = singleIndex & sampleMask;
78 // shift single index to have the loop index at the rightmost position and apply the bitwise and with the loop
79 // mask
80 const int loopAndCurveIndex = singleIndex >> sampleBits;
81 int loopIndex = loopAndCurveIndex & loopMask;
82 // shift again to have the curve index at the rightmost position and apply the bitwise and with the curve mask
83 int curveIndex = (loopAndCurveIndex >> loopBits) & curveMask;
84 return {loopIndex, curveIndex, sampleIndex};
85 }
86 };

96 class CurveManager
97 {
98 friend PatchTrimDataImpl;
99
100 public:
101 using idx_t = u_int64_t;
102 static constexpr idx_t indexOffSet = 5; // Since the 0 is reserved for untouched points and the following 4
103 // values are reserved for the corners of the untrimmed square all indices start at 5
104 void addLoop(const Impl::BoundaryLoop<TrimmingCurve>& loop) {
105 Clipper2Lib::PathD path;
106 // since the first
107 for (int curveIndex = 0; const auto& curve : loop.curves()) {
108 if (curve.affine() == 1 and loops_.empty()) {
109 auto p1 = curve.corner(0);
110 auto p2 = curve.corner(1);
111 path.emplace_back(p1[0], p1[1], getZValue(loops_.size(), curveIndex, 0));
112 path.emplace_back(p2[0], p2[1], getZValue(loops_.size(), curveIndex, 1));
113 } else {
114 for (int vi = 0; const auto v : Utilities::linspace(curve.domain()[0], splitter_)) {
115 auto fV = curve.global({v});
116 path.emplace_back(fV[0], fV[1], getZValue(loops_.size(), curveIndex, vi++));
117 }
118 }
119 ++curveIndex;
120 }
121 loops_.push_back(path);
122 }
123
124 [[nodiscard]] auto getIndices(const idx_t val) const -> Impl::CurveLoopIndexEncoder::IndexResult {
125 if (val == 0) // untouched point usually created by Clipperlib as Intersection point
126 return Impl::CurveLoopIndexEncoder::IndexResult{-1, -1, -1};
127 assert(val >= indexOffSet &&
128 "Passed index not decodable, since it is smaller than the offset and non-zero, which usually means that "
129 "the point is a host vertex");
130 return Impl::CurveLoopIndexEncoder::decode(val - indexOffSet);
131 }
132
133 [[nodiscard]] auto getZValue(int loopIndex, int curveIndex, int sampleIndex) const {
134 return Impl::CurveLoopIndexEncoder::encode(loopIndex, curveIndex, sampleIndex) + indexOffSet;
135 }

137 private:
138 Clipper2Lib::PathsD loops_;
139 idx_t splitter_{};
140 };

179 void prepare(const ParameterType& parameters, const std::array<std::vector<ctype>, 2>& coordinates) {
180 manager_.splitter_ = parameters.splitter;
181 std::ranges::for_each(loops_, [&](const auto& loop) { manager_.addLoop(loop); });
182
183 std::array domainU{coordinates[0].front(), coordinates[0].back()};
184 std::array domainV{coordinates[1].front(), coordinates[1].back()};
185
186 // Determine points of the outer boundary that are not on the edges
187 auto isInsidePatch = [&](const FieldVector<ctype, 2>& pt) {
188 const bool isInsideU = FloatCmp::gt(pt[0], domainU.front()) and FloatCmp::lt(pt[0], domainU.back());
189 const bool isInsideV = FloatCmp::gt(pt[1], domainV.front()) and FloatCmp::lt(pt[1], domainV.back());
190
191 return isInsideU and isInsideV;
192 };
193
194 for (const auto& loop : loops()) {
195 pointsInPatch_.push_back({});
196 for (size_t i = 0; const auto& curve : loop.curves()) {
197 if (const auto pt = curve.corner(1); isInsidePatch(pt)) {
198 pointsInPatch_.back().emplace_back(pt, i, (i + 1) % loop.size());
199 }
200 ++i;
201 }
202 }
203
204 finished_ = true;
205 }
206
207private:
208 std::vector<std::vector<Impl::PointInPatch<ctype>>> pointsInPatch_;
209 bool finished_ = false;
210 std::vector<Impl::BoundaryLoop<TrimmingCurve>> loops_;
211 CurveManager manager_;
212};
