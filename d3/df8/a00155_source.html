19auto clipElementRectangle(const auto& element, const auto& patchTrimData)
20 -> std::tuple<ElementTrimFlag, ClippingResult> {
21 using namespace Clipper2Lib;
22
23 auto eleGeo = element.geometry();
24 static constexpr int numberOfCorners = 4;
25
26 PathD eleRect;
27 for (const auto vertexIndex : Dune::range(numberOfCorners)) {
28 auto corner = eleGeo.corner(vertexIndexMapping[vertexIndex]);
29 eleRect.emplace_back(corner[0], corner[1], vertexIndex + 1);
30 }
31
32 const auto& trimmingCurves = patchTrimData.clipperLoops();
33
34 // Element is trimmed
35 ClipperD clipper(8);
36 ClippingResult result(eleRect);
37
38 auto checkParallel = [&](const auto& curve, const int edgeIndex) -> bool {
39 return curve.affine() and std::get<0>(findIntersectionLinearCurveAndLine(
40 curve, eleGeo.corner(vertexIndexMapping[edgeIndex]), edgeDirections[edgeIndex],
41 {curve.domainMidPoint()[0], 0.5})) == IntersectionCurveAndLine::parallel;
42 };
43
44 clipper.SetZCallback([&](const PointD& rectangleFirstPoint, const PointD& rectangleSecondPoint,
45 const PointD& firstTrimmingCurvePoint, const PointD& secondTrimmingCurvePoint,
46 const PointD& intersectionPoint) {
47 assert(rectangleFirstPoint.z > 0 and rectangleFirstPoint.z < 5 &&
48 "Intersection should only occur with the rectangle edges");
49 assert(rectangleSecondPoint.z > 0 and rectangleSecondPoint.z < 5 &&
50 "Intersection should only occur with the rectangle edges");
51
52 const auto edgeIdx = giveEdgeIdx(rectangleFirstPoint.z - 1, rectangleSecondPoint.z - 1);
53
54 const auto vertexZValue = secondTrimmingCurvePoint.z;
55
56 // Now check that we don't have a parallel intersection
57 if (checkParallel(patchTrimData.getCurve(vertexZValue), edgeIdx)) {
58 return;
59 }
60
61 const auto [isHost, idx] = isCornerVertex(intersectionPoint, eleRect);
62 if (isHost) {
63 result.addOriginalVertex(idx, vertexZValue);
64 return;
65 }
66
67 auto intersectionIndices = patchTrimData.getIndices(intersectionPoint.z);
68
69 auto firstTrimmingCurveIndices = patchTrimData.getIndices(firstTrimmingCurvePoint.z);
70 auto secondTrimmingCurveIndices = patchTrimData.getIndices(secondTrimmingCurvePoint.z);
71
72 assert(firstTrimmingCurveIndices.loop == secondTrimmingCurveIndices.loop &&
73 "The points of the trimming curves should be on the same loop");
74
75 assert(firstTrimmingCurveIndices.curve == secondTrimmingCurveIndices.curve or
76 secondTrimmingCurveIndices.curve == intersectionIndices.curve or
77 firstTrimmingCurveIndices.curve == intersectionIndices.curve &&
78 "The indices of the trimming curves should be the same or the intersection point should be on one of "
79 "the two second curves");
80
81 result.addNewVertex(edgeIdx, intersectionPoint, vertexZValue);
82 });
83
84 PathsD resultClosedPaths{};
85 PathsD resultOpenPaths{};
86
87 clipper.AddClip(trimmingCurves);
88 clipper.AddSubject({eleRect});
89
90 clipper.Execute(ClipType::Intersection, FillRule::NonZero, resultClosedPaths, resultOpenPaths);
91 assert(resultOpenPaths.empty() && "There should be no open paths in the clipping result");
92 if (resultClosedPaths.size() > 1)
93 DUNE_THROW(Dune::GridError,
94 "Clipping was not successful: Multiple closed regions per element, please refine accordingly");
95
96 auto isFullElement = [&](const auto& clippedEdges) {
97 return clippedEdges.size() == 4 and
98 (clippedEdges[0].z < 5 and clippedEdges[1].z < 5 and clippedEdges[2].z < 5 and clippedEdges[3].z < 5);
99 };
100
101 if (resultClosedPaths.empty())
102 return std::make_tuple(ElementTrimFlag::empty, ClippingResult{eleRect});
103
104 if (resultClosedPaths.size() == 1 and isFullElement(resultClosedPaths.front())) {
105 return std::make_tuple(ElementTrimFlag::full, ClippingResult{eleRect});
106 }
107
108 // Add corner vertices that are not yet present in the clip path
109 for (const auto& pt : resultClosedPaths.front()) {
110 if (auto [isHost, idx] = isCornerVertex(pt, eleRect); isHost)
111 result.addOriginalVertex(idx);
112 // result.addOriginalVertex(idx, pt.z > 4 ? std::optional<size_t>(pt.z) : std::nullopt);
113
114 }
115
116 // Now we have to check for points inside the elements
117 for (auto i : Dune::range(patchTrimData.loops().size())) {
118 for (const auto& pt : patchTrimData.getPointsInPatch(i)) {
119 if (const PointD cp{pt.pt[0], pt.pt[1]}; PointInPolygon(cp, eleRect) == PointInPolygonResult::IsInside) {
120 result.addInsideVertex(cp, pt.curveIdxI, pt.curveIdxJ, i);
121 }
122 }
123 }
124
125 // Add startpoints of trimming curves to the vertices if they are on one of the element edges
126 for (auto cI = 0; cI < patchTrimData.loops().front().curves().size(); ++cI) {
127 const auto& curve = patchTrimData.loops().front().curves()[cI];
128
129 for (int i = 0; i < 2; ++i) {
130 auto pt = curve.corner(i);
131 PointD ptClipper{pt[0], pt[1]};
132
133 auto [isHost, idx] = isCornerVertex(ptClipper, eleRect);
134 if (isHost) {
135 size_t additionalZVal = patchTrimData.getZValue(0, cI,
136 i == 0 ? 0
137 : curve.affine() ? 1
138 : patchTrimData.getSplitter() - 1);
139 result.addOriginalVertex(idx, additionalZVal);
140 continue;
141 }
142
143 for (auto e = 0; e < edgeLookUp.size(); ++e) {
144 if (const auto& edgeIdx = edgeLookUp[e];
145 GeometryKernel::isPointOnLineSegment(pt, eleGeo.corner(edgeIdx.front()), eleGeo.corner(edgeIdx.back())) &&
146 !checkParallel(curve, e)) {
147 result.addNewVertex(e, ptClipper, patchTrimData.getZValue(0, cI, std::numeric_limits<u_int64_t>::infinity()));
148 break;
149 }
150 }
151 }
152 }
153
154 // Check again if element is full after adding additional vertices
155 if (result.vertices_.size() == 4 and
156 std::ranges::all_of(result.vertices_, [](const ClippingResult::Vertex& v) { return v.isHost(); })) {
157 return std::make_tuple(ElementTrimFlag::full, ClippingResult{eleRect});
158 }
159
160 result.finish(patchTrimData);
161
162 // While developing
163 // result.report();
164
165 return std::make_tuple(ElementTrimFlag::trimmed, result);
166}
