601 std::vector<ElementTrimData> trimElements(std::optional<int> level_ = std::nullopt) const {
602 int level = level_.value_or(maxLevel());
603 std::vector<ElementTrimData> elementTrimDatas;
604 auto gv = parameterSpaceGrid_->levelGridView(level);
605 for (const auto& ele : elements(gv)) {
606 if (trimData_.has_value())
607 elementTrimDatas.emplace_back(trimElement(ele, trimData_.value()));
608 else
609 elementTrimDatas.emplace_back(ElementTrimFlag::full, ele);
610 }
611 return elementTrimDatas;
612 }
