37template <int dim, int dimworld, typename PatchGrid>
38requires(dim == 2) && (dimworld == 2 || dimworld == 3) && (dim <= dimworld)
39class IbraReader<dim, dimworld, PatchGrid, true>
40{
41 using GridFamily = typename PatchGrid::GridFamily;
42 using PatchData = NURBSPatchData<dim, dimworld, typename GridFamily::ctype>;
43 using PatchTrimData = typename GridFamily::TrimmerTraits::PatchTrimData;
44 using TrimmingCurve = typename GridFamily::TrimmerTraits::TrimmingCurve;
45
46 using ControlPointType = typename PatchData::ControlPointType;
47 using ControlPointNetType = typename PatchData::ControlPointNetType;
48
49public:
50 static auto read(const std::string& fileName, const bool trim = true, std::array<int, 2> preKnotRefine = {0, 0},
51 std::array<int, 2> degreeElevate = {0, 0},
52 std::array<int, 2> postKnotRefine = {0, 0}) -> std::tuple<PatchData, std::optional<PatchTrimData>> {
53 Ibra::Brep brep = readJson<dimworld>(fileName);
54 // Each surface in a brep is one Patch, as for now only one brep is supported in NURBSGrid
55 assert(brep.surfaces.size() == 1);
56
57 std::cout << "IbraReader successfully read in 1 patch with " << brep.trims.size() << " edges." << std::endl;
58
59 // Reader has done its job, now NURBSGrid can be constructed
60
61 Ibra::Surface<dimworld> _surface = brep.surfaces[0];
62 const std::array<std::vector<double>, dim> knotSpans = _surface.compileKnotVectors();
63 const std::vector<std::vector<ControlPointType>> controlPoints =
64 _surface.template transformControlPoints<ControlPointType>();
65 std::array<int, dim> dimsize = _surface.n_controlPoints;
66
67 auto controlNet = ControlPointNetType(dimsize, controlPoints);
68 PatchData _patchData{knotSpans, controlNet, _surface.degree};
69
70 // Optional preKnot refinement
71 for (const auto i : Dune::range(dim)) {
72 if (preKnotRefine[i] > 0) {
73 auto newKnots = Splines::generateRefinedKnots(knotSpans, i, preKnotRefine[i]);
74 _patchData = Splines::knotRefinement(_patchData, newKnots, i);
75 }
76 }
77
78 // Optional degree elevate
79 for (const auto i : Dune::range(dim)) {
80 if (degreeElevate[i] > 0) {
81 _patchData = Splines::degreeElevate(_patchData, i, degreeElevate[i]);
82 }
83 }
84
85 // Optional postKnot refinement
86 for (const auto i : Dune::range(dim)) {
87 if (postKnotRefine[i] > 0) {
88 auto newKnots = Splines::generateRefinedKnots(_patchData.knotSpans, i, postKnotRefine[i]);
89 _patchData = Splines::knotRefinement(_patchData, newKnots, i);
90 }
91 }
92
93 if (trim) {
94 PatchTrimData trimData{};
95 constructTrimmingCurves(brep, trimData);
96 return std::make_tuple(_patchData, trimData);
97 }
98 return std::make_tuple(_patchData, std::nullopt);
99 }
100
101private:
102 static void constructTrimmingCurves(const Ibra::Brep<dimworld>& brep, PatchTrimData& trimData) {
103 const std::vector<Ibra::BrepLoop> loops = brep.loops;
104 for (int i = 0; const Ibra::BrepLoop& loop : loops) {
105 trimData.addLoop();
106 for (const Ibra::BrepTrim& trim : loop.trims)
107 trimData.insertTrimCurve(trim.asCurve<TrimmingCurve>(), i);
108 i++;
109 }
110 }
111};
