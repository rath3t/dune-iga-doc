15inline int giveEdgeIdx(const std::size_t e1, const std::size_t e2) {
16 if ((e1 == 0 and e2 == 1) or (e1 == 1 and e2 == 0))
17 return 0;
18 if ((e1 == 1 and e2 == 2) or (e1 == 2 and e2 == 1))
19 return 1;
20 if ((e1 == 2 and e2 == 3) or (e1 == 3 and e2 == 2))
21 return 2;
22 if ((e1 == 3 and e2 == 0) or (e1 == 0 and e2 == 3))
23 return 3;
24 DUNE_THROW(Dune::GridError, "No corresponding edge Index for Input");
25}

154 void finish(const auto& patchTrimData) {
155 // Sort the points in counter clockwise manner such that the first point is in the lower left
156 const auto minVertex = lowerLeftVertex();
157 std::ranges::sort(vertices_, [&](const Vertex& aV, const Vertex& bV) { return comparePoints(aV, bV, minVertex); });
158
159 // Resort that we always start on a hostVertex (if there is one)
160 if (const auto it = std::ranges::find_if(vertices_, [](const Vertex& vV) { return vV.isHost(); });
161 it != vertices_.end())
162 std::ranges::rotate(vertices_, it);
163
164 // Now resort InsideVertices (sometimes they end up on the wrong position)
165 for (const auto i : std::views::iota(0ul, vertices_.size())) {
166 if (vertices_[i].isInside()) {
167 const auto insideVertex = vertices_[i];
168 auto it = vertices_.begin() + static_cast<long>(i);
169 const size_t indexBeforeIt =
170 it == vertices_.begin() ? vertices_.size() - 1 : std::ranges::distance(vertices_.begin(), it) - 1;
171 const size_t indexAfterIt =
172 it == std::prev(vertices_.end()) ? 0 : std::ranges::distance(vertices_.begin(), it) + 1;
173
174 // Check vertex before if they line up correctly
175 const auto vertexBefore = vertices_[indexBeforeIt];
176 const auto vertexAfter = vertices_[indexAfterIt];
177
178 if (vertexBefore.isNew() and vertexAfter.isNew()) {
179 auto indices = patchTrimData.getIndices(vertexBefore.zValue());
180 if (indices.loop == insideVertex.loop() and indices.curve == insideVertex.formerCurve())
181 continue;
182 }
183 // If we end up here sort the vertex to a correct place
184 auto correctVertexIt = findCorrespondingVertexIt(insideVertex, patchTrimData);
185 assert(correctVertexIt != vertices_.end() && "No corresponding newVertex found");
186
187 // Add the insideVertex at correctVertexIt+1
188 const auto correctItForInsideVertex = correctVertexIt + 1;
189 if (std::distance(vertices_.begin(), correctItForInsideVertex) < i) {
190 vertices_.insert(correctItForInsideVertex, insideVertex);
191 vertices_.erase(vertices_.begin() + static_cast<long>(i + 1));
192 } else {
193 std::ranges::rotate(it, it + 1, correctItForInsideVertex);
194 }
195 }
196 }
197 }

226 // Sort the points based on polar angle with respect to the reference point
227 inline static auto comparePoints = [](const Vertex& aV, const Vertex& bV, const Vertex& referenceV) -> bool {
228 auto polarAngle = [](const auto& p, const auto& reference) -> double {
229 return atan2(p.y - reference.y, p.x - reference.x);
230 };
231
232 const auto a = aV.pt;
233 const auto b = bV.pt;
234 const auto reference = referenceV.pt;
235 const double angleA = polarAngle(a, reference);
236 const double angleB = polarAngle(b, reference);
237
238 if (angleA != angleB)
239 return angleA < angleB;
240
241 // If two points have the same angle, sort based on distance to the reference point
242 return (a.x - reference.x) * (a.x - reference.x) + (a.y - reference.y) * (a.y - reference.y) <
243 (b.x - reference.x) * (b.x - reference.x) + (b.y - reference.y) * (b.y - reference.y);
244 };

246 auto findCorrespondingVertexIt(const Vertex& insideVertex, const auto& patchTrimData) {
247 return std::ranges::find_if(vertices_, [&](const Vertex& vV) {
248 if (vV.isNew()) {
249 const auto indices = patchTrimData.getIndices(vV.zValue());
250 return indices.loop == insideVertex.loop() and indices.curve == insideVertex.formerCurve();
251 } else
252 return false;
253 });
254 }

256 [[nodiscard]] const Vertex& lowerLeftVertex() const {
257 return *std::ranges::min_element(vertices_, [&](const Vertex& aV, const Vertex& bV) {
258 const auto a = aV.pt;
259 const auto b = bV.pt;
260 return (a.y < b.y) || (a.y == b.y && a.x < b.x);
261 });
262 }
