110 /* this is for testing purposes only */
111 void drawResult(const std::string& filename, bool inParameterSpace, bool newFig = true) {
112 if (flag_ == ElementTrimFlag::empty)
113 return;
114 auto eleGeometry = hostEntity_.geometry();
115 auto lowerLeftCorner = inParameterSpace ? eleGeometry.corner(0) : Dune::FieldVector<double, 2>({0, 0});
116 auto upperRightCorner = eleGeometry.corner(3);
117 Dune::DiagonalMatrix<double, 2> scaling({inParameterSpace ? 1.0 / (upperRightCorner[0] - lowerLeftCorner[0]) : 1,
118 inParameterSpace ? 1.0 / (upperRightCorner[1] - lowerLeftCorner[1]) : 1});
119
120 if (newFig)
121 auto figure = matplot::figure(true);
122 matplot::hold("on");
123
124 matplot::text(eleGeometry.center()[0], eleGeometry.center()[1], std::to_string(edges_.size()))
125 ->font("Arial")
126 .font_size(18);
127
128 constexpr std::array idxLookUp = {0, 1, 3, 2};
129
130 const double scale = eleGeometry.volume() / 5;
131
132 auto plotEllipse = [&](const Vertex& v) {
133 const auto w = 0.025 * scale;
134 const auto c = matplot::ellipse(v[0] - (w / 2), v[1] - (w / 2), w, w);
135 c->color("blue");
136 };
137
138 auto plotLine = [&](std::vector<Vertex>& vs, bool thin = false) {
139 std::vector<double> x;
140 std::ranges::transform(vs, std::back_inserter(x), [](auto& v) { return v[0]; });
141
142 std::vector<double> y;
143 std::ranges::transform(vs, std::back_inserter(y), [](auto& v) { return v[1]; });
144 if (thin)
145 matplot::plot(x, y)->line_width(0.5).color("grey");
146 else
147 matplot::plot(x, y)->line_width(2).color("red");
148
149 // Annotate
150 if (thin)
151 return;
152 matplot::text(x.front(), y.front(), "S")->font("Arial").color("blue").font_size(18);
153 };
154
155 for (auto c : edgeLookUp) {
156 Dune::FieldVector<double, 2> firstVertexPos = eleGeometry.corner(c[0]) - lowerLeftCorner;
157 Dune::FieldVector<double, 2> secondVertexPos = eleGeometry.corner(c[1]) - lowerLeftCorner;
158 scaling.mv(firstVertexPos, firstVertexPos);
159 scaling.mv(secondVertexPos, secondVertexPos);
160 std::vector<Vertex> vs{firstVertexPos, secondVertexPos};
161 plotLine(vs, true);
162 }
163
164 for (auto& v : vertices_) {
165 if (v.geometry.has_value()) {
166 Dune::FieldVector<double, 2> pos = (v.geometry.value() - lowerLeftCorner);
167 scaling.mv(pos, pos);
168 plotEllipse(pos);
169 } else {
170 Dune::FieldVector<double, 2> pos = eleGeometry.corner(idxLookUp[v.idx]) - lowerLeftCorner;
171 scaling.mv(pos, pos);
172 plotEllipse(pos);
173 }
174 }
175
176 for (auto& curve : edges_) {
177 if (curve.geometry.has_value()) {
178 std::vector<Vertex> vs;
179 for (double u : Utilities::linspace(curve.geometry.value().domain().front(), 40)) {
180 Dune::FieldVector<double, 2> pos = (curve.geometry.value().global(u) - lowerLeftCorner);
181 scaling.mv(pos, pos);
182 vs.push_back(pos);
183 }
184 plotLine(vs);
185 } else {
186 auto cornerIdx = edgeLookUp[curve.idx];
187 Dune::FieldVector<double, 2> firstVertexPos = (eleGeometry.corner(cornerIdx.front()) - lowerLeftCorner);
188 Dune::FieldVector<double, 2> secondVertexPos = (eleGeometry.corner(cornerIdx.back()) - lowerLeftCorner);
189 scaling.mv(firstVertexPos, firstVertexPos);
190 scaling.mv(secondVertexPos, secondVertexPos);
191 std::vector<Vertex> vs{firstVertexPos, secondVertexPos};
192 plotLine(vs);
193 }
194 }
195
196 matplot::axis(matplot::equal);
197 static int elleC = 0;
198 if (newFig)
199 matplot::save(filename + std::to_string(elleC++) + (inParameterSpace ? "inParameterSpace" : ""), "gif");
200 }

251private:
252 auto sampleCurves() {
253 auto eleGeo = hostEntity_.geometry();
254 auto lowerLeftCorner = eleGeo.corner(0);
255 auto upperRightCorner = eleGeo.corner(3);
256 Dune::DiagonalMatrix<double, 2> scaling(
257 {1.0 / (upperRightCorner[0] - lowerLeftCorner[0]), 1.0 / (upperRightCorner[1] - lowerLeftCorner[1])});
258
259 for (auto& edge : edges_) {
260 if (edge.geometry.has_value()) {
261 for (const auto v : Utilities::linspace(edge.geometry->domain()[0], 100)) {
262 auto fV = edge.geometry->global({v}) - lowerLeftCorner;
263 scaling.mv(fV, fV);
264 path_.emplace_back(fV[0], fV[1]);
265 }
266 } else {
267 auto cornerIdx = edgeLookUp[edge.idx];
268
269 auto firstVertex = eleGeo.corner(cornerIdx.front()) - lowerLeftCorner;
270 auto secondVertex = eleGeo.corner(cornerIdx.back()) - lowerLeftCorner;
271 scaling.mv(firstVertex, firstVertex);
272 scaling.mv(secondVertex, secondVertex);
273
274 path_.emplace_back(firstVertex[0], firstVertex[1]);
275 path_.emplace_back(secondVertex[0], secondVertex[1]);
276 }
277 }
278 }
