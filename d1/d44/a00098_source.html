41 class CurveLoopIndexEncoder
42 {
43 static constexpr int loopBits = 4;
44 static constexpr int curveBits = 8;
45 static constexpr int sampleBits = 12;
46
47 static constexpr u_int64_t loopMask{(1ULL << loopBits) - 1};
48 static constexpr u_int64_t curveMask{(1ULL << curveBits) - 1};
49 static constexpr u_int64_t sampleMask{(1ULL << sampleBits) - 1};
50
51 public:
52 static int64_t encode(int loopIndex, int curveIndex, int sampleIndex) {
53 // the left hand side shifts the curve index value away from loop values and then inserts the curve index
54 // example curveBits = 4 and loopBits = 3 and sampleBits=12;
55 // curveIndex = 3 and loopIndex = 2 = 00010_b , sampleIndex= 200= 11001000_b
56 // curveIndex = 3 = 00011_b
57 // curveIndex<< (loopBits+ sampleBits) = 110,000,000,000,000,000_b // shift the curve index 16 bits to the left
58 // loopIndex<< sampleBits = 10,000,000,000,000_b // shift the loop index 12 bits to the left
59 // sampleIndex = 11,001,000_b // shift the loop index 12 bits to the left
60 // bitwise or of the three values: 110,010,000,011,001,000_b //
61 // (curveIndex<< loopBits) | loopIndex = 11000_b | 00010_b = 11010_b = 205000_d
62 return (static_cast<u_int64_t>(curveIndex) << (loopBits + sampleBits)) |
63 (static_cast<u_int64_t>(loopIndex) << sampleBits) | sampleIndex;
64 }
65
66 struct IndexResult
67 {
68 int loop;
69 int curve;
70 int sample;
71
72 friend std::ostream& operator<<(std::ostream& os, const IndexResult& res) {
73 os << "Loop: " << res.loop << " Curve: " << res.curve << " Sample: " << res.sample;
74 return os;
75 }
76 };
77
78 static IndexResult decode(int64_t singleIndex) {
79 // since the sample index is stored in the right-most bits, we only have to apply the bitwise "and" with
80 // sampleMask, which returns simply the sample index
81 int sampleIndex = singleIndex & sampleMask;
82 // shift single index to have the loop index at the rightmost position and apply the bitwise and with the loop
83 // mask
84 const int loopAndCurveIndex = singleIndex >> sampleBits;
85 int loopIndex = loopAndCurveIndex & loopMask;
86 // shift again to have the curve index at the rightmost position and apply the bitwise and with the curve mask
87 int curveIndex = (loopAndCurveIndex >> loopBits) & curveMask;
88 return {loopIndex, curveIndex, sampleIndex};
89 }
90 };

94struct PatchTrimDataImpl

96 using TrimmingCurve = typename GridImp::GridFamily::ParameterSpaceTraits::TrimmingCurve;

100 class CurveManager
101 {
102 friend PatchTrimDataImpl;
103
104 public:
105 using idx_t = u_int64_t;
106 static constexpr idx_t indexOffSet = 5; // Since the 0 is reserved for untouched points and the following 4
107 // values are reserved for the corners of the untrimmed square all indices start at 5
108 void addLoop(const Impl::BoundaryLoop<TrimmingCurve>& loop) {
109 Clipper2Lib::PathD path;
110 // since the first
111 for (int curveIndex = 0; const auto& curve : loop.curves()) {
112 if (curve.affine() == 1 and loops_.empty()) {
113 auto p1 = curve.corner(0);
114 auto p2 = curve.corner(1);
115 path.emplace_back(p1[0], p1[1], getZValue(loops_.size(), curveIndex, 0));
116 path.emplace_back(p2[0], p2[1], getZValue(loops_.size(), curveIndex, 1));
117 } else {
118 for (int vi = 0; const auto v : Utilities::linspace(curve.domain()[0], splitter_)) {
119 auto fV = curve.global({v});
120 path.emplace_back(fV[0], fV[1], getZValue(loops_.size(), curveIndex, vi++));
121 }
122 }
123 ++curveIndex;
124 }
125 loops_.push_back(path);
126 }
127
128 [[nodiscard]] auto getIndices(const idx_t val) const -> Impl::CurveLoopIndexEncoder::IndexResult {
129 if (val == 0) // untouched point usually created by Clipperlib as Intersection point
130 return Impl::CurveLoopIndexEncoder::IndexResult{-1, -1, -1};
131 assert(val >= indexOffSet &&
132 "Passed index not decodable, since it is smaller than the offset and non-zero, which usually means that "
133 "the point is a host vertex");
134 return Impl::CurveLoopIndexEncoder::decode(val - indexOffSet);
135 }
136
137 [[nodiscard]] auto getZValue(int loopIndex, int curveIndex, int sampleIndex) const {
138 return Impl::CurveLoopIndexEncoder::encode(loopIndex, curveIndex, sampleIndex) + indexOffSet;
139 }
140
141 private:
142 Clipper2Lib::PathsD loops_;
143 idx_t splitter_{};
144 };

189 void prepare(const ParameterType& parameters, const std::array<std::vector<ctype>, 2>& coordinates) {
190 manager_.splitter_ = parameters.splitter;
191 std::ranges::for_each(loops_, [&](const auto& loop) { manager_.addLoop(loop); });
192
193 std::array domainU{coordinates[0].front(), coordinates[0].back()};
194 std::array domainV{coordinates[1].front(), coordinates[1].back()};
195
196 // Determine points of the outer boundary that are not on the edges
197 auto isInsidePatch = [&](const FieldVector<ctype, 2>& pt) {
198 const bool isInsideU = FloatCmp::gt(pt[0], domainU.front()) and FloatCmp::lt(pt[0], domainU.back());
199 const bool isInsideV = FloatCmp::gt(pt[1], domainV.front()) and FloatCmp::lt(pt[1], domainV.back());
200
201 return isInsideU and isInsideV;
202 };
203
204 for (const auto& loop : loops()) {
205 pointsInPatch_.push_back({});
206 for (size_t i = 0; const auto& curve : loop.curves()) {
207 if (const auto pt = curve.corner(1); isInsidePatch(pt)) {
208 pointsInPatch_.back().emplace_back(pt, i, (i + 1) % loop.size());
209 }
210 ++i;
211 }
212 }
213
214 finished_ = true;
215 }
216
217private:
218 std::vector<std::vector<Impl::PointInPatch<ctype>>> pointsInPatch_;
219 bool finished_ = false;
220 std::vector<Impl::BoundaryLoop<TrimmingCurve>> loops_;
221 CurveManager manager_;
222};
