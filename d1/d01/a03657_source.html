507 std::vector<ElementTrimData> trimElements(std::optional<int> level_ = std::nullopt) {
508 int level = level_.value_or(maxLevel());
509 bool initFlag = level == 0;
510 if (initFlag)
511 numBoundarySegments_ = untrimmedParameterSpaceGrid_->numBoundarySegments();
512
513 std::vector<ElementTrimData> elementTrimDatas;
514 auto gv = untrimmedParameterSpaceGrid_->levelGridView(level);
515 for (const auto& ele : elements(gv)) {
516 if (trimData_.has_value())
517 elementTrimDatas.emplace_back(trimElement(ele, gv, trimData_.value(), initFlag));
518 else
519 elementTrimDatas.emplace_back(ElementTrimFlag::full, ele);
520 }
521 return elementTrimDatas;
522 }
