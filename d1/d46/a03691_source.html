357 globalIdSet_->update();

359 // //////////////////////////////////////////
360 // Create the index sets
361 // //////////////////////////////////////////
362 for (int i = levelIndexSets_.size(); i <= maxLevel(); i++) {

364 levelIndexSets_.emplace_back(std::move(p));
365 }
366
367 for (int i = 0; i <= maxLevel(); i++)

369 levelIndexSets_[i]->update(*grid_, i);

371 leafIndexSet_->update(*grid_);
