354 void update(GridImp* grid) {
355 grid_ = grid;
356 localIdSet_->update();
357
358 globalIdSet_->update();
359
360 // //////////////////////////////////////////
361 // Create the index sets
362 // //////////////////////////////////////////
363 for (int i = levelIndexSets_.size(); i <= maxLevel(); i++) {
364 auto p = std::make_unique<LevelIndexSet>();
365 levelIndexSets_.emplace_back(std::move(p));
366 }
367
368 for (int i = 0; i <= maxLevel(); i++)
369 if (levelIndexSets_[i])
370 levelIndexSets_[i]->update(*grid_, i);
371
372 leafIndexSet_->update(*grid_);
373 }
