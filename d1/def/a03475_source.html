506 std::vector<ElementTrimData> trimElements(std::optional<int> level_ = std::nullopt) const {
507 int level = level_.value_or(maxLevel());
508 std::vector<ElementTrimData> elementTrimDatas;
509 auto gv = untrimmedParameterSpaceGrid_->levelGridView(level);
510 for (const auto& ele : elements(gv)) {
511 if (trimData_.has_value())
512 elementTrimDatas.emplace_back(trimElement(ele, trimData_.value()));
513 else
514 elementTrimDatas.emplace_back(ElementTrimFlag::full, ele);
515 }
516 return elementTrimDatas;
517 }

519 // compute the grid indices and ids
520 void update(GridImp* grid) {
521 grid_ = grid;
522 localIdSet_->update();
523
524 globalIdSet_->update();
525
526 // //////////////////////////////////////////
527 // Create the index sets
528 // //////////////////////////////////////////
529 for (int i = levelIndexSets_.size(); i <= maxLevel(); i++) {
530 auto p = std::make_unique<typename GridFamily::LevelIndexSet>();
531 levelIndexSets_.emplace_back(std::move(p));
532 }
533
534 for (int i = 0; i <= maxLevel(); i++)
535 if (levelIndexSets_[i])
536 levelIndexSets_[i]->update(*grid_, i);
537
538 leafIndexSet_->update(*grid_);
539 }
