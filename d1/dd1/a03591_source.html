293 void globalRefine(int refCount) {
294 for (int i = 0; i < refCount; ++i) {
295 const auto& finestPatchData = patchGeometries_.back().patchData();
296 auto newfinestPatchData = finestPatchData;
297
298 auto& olduniqueKnotVector = patchGeometries_.back().uniqueKnotVector();
299 std::array<std::vector<ctype>, dim> newUniqueKnotVecs;
300 for (int refDirection = 0; refDirection < dim; ++refDirection) {
301 auto additionalKnots = Splines::generateRefinedKnots(finestPatchData.knotSpans, refDirection, 1);
302 newfinestPatchData = Splines::knotRefinement<dim>(newfinestPatchData, additionalKnots, refDirection);
303 auto& newKnotVec = newUniqueKnotVecs[refDirection];
304
305 // compute new unique Knot vectors from refinement
306 newKnotVec.resize(olduniqueKnotVector[refDirection].size() + additionalKnots.size());
307 std::ranges::merge(olduniqueKnotVector[refDirection], additionalKnots, std::begin(newKnotVec));
308 }
309
310 patchGeometries_.emplace_back(newfinestPatchData, newUniqueKnotVecs);
311 patchGeometriesUnElevated.emplace_back(patchGeometries_.back());
312
313 parameterSpace_->globalRefine(1);
314 }
315 }

354 void degreeElevate(const std::array<int, dim>& elevationFactors, int lvl) {
355 if (lvl > maxLevel() and lvl >= 0)
356 DUNE_THROW(Dune : RangeError, "This level does not exist");
357 auto& patchData = patchGeometries_[lvl].patchData();
358 patchGeometriesUnElevated[lvl] = GeometryKernel::NURBSPatch<dim, dimworld, ctype>(patchData);
359 for (int dir = 0; auto elevatesInDirection : elevationFactors) {
360 if (elevatesInDirection == 0) {
361 ++dir;
362 continue;
363 }
364 patchData = Splines::degreeElevate(patchData, dir, elevatesInDirection);
365 ++dir;
366 }
367 patchGeometries_[lvl] = GeometryKernel::NURBSPatch<dim, dimworld, ctype>(patchData);
368 }
