112 /* this is for testing purposes only */
113 void drawResult(const std::string& filename, bool inParameterSpace, bool newFig = true) {
114 if (flag_ == ElementTrimFlag::empty)
115 return;
116 auto eleGeometry = hostEntity_.geometry();
117 auto lowerLeftCorner = inParameterSpace ? eleGeometry.corner(0) : Dune::FieldVector<double, 2>({0, 0});
118 auto upperRightCorner = eleGeometry.corner(3);
119 Dune::DiagonalMatrix<double, 2> scaling({inParameterSpace ? 1.0 / (upperRightCorner[0] - lowerLeftCorner[0]) : 1,
120 inParameterSpace ? 1.0 / (upperRightCorner[1] - lowerLeftCorner[1]) : 1});
121
122 if (newFig)
123 auto figure = matplot::figure(true);
124 matplot::hold("on");
125
126 matplot::text(eleGeometry.center()[0], eleGeometry.center()[1], std::to_string(edges_.size()))
127 ->font("Arial")
128 .font_size(18);
129
130 constexpr std::array idxLookUp = {0, 1, 3, 2};
131
132 const double scale = eleGeometry.volume() / 5;
133
134 auto plotEllipse = [&](const Vertex& v) {
135 const auto w = 0.025 * scale;
136 const auto c = matplot::ellipse(v[0] - (w / 2), v[1] - (w / 2), w, w);
137 c->color("blue");
138 };
139
140 auto plotLine = [&](std::vector<Vertex>& vs, bool thin = false) {
141 std::vector<double> x;
142 std::ranges::transform(vs, std::back_inserter(x), [](auto& v) { return v[0]; });
143
144 std::vector<double> y;
145 std::ranges::transform(vs, std::back_inserter(y), [](auto& v) { return v[1]; });
146 if (thin)
147 matplot::plot(x, y)->line_width(0.5).color("grey");
148 else
149 matplot::plot(x, y)->line_width(2).color("red");
150
151 // Annotate
152 if (thin)
153 return;
154 matplot::text(x.front(), y.front(), "S")->font("Arial").color("blue").font_size(18);
155 };
156
157 for (auto c : edgeLookUp) {
158 Dune::FieldVector<double, 2> firstVertexPos = eleGeometry.corner(c[0]) - lowerLeftCorner;
159 Dune::FieldVector<double, 2> secondVertexPos = eleGeometry.corner(c[1]) - lowerLeftCorner;
160 scaling.mv(firstVertexPos, firstVertexPos);
161 scaling.mv(secondVertexPos, secondVertexPos);
162 std::vector<Vertex> vs{firstVertexPos, secondVertexPos};
163 plotLine(vs, true);
164 }
165
166 for (auto& v : vertices_) {
167 if (v.geometry.has_value()) {
168 Dune::FieldVector<double, 2> pos = (v.geometry.value() - lowerLeftCorner);
169 scaling.mv(pos, pos);
170 plotEllipse(pos);
171 } else {
172 Dune::FieldVector<double, 2> pos = eleGeometry.corner(idxLookUp[v.idx]) - lowerLeftCorner;
173 scaling.mv(pos, pos);
174 plotEllipse(pos);
175 }
176 }
177
178 for (auto& curve : edges_) {
179 if (curve.geometry.has_value()) {
180 std::vector<Vertex> vs;
181 for (double u : Utilities::linspace(curve.geometry.value().domain().front(), 40)) {
182 Dune::FieldVector<double, 2> pos = (curve.geometry.value().global(u) - lowerLeftCorner);
183 scaling.mv(pos, pos);
184 vs.push_back(pos);
185 }
186 plotLine(vs);
187 } else {
188 auto cornerIdx = edgeLookUp[curve.idx];
189 Dune::FieldVector<double, 2> firstVertexPos = (eleGeometry.corner(cornerIdx.front()) - lowerLeftCorner);
190 Dune::FieldVector<double, 2> secondVertexPos = (eleGeometry.corner(cornerIdx.back()) - lowerLeftCorner);
191 scaling.mv(firstVertexPos, firstVertexPos);
192 scaling.mv(secondVertexPos, secondVertexPos);
193 std::vector<Vertex> vs{firstVertexPos, secondVertexPos};
194 plotLine(vs);
195 }
196 }
197
198 matplot::axis(matplot::equal);
199 static int elleC = 0;
200 if (newFig)
201 matplot::save(filename + std::to_string(elleC++) + (inParameterSpace ? "inParameterSpace" : ""), "gif");
202 }

256private:
257 auto sampleCurves() {
258 auto eleGeo = hostEntity_.geometry();
259 auto lowerLeftCorner = eleGeo.corner(0);
260 auto upperRightCorner = eleGeo.corner(3);
261 Dune::DiagonalMatrix<double, 2> scaling(
262 {1.0 / (upperRightCorner[0] - lowerLeftCorner[0]), 1.0 / (upperRightCorner[1] - lowerLeftCorner[1])});
263
264 for (auto& edge : edges_) {
265 if (edge.geometry.has_value()) {
266 for (const auto v : Utilities::linspace(edge.geometry->domain()[0], 100)) {
267 auto fV = edge.geometry->global({v}) - lowerLeftCorner;
268 scaling.mv(fV, fV);
269 path_.emplace_back(fV[0], fV[1]);
270 }
271 } else {
272 auto cornerIdx = edgeLookUp[edge.idx];
273
274 auto firstVertex = eleGeo.corner(cornerIdx.front()) - lowerLeftCorner;
275 auto secondVertex = eleGeo.corner(cornerIdx.back()) - lowerLeftCorner;
276 scaling.mv(firstVertex, firstVertex);
277 scaling.mv(secondVertex, secondVertex);
278
279 path_.emplace_back(firstVertex[0], firstVertex[1]);
280 path_.emplace_back(secondVertex[0], secondVertex[1]);
281 }
282 }
283 }
