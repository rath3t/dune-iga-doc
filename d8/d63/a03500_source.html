291 void globalRefine(int refCount) {
292 for (int i = 0; i < refCount; ++i) {
293 const auto& finestPatchData = patchGeometries_.back().patchData();
294 auto newfinestPatchData = finestPatchData;
295
296 auto& olduniqueKnotVector = patchGeometries_.back().uniqueKnotVector();
297 std::array<std::vector<ctype>, dim> newUniqueKnotVecs;
298 for (int refDirection = 0; refDirection < dim; ++refDirection) {
299 auto additionalKnots = Splines::generateRefinedKnots(finestPatchData.knotSpans, refDirection, 1);
300 newfinestPatchData = Splines::knotRefinement<dim>(newfinestPatchData, additionalKnots, refDirection);
301 auto& newKnotVec = newUniqueKnotVecs[refDirection];
302
303 // compute new unique Knot vectors from refinement
304 newKnotVec.resize(olduniqueKnotVector[refDirection].size() + additionalKnots.size());
305 std::ranges::merge(olduniqueKnotVector[refDirection], additionalKnots, std::begin(newKnotVec));
306 }
307
308 patchGeometries_.emplace_back(newfinestPatchData, newUniqueKnotVecs);
309 patchGeometriesUnElevated.emplace_back(patchGeometries_.back());
310 }
311 // Here it is exploited that the knot refinem above is in sync with the globaRefine of the trimmer
312 trimmer_->globalRefine(refCount);
313 }

330 void globalRefineInDirection(const std::array<int, dim>& refines) {
331 const auto& finestPatchData = patchGeometries_.back().patchData();
332 auto newfinestPatchData = finestPatchData;
333 for (int dir = 0; auto refinesInDirection : refines) {
334 if (refinesInDirection == 0) {
335 ++dir;
336 continue;
337 }
338 auto additionalKnots = Splines::generateRefinedKnots(finestPatchData.knotSpans, dir, refinesInDirection);
339 newfinestPatchData = Splines::knotRefinement<dim>(newfinestPatchData, additionalKnots, dir);
340 ++dir;
341 }
342 auto newGrid = PatchGrid(newfinestPatchData);
343 *this = std::move(newGrid);
344 }

351 void degreeElevate(const std::array<int, dim>& elevationFactors, int lvl) {
352 if (lvl > maxLevel() and lvl >= 0)
353 DUNE_THROW(Dune : RangeError, "This level does not exist");
354 auto& patchData = patchGeometries_[lvl].patchData();
355 patchGeometriesUnElevated[lvl] = GeometryKernel::NURBSPatch<dim, dimworld, ctype>(patchData);
356 for (int dir = 0; auto elevatesInDirection : elevationFactors) {
357 if (elevatesInDirection == 0) {
358 ++dir;
359 continue;
360 }
361 patchData = Splines::degreeElevate(patchData, dir, elevatesInDirection);
362 ++dir;
363 }
364 patchGeometries_[lvl] = GeometryKernel::NURBSPatch<dim, dimworld, ctype>(patchData);
365 }
