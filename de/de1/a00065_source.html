25auto findGoodStartingPoint(const auto& curve, const Clipper2Lib::PointD& pt) -> double {
26 int N = std::max(curve.numberOfControlPoints().front() * 15, 200);
27 auto linSpace = Utilities::linspace(curve.domain().front(), N);
28 auto distances = std::ranges::transform_view(linSpace, [&](const auto u) { return distance(pt, curve.global(u)); });
29 auto min_idx = std::ranges::distance(distances.begin(), std::ranges::min_element(distances));
30 return linSpace[min_idx];
31}

51auto callFindIntersection(const auto& curvePatchGeo, int edgeIdx, const auto& ip,
52 const auto& corners) -> std::pair<double, FieldVector<double, 2>> {
53 auto pos = corners[edgeIdx];
54 auto dir = edgeDirections[edgeIdx];
55
56 double lineGuess = (edgeIdx == 0 or edgeIdx == 2) ? (ip.x - pos[0]) / dir[0] : (ip.y - pos[1]) / dir[1];
57 // Todo use z-Value to get a good starting point, not brute-force
58 auto guessTParam = FieldVector<double, 2>{findGoodStartingPoint(curvePatchGeo, ip), lineGuess};
59
60 auto [success, tParam, curvePoint] = findIntersectionCurveAndLine(curvePatchGeo, pos, dir, guessTParam);
61 if (success == IntersectionCurveAndLine::intersect)
62 return std::make_pair(tParam[0], curvePoint);
63 if (success == IntersectionCurveAndLine::parallel) {
64 DUNE_THROW(Dune::GridError, "Couldn't find intersection Point, lines are parallel");
65 }
66
67 auto domain = curvePatchGeo.domain();
68 if (FloatCmp::eq(curvePoint, curvePatchGeo.global({domain[0].front()})))
69 return std::make_pair(domain[0][0], curvePoint);
70 if (FloatCmp::eq(curvePoint, curvePatchGeo.global({domain[0].back()})))
71 return std::make_pair(domain[0].back(), curvePoint);
72
73 DUNE_THROW(Dune::GridError, "Couldn't find intersection Point");
74};
