13template <int dimworld, typename InputStringType>
14requires(not std::convertible_to<std::string, InputStringType> and
15 not std::convertible_to<InputStringType, const char*>)
16auto readJson(InputStringType& ibraInputFile) {
17 using json = nlohmann::json;
18
19 std::vector<Ibra::Surface<dimworld>> surfaces;
20 std::vector<Ibra::Curve2D> curves2D;
21
22 std::vector<Ibra::BrepRepresentation> brepRepresentations;
23 std::vector<Ibra::BrepLoopRepresentation> brepLoopRepresentations;
24 std::vector<Ibra::BrepTrimRepresentation> brepTrimRepresentations;
25
26 json ibraJson;
27 try {
28 ibraJson = json::parse(ibraInputFile);
29
30 for (auto& j : ibraJson) {
31 auto geo = j.get<Ibra::IbraBase>();
32
33 switch (geo.type) {
34 case Ibra::Type::NurbsSurfaceGeometry3D:
35 surfaces.push_back(j.get<Ibra::Surface<dimworld>>());
36 break;
37 case Ibra::Type::NurbsCurveGeometry2D:
38 curves2D.push_back(j.get<Ibra::Curve2D>());
39 break;
40 case Ibra::Type::BrepType:
41 brepRepresentations.push_back(j.get<Ibra::BrepRepresentation>());
42 break;
43 case Ibra::Type::BrepLoopType:
44 brepLoopRepresentations.push_back(j.get<Ibra::BrepLoopRepresentation>());
45 break;
46 case Ibra::Type::BrepTrimType:
47 brepTrimRepresentations.push_back(j.get<Ibra::BrepTrimRepresentation>());
48 break;
49 default:
50 // Do nothing -- keep the compiler happy
51 break;
52 }
53 }
54 } catch (json::parse_error& ex) {
55 DUNE_THROW(Dune::IOError, "Error in reading input stream: "
56 << ", parse error at byte " << ex.byte << " What: " << ex.what());
57 }
58
59 // Make Connections
60 auto brepRepresentation = brepRepresentations[0];
61
62 Ibra::Brep brep{brepRepresentation, curves2D, surfaces, brepLoopRepresentations, brepTrimRepresentations};
63 return brep;
64}
65template <int dimworld>
66auto readJson(const std::string& fileName) {
67 std::ifstream ibraInputFile;
68 ibraInputFile.open(fileName);
69 return readJson<dimworld>(ibraInputFile);
70}
