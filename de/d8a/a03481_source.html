357 void update(GridImp* grid) {
358 grid_ = grid;
359 localIdSet_->update();
360
361 globalIdSet_->update();
362
363 // //////////////////////////////////////////
364 // Create the index sets
365 // //////////////////////////////////////////
366 for (int i = levelIndexSets_.size(); i <= maxLevel(); i++) {
367 auto p = std::make_unique<LevelIndexSet>();
368 levelIndexSets_.emplace_back(std::move(p));
369 }
370
371 for (int i = 0; i <= maxLevel(); i++)
372 if (levelIndexSets_[i])
373 levelIndexSets_[i]->update(*grid_, i);
374
375 leafIndexSet_->update(*grid_);
376 }
