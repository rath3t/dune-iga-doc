26 template <int dim>
27 auto& voigtIndices() {
28 if constexpr (dim == 1) {
29 static std::array<IndexPair, dim*(dim + 1) / 2> voigt1D = {
30 {0, 0}
31 };
32 return voigt1D;
33 } else if constexpr (dim == 2) {
34 static std::array<IndexPair, dim*(dim + 1) / 2> voigt2D = {
35 {{0, 0}, {1, 1}, {0, 1}}
36 };
37 return voigt2D;
38 } else if constexpr (dim == 3) {
39 static std::array<IndexPair, dim*(dim + 1) / 2> voigt3D = {
40 {{0, 0}, {1, 1}, {2, 2}, {1, 2}, {0, 2}, {0, 1}}
41 };
42 return voigt3D;
43 }

87 // if we have codim==0, then the Jacobian in the parameter space of the grid entity itself is a DiagonalMatrix,
88 // and
89 // Coordinates in a single knot span differ from coordinates on the B-spline patch
90 // by an affine transformation. This transformation is stored in the diagonal entries.
91 // If trimming is disabled the Jacobian in the parameter space of subentities (edges or surfaces) is a
92 // DiagonalMatrixBlock but is treated as a FieldMatrix, since is also just cubes and an axis-aligned geometry, if
93 // trimming is enabled the Jacobian of the parameterspace for subentities is a potentially fully populated
94 // FieldMatrix, Think about an arbitrary curve in the axis-aligned cube parameter grid
95 // ------->
96 // `____________________C_______D_______`
97 // ::````````````````::``````````'|````::
98 // :: :: `| ::
99 // :: ^F:: `| ::
100 // :: | :: `| ::
101 // :: | :: `| ::
102 // :: |E:: `| ::
103 // :: :: .| x ::
104 // :: :: _: ::
105 // ::................::..........|:....::
106 // ::''''''''''''''''::''''''''''/'''''::
107 // :: :: _:B ::
108 // :: :: .| ::
109 // :: :: `::A ::
110 // :: '::::||::::::' ::
111 // :: ::` :: x ::
112 // :: |' :: ::
113 // :: | x :: ::
114 // ::````````|```````::````````````````::
115 // `____________________________________`
116 // at point A the tangent of the curve (the JacobianInParameterSpace) points into the oblique direction (to B)
117 // for the axis-aligned geometry these tangents are from E to F or from C to D, depending on the orientation of
118 // the subentity, which are aligned with the axis
119 using JacobianTransposedInParameterSpace = typename ParameterSpaceGeometry::JacobianTransposed;
120 using GlobalInParameterSpace = typename ParameterSpaceGeometry::GlobalCoordinate;

212 [[nodiscard]] double volume() const {
213 int order = mydimension * *std::ranges::max_element(patchGeometry_->patchData_.degree);
214
215 auto rule = [&]() {
216 if constexpr (ParameterSpace::isAlwaysTrivial or mydimension != 2)
217 return QuadratureRules<ctype, mydimension>::rule(GeometryTypes::cube(mydimension), order);
218 else {
219 return parameterSpaceGeometry->getQuadratureRule(order);
220 }
221 }();
222
223 Volume vol = 0.0;
224 for (auto& gp : rule)
225 vol += integrationElement(gp.position()) * gp.weight();
226 return vol;
227 }

335 Hessian hessian(const LocalCoordinate& local) const {
336 checkState();
337 const auto ouInPatch = globalInParameterSpace(local);
338
339 const auto [p, dfdg, dfdgdg] =
340 patchGeometry_->zeroFirstAndSecondDerivativeOfPositionImpl(ouInPatch, nurbsLocalView_, localControlPointNet);
341
342 const JacobianTransposedInParameterSpace dgdt = jacobianTransposedInParameterSpace(local);
343 static_assert(JacobianTransposedInParameterSpace::rows == mydimension);
344 static_assert(JacobianTransposedInParameterSpace::cols == gridDimension);
345 // dgdt (1x2, curve on surface), (2x2 surface in surface), (2x3 surface in 3D), (3x3 volume in 3D)
346 FieldMatrix<ctype, patchNumberOfSecondDerivatives, numberOfSecondDerivatives> dgdtSquared(0);
347 for (int patchIndex = 0; auto [patchRow, patchCol] : Impl::voigtIndices<gridDimension>()) {
348 for (int myIndex = 0; auto [myRow, myCol] : Impl::voigtIndices<mydimension>()) {
349 if constexpr (not std::is_same_v<JacobianTransposedInParameterSpace, DiagonalMatrix<ctype, gridDimension>>)
350 dgdtSquared[patchIndex][myIndex] =
351 (dgdt[myRow][patchRow] * dgdt[myCol][patchCol] + dgdt[myCol][patchRow] * dgdt[myRow][patchCol]) *
352 ((patchRow == patchCol) ? 0.5 : 1.0);
353 else {
354 const auto dgdtmyRowpatchRow = (myRow == patchRow) ? dgdt[myRow][patchRow] : 0;
355 const auto dgdtmyColpatchCol = (myCol == patchCol) ? dgdt[myCol][patchCol] : 0;
356 const auto dgdtmyColpatchRow = (myCol == patchRow) ? dgdt[myCol][patchRow] : 0;
357 const auto dgdtmyRowpatchCol = (myRow == patchCol) ? dgdt[myRow][patchCol] : 0;
358 dgdtSquared[patchIndex][myIndex] =
359 (dgdtmyRowpatchRow * dgdtmyColpatchCol + dgdtmyColpatchRow * dgdtmyRowpatchCol) *
360 ((patchRow == patchCol) ? 0.5 : 1.0);
361 }
362 ++myIndex;
363 }
364 ++patchIndex;
365 }
366
367 // h = mydimension * (mydimension + 1) / 2 X worlddimension = 1x2 (curve in 2D), 1x3 (curve in 3D), (3x2 surface
368 // in 2D), (3x3 surface in 3D), (6x3 volume in 3D)
369 //
370 // dfdgdg = gridDimension*(gridDimension + 1) / 2 X worlddimension
371 // = 3x2 (curve on surface flat),3x3 (curve on surface in 3D), 6x3 (curve in volume), 6x3 (surface in
372 // volume),(3x3 surface in surface in 3D)
373 //
374 // dgdtSquared = gridDimension * (gridDimension + 1) / 2 X mydimension * (mydimension +
375 // 1) / 2 = 3x1 (curve on surface), 6x1 (curve in volume), 6x3 (surface in volume), 3x3 ( surface in surface),
376 // 6x6 ( volume in volume)
377 Hessian h = transposedView(dgdtSquared) * dfdgdg;
378
379 /* if trimming is enabled the parameter space geometry is potentially non-linear,
380 * the resulting Hessian has another contribution due to chain-rule, namely the second derivative of g */
381 if constexpr (not ParameterSpace::template isLocalGeometryLinear<codim>) {
382 // const auto dgdtdt = parameterSpaceGeometry->hessian(ouInPatch);
383 //
384 // assert(ParameterSpaceType::template isLocalGeometryLinear<
385 // codim> && "This can not be checked yet. Check if this works with trimming and then remove
386 // assert");
387 // h += transpose(transposedView(dfdg) * dgdtdt);
388 }
389 return h;
390 }
