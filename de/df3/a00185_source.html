144 [[nodiscard]] double volume() const {
145 auto dom = domain();
146 double volParamterSpace = 1.0;
147 for (int i = 0; i < mydimension; ++i)
148 volParamterSpace *= dom[i].volume();
149 const auto rule = Dune::QuadratureRules<ctype, mydimension>::rule(
150 this->type(), mydimension * (*std::ranges::max_element(patchData_.degree)));
151 ctype vol = 0.0;
152 const std::array<Utilities::Domain<double>, mydimension> input{};
153 for (auto& gp : rule) {
154 vol += integrationElement(Utilities::mapToRange(gp.position(), input, domain())) * gp.weight() * volParamterSpace;
155 }
156 return vol;
157 }
161 [[nodiscard]] double curveLength() const
162 requires(mydimension == 1)
163 {
164 if (affine())
165 return (corner(1) - corner(0)).two_norm();
166
167 auto dom = domain();
168 double volParamterSpace = dom[0].volume();
169
170 const auto rule =
171 Dune::QuadratureRules<ctype, mydimension>::rule(this->type(), 4 * *std::ranges::max_element(patchData_.degree));
172 ctype vol = 0.0;
173 const std::array<Utilities::Domain<double>, mydimension> input{};
174 for (auto& gp : rule) {
175 vol += integrationElement(Utilities::mapToRange(gp.position(), input, domain())) * gp.weight() * volParamterSpace;
176 }
177 return vol;
178 }

231 b[0][0] = secDerivatives[0] * unitnormal;

413 ConnectionResult isConnectedAtBoundary(int boundary, double tol = 1e-8) const {
414 const auto& cps = patchData_.controlPoints;
415 if constexpr (mydimension == 1) {
416 if (auto cp = cps.directGetAll(); FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(
417 cp.front().p, cp.back().p, tol))
418 return ConnectionResult(true, !boundary);
419 else
420 return ConnectionResult(false, -1);
421 } else if constexpr (mydimension == 2) {
422 auto edgeRange = [&cps](int b) {
423 switch (b) {
424 case 0:
425 return cps.hyperSurfFront({1});
426 case 1:
427 return cps.hyperSurfBack({1});
428 case 2:
429 return cps.hyperSurfFront({0});
430 case 3:
431 return cps.hyperSurfBack({0});
432 default:
433 assert("No valid boundary index");
434 }
435 __builtin_unreachable();
436 };
437
438 auto theEdgeToCheck = edgeRange(boundary);
439 for (auto i : Dune::range(4)) {
440 if (boundary == i)
441 continue;
442 if (std::ranges::equal(theEdgeToCheck, edgeRange(i), [tol](auto f, auto s) {
443 return FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(f.p, s.p, tol);
444 }))
445 return ConnectionResult(true, i);
446 }
447 return ConnectionResult(false, -1);
448 } else if constexpr (mydimension == 3) {
449 auto edgeRange = [&cps](int b) {
450 switch (b) {
451 case 0:
452 return cps.hyperSurfFront({1, 2});
453 case 1:
454 return cps.hyperSurfBack({1, 2});
455 case 2:
456 return cps.hyperSurfFront({0, 1});
457 case 3:
458 return cps.hyperSurfBack({0, 1});
459 case 4:
460 return cps.hyperSurfFront({0, 2});
461 case 5:
462 return cps.hyperSurfBack({0, 2});
463 default:
464 assert("No valid boundary index");
465 }
466 __builtin_unreachable();
467 };
468
469 auto theEdgeToCheck = edgeRange(boundary);
470 for (auto i : Dune::range(6)) {
471 if (boundary == i)
472 continue;
473 if (std::ranges::equal(theEdgeToCheck, edgeRange(i), [tol](auto f, auto s) {
474 return FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(f.p, s.p, tol);
475 }))
476 return ConnectionResult(true, i);
477 }
478 return ConnectionResult(false, -1);
479 } else
480 DUNE_THROW(Dune::NotImplemented,
481 "Higher dimensions not implemented yet. But should be straight-forward to do so, somebody just has "
482 "to come up with a resenable indexing of hyper surfaces.");
483 }

490 [[nodiscard]] GlobalCoordinate unitNormal(const LocalCoordinate& local) const
491 requires(mydimension == 2) && (coorddimension == 3)
492 {
493 auto N = normal(local);

497 [[nodiscard]] GlobalCoordinate normal(const LocalCoordinate& local) const
498 requires(mydimension == 2) && (coorddimension == 3)
499 {

504 /* @brief Calculate NURBS and control point net for a given local coordinate. */
505 auto calculateNurbsAndControlPointNet(const LocalCoordinate& u) const {
506 auto subNetStart = Splines::findSpan(patchData_.degree, u, patchData_.knotSpans);
507 auto cpCoordinateNet = Splines::netOfSpan(subNetStart, patchData_.degree,
508 Splines::extractControlCoordinates(patchData_.controlPoints));
509 auto nurbsLocalView = nurbs_.localView();
510 nurbsLocalView.bind(subNetStart);
511 return std::make_tuple(nurbsLocalView, cpCoordinateNet, subNetStart);
512 }

514 /* @brief Calculate zero first and second derivatives of the position for a given local coordinate. */
515 static auto zeroFirstAndSecondDerivativeOfPositionImpl(const LocalCoordinate& u, const NurbsLocalView& nurbsLocalView,
516 const ControlPointCoordinateNetType& localControlPointNet) {
517 return std::make_tuple(GeometryKernel::position(u, nurbsLocalView, localControlPointNet),
518 GeometryKernel::jacobianTransposed(u, nurbsLocalView, localControlPointNet),
519 GeometryKernel::hessian(u, nurbsLocalView, localControlPointNet));
520 }
