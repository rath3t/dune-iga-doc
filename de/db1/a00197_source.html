133 [[nodiscard]] double volume() const {
134 auto dom = domain();
135 double volParamterSpace = 1.0;
136 for (int i = 0; i < mydimension; ++i)
137 volParamterSpace *= dom[i].volume();
138 const auto rule = Dune::QuadratureRules<ctype, mydimension>::rule(
139 this->type(), mydimension * (*std::ranges::max_element(patchData_.degree)));
140 ctype vol = 0.0;
141 const std::array<Utilities::Domain<double>, mydimension> input{};
142 for (auto& gp : rule) {
143 vol += integrationElement(Utilities::mapToRange(gp.position(), input, domain())) * gp.weight() * volParamterSpace;
144 }
145 return vol;
146 }
150 [[nodiscard]] double curveLength() const
151 requires(mydimension == 1)
152 {
153 if (affine())
154 return (corner(1) - corner(0)).two_norm();
155
156 auto dom = domain();
157 double volParamterSpace = dom[0].volume();
158
159 const auto rule =
160 Dune::QuadratureRules<ctype, mydimension>::rule(this->type(), 4 * *std::ranges::max_element(patchData_.degree));
161 ctype vol = 0.0;
162 const std::array<Utilities::Domain<double>, mydimension> input{};
163 for (auto& gp : rule) {
164 vol += integrationElement(Utilities::mapToRange(gp.position(), input, domain())) * gp.weight() * volParamterSpace;
165 }
166 return vol;
167 }

368 ConnectionResult isConnectedAtBoundary(int boundary, double tol = 1e-8) const {
369 const auto& cps = patchData_.controlPoints;
370 if constexpr (mydimension == 1) {
371 if (auto cp = cps.directGetAll(); FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(
372 cp.front().p, cp.back().p, tol))
373 return ConnectionResult(true, !boundary);
374 else
375 return ConnectionResult(false, -1);
376 } else if constexpr (mydimension == 2) {
377 auto edgeRange = [&cps](int b) {
378 switch (b) {
379 case 0:
380 return cps.hyperSurfFront({1});
381 case 1:
382 return cps.hyperSurfBack({1});
383 case 2:
384 return cps.hyperSurfFront({0});
385 case 3:
386 return cps.hyperSurfBack({0});
387 default:
388 assert("No valid boundary index");
389 }
390 __builtin_unreachable();
391 };
392
393 auto theEdgeToCheck = edgeRange(boundary);
394 for (auto i : Dune::range(4)) {
395 if (boundary == i)
396 continue;
397 if (std::ranges::equal(theEdgeToCheck, edgeRange(i), [tol](auto f, auto s) {
398 return FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(f.p, s.p, tol);
399 }))
400 return ConnectionResult(true, i);
401 }
402 return ConnectionResult(false, -1);
403 } else if constexpr (mydimension == 3) {
404 auto edgeRange = [&cps](int b) {
405 switch (b) {
406 case 0:
407 return cps.hyperSurfFront({1, 2});
408 case 1:
409 return cps.hyperSurfBack({1, 2});
410 case 2:
411 return cps.hyperSurfFront({0, 1});
412 case 3:
413 return cps.hyperSurfBack({0, 1});
414 case 4:
415 return cps.hyperSurfFront({0, 2});
416 case 5:
417 return cps.hyperSurfBack({0, 2});
418 default:
419 assert("No valid boundary index");
420 }
421 __builtin_unreachable();
422 };
423
424 auto theEdgeToCheck = edgeRange(boundary);
425 for (auto i : Dune::range(6)) {
426 if (boundary == i)
427 continue;
428 if (std::ranges::equal(theEdgeToCheck, edgeRange(i), [tol](auto f, auto s) {
429 return FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(f.p, s.p, tol);
430 }))
431 return ConnectionResult(true, i);
432 }
433 return ConnectionResult(false, -1);
434 } else
435 DUNE_THROW(Dune::NotImplemented,
436 "Higher dimensions not implemented yet. But should be straight-forward to do so, somebody just has "
437 "to come up with a resenable indexing of hyper surfaces.");
438 }

445 /* @brief Calculate NURBS and control point net for a given local coordinate. */
446 auto calculateNurbsAndControlPointNet(const LocalCoordinate& u) const {
447 auto subNetStart = Splines::findSpan(patchData_.degree, u, patchData_.knotSpans);
448 auto cpCoordinateNet = Splines::netOfSpan(subNetStart, patchData_.degree,
449 Splines::extractControlCoordinates(patchData_.controlPoints));
450 auto nurbsLocalView = nurbs_.localView();
451 nurbsLocalView.bind(subNetStart);
452 return std::make_tuple(nurbsLocalView, cpCoordinateNet, subNetStart);
453 }
454
455 /* @brief Calculate zero first and second derivatives of the position for a given local coordinate. */
456 static auto zeroFirstAndSecondDerivativeOfPositionImpl(const LocalCoordinate& u, const NurbsLocalView& nurbsLocalView,
457 const ControlPointCoordinateNetType& localControlPointNet) {
458 return std::make_tuple(GeometryKernel::position(u, nurbsLocalView, localControlPointNet),
459 GeometryKernel::jacobianTransposed(u, nurbsLocalView, localControlPointNet),
460 GeometryKernel::hessian(u, nurbsLocalView, localControlPointNet));
461 }
