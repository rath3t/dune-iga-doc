18 template <std::integral auto dim>
19 auto ordersFromDegrees(const std::array<int, dim>& degree) {
20 std::array<int, dim> order;
21 std::ranges::transform(degree, order.begin(), [](const auto& p) { return p + 1; });
22 return order;

25 template <std::integral auto dim>
26 void createPartialSubDerivativPermutations(const FieldVector<int, dim>& v, std::vector<FieldVector<int, dim>>& perm) {
27 perm.resize(0);
28 for (int i = 1; i < std::pow(2, v.size()); ++i) {
29 FieldVector<int, dim> x{};
30 for (int j = 0; j < v.size(); ++j)
31 if ((i & (1 << j)) != 0) {
32 x[j] = v[j];
33 if (v[j] == 0)
34 goto outer;
35 }
36 perm.push_back(x);
37 outer:;
38 }
39 }

42 int binom(const VectorType& n, const VectorType& k) {
43 return std::inner_product(n.begin(), n.end(), k.begin(), 1, std::multiplies{},
44 [](auto& ni, auto& ki) { return Dune::binomial(ni, ki); });

63template <std::floating_point ScalarType, std::integral auto dim>
64auto generateRefinedKnots(const std::array<std::vector<ScalarType>, dim>& knotSpans, const int dir,
65 const int refinementLevel) {
66 const int newKnotsSizeForEachSpan = Dune::power(2, refinementLevel);
67 auto unique_Knots = knotSpans;
68 std::vector<double> additionalKnots;
69 auto& unique_KnotPerDim = unique_Knots[dir];
70 unique_KnotPerDim.erase(std::ranges::begin(std::ranges::unique(unique_KnotPerDim)), std::end(unique_KnotPerDim));
71
72 for (int i = 0; i < unique_KnotPerDim.size() - 1; ++i) {
73 const double spanLength = unique_KnotPerDim[i + 1] - unique_KnotPerDim[i];
74 const double increment = spanLength / newKnotsSizeForEachSpan;
75 for (int j = 1; j < newKnotsSizeForEachSpan; ++j) {
76 additionalKnots.emplace_back(unique_KnotPerDim[i] + increment * j);
77 }
78 }
79 return additionalKnots;
80}

94template <std::integral auto dim, std::integral auto dim2, typename NetValueType>
95requires((std::floating_point<NetValueType> || Concept::Vector<NetValueType> ||
96 is_instantiation_of<ControlPoint, NetValueType>::value) and
97 (dim == dim2))
98auto netOfSpan(std::array<int, dim> subNetStart, const std::array<int, dim>& degree,
99 const MultiDimensionalNet<dim2, NetValueType>& net) {
100 std::array<int, dim> order = Impl::ordersFromDegrees(degree);
101 for (std::size_t i = 0; i < dim; ++i)
102 subNetStart[i] -= degree[i];
103 return net.subNet(subNetStart, order);
104}

311 static auto basisFunctions(const FieldVector<ScalarType, dim>& u,
312 const std::array<std::vector<ScalarType>, dim>& knots, const std::array<int, dim>& degree,
313 const MultiDimensionalNet<dim, ScalarType>& weights, std::array<int, dim> spIndex) {
314 std::array<DynamicVectorType, (size_t)dim> bSplines;
315
316 for (std::size_t i = 0; i < dim; ++i)
317 bSplines[i] = BsplineBasis<ScalarType>::basisFunctions(u[i], knots[i], degree[i], spIndex[i]);
318
319 auto Nnet = MultiDimensionalNet<dim, ScalarType>(bSplines);
320 // @todo Alex this should also be cached when called with a localView
321 const auto subNetWeights = netOfSpan(spIndex, degree, weights);
322
323 const ScalarType invSumWeight = dot(Nnet, subNetWeights);
324 Nnet *= subNetWeights;
325 Nnet /= invSumWeight;
326 return Nnet;
327 }

339 static auto basisFunctionDerivatives(const FieldVector<ScalarType, dim>& u,
340 const std::array<std::vector<ScalarType>, dim>& knots,
341 const std::array<int, dim>& degree,
342 const MultiDimensionalNet<dim, double>& weights, const int derivativeOrder,
343 std::optional<std::array<int, dim>> spIndex = std::nullopt) {
344 std::array<DynamicMatrixType, dim> bSplineDerivatives;
345 std::array<int, dim> spanIndex = spIndex ? spIndex.value() : findSpan(degree, u, knots);
346
347 for (int i = 0; i < dim; ++i)
348 bSplineDerivatives[i] =
349 BsplineBasis<ScalarType>::basisFunctionDerivatives(u[i], knots[i], degree[i], derivativeOrder, spanIndex[i]);
350
351 std::array<std::vector<ScalarType>, dim> dimArrayOfVectors;
352 FieldVector<int, dim> dimSize(derivativeOrder + 1);
353 MultiDimensionalNet<dim, MultiDimensionalNet<dim, ScalarType>> netOfDerivativeNets(dimSize);
354
355 for (int j = 0; auto& derivNet : netOfDerivativeNets.directGetAll()) {
356 auto multiIndex = netOfDerivativeNets.directToMultiIndex(j++);
357 for (int i = 0; i < multiIndex.size(); ++i)
358 dimArrayOfVectors[i] = bSplineDerivatives[i][multiIndex[i]].container();
359 derivNet = MultiDimensionalNet<dim, ScalarType>(dimArrayOfVectors);
360 }
361
362 MultiDimensionalNet<dim, MultiDimensionalNet<dim, ScalarType>> R = netOfDerivativeNets;
363 MultiDimensionalNet<dim, ScalarType> netsOfWeightfunctions(dimSize);
364 const auto subNetWeights = netOfSpan(spanIndex, degree, weights);
365
366 for (int j = 0; j < R.size(); ++j) {
367 R.directGet(j) *= subNetWeights;
368 netsOfWeightfunctions.directGet(j) = dot(netOfDerivativeNets.directGet(j), subNetWeights);
369 }
370
371 std::vector<FieldVector<int, dim>> perms;
372 // The following is inefficient, it computes too much derivative directions and recycles nothing
373 for (int j = 0; j < R.size(); ++j) {
374 const auto derivOrders = R.template directToMultiIndex<FieldVector<int, dim>>(j);
375 Impl::createPartialSubDerivativPermutations(derivOrders, perms);
376
377 for (const auto& perm : perms) {
378 const MultiDimensionalNetIndex<dim> kNet(perm + FieldVector<int, dim>(1));
379 auto startMultiIndex = perm;
380 std::ranges::transform(startMultiIndex, startMultiIndex.begin(), [](const auto& v) { return (v != 0); });
381 for (int kk = kNet.index(startMultiIndex); kk < kNet.size(); ++kk) {
382 const auto multik = kNet.template directToMultiIndex<FieldVector<int, dim>>(kk);
383 const ScalarType fac = (Impl::binom(perm, multik) * netsOfWeightfunctions.get(multik));
384 const auto& Rdirect_d = R.get(derivOrders - multik);
385 auto& Rdirect = R.directGet(j);
386 for (int i = 0; i < R.directGet(j).size(); ++i)
387 Rdirect.directGet(i) -= fac * Rdirect_d.directGet(i); // generalized Piegl Tiller (4.20)
388 }
389 }
390 R.directGet(j) /= netsOfWeightfunctions.directGet(0);
391 }
392 return R;
393 }

444template <std::integral auto dim, std::integral auto dimworld, typename ScalarType>
445auto degreeElevate(const NURBSPatchData<dim, dimworld, ScalarType>& oldData, const int elevationDirection,
446 const int elevationFactor) {
447 using DynamicMatrixType = DynamicMatrix<ScalarType>;
448 using NURBSPatchData = NURBSPatchData<dim, dimworld, ScalarType>;
449 using ControlPointNetType = typename NURBSPatchData::ControlPointNetType;
450 using ControlPointType = typename NURBSPatchData::ControlPointType;
451 const int t = elevationFactor;
452 const int p = oldData.degree[elevationDirection];
453 const int m = oldData.controlPoints.strideSizes()[elevationDirection] + p;
454 const int ph = p + t;
455 const int ph2 = ph / 2;
456 /* Compute Bezier degree elevation coefficients */
457 DynamicMatrixType bezalfs(p + t + 1, p + 1);
458 bezalfs[0][0] = bezalfs[ph][p] = 1.0;
459 for (int i = 1; i <= ph2; ++i) {
460 const ScalarType inv = 1.0 / binomial(ph, i);
461 for (int j = std::max(0, i - t); j <= std::min(p, i); ++j)
462 bezalfs[i][j] = inv * binomial(p, j) * binomial(t, i - j);
463 }
464 for (int i = ph2 + 1; i <= ph - 1; ++i)
465 for (int j = std::max(0, i - t); j <= std::min(p, i); ++j)
466 bezalfs[i][j] = bezalfs[ph - i][p - j];
467
468 ControlPointNetType newCPv(oldData.controlPoints.strideSizes());
469
470 auto oldCPs = oldData.controlPoints;
471
472 auto scaleCPWithW = [](const auto& cp) -> ControlPointType { return {.p = cp.w * cp.p, .w = cp.w}; };
473 std::ranges::transform(oldCPs.directGetAll(), oldCPs.directGetAll().begin(), scaleCPWithW);
474
475 std::vector<ControlPointType> bpts(p + 1), ebpts(ph + 1), nextbpts(p - 1);
476
477 NURBSPatchData newData;
478 newData.degree = oldData.degree;
479 newData.degree[elevationDirection] += t;
480 const auto& U = oldData.knotSpans[elevationDirection];
481 double ua = U[0];
482 std::vector<ScalarType> Uh;
483 for (int j = 0, i = 0; i < U.size(); ++i) // insert knot t times for each unique knot
484 if (j < U.size()) {
485 do {
486 Uh.push_back(U[j]);
487 ++j;
488 } while (j != U.size() && FloatCmp::eq(U[j - 1], U[j]));
489 Uh.insert(Uh.end(), t, U[j - 1]);
490 if (j == U.size())
491 break;
492 }
493
494 for (int i = 0; i < dim; ++i)
495 if (i == elevationDirection)
496 newData.knotSpans[i] = Uh;
497 else
498 newData.knotSpans[i] = oldData.knotSpans[i];
499
500 auto dimSize = oldData.controlPoints.strideSizes();
501 dimSize[elevationDirection] = Uh.size() - ph - 1;
502 newData.controlPoints = MultiDimensionalNet<dim, ControlPointType>(dimSize);
503 auto& newCPs = newData.controlPoints;
504
505 int totalCurvesInPatch = 1;
506 std::array<int, dim - 1> indicesOther;
507 for (int counter = 0, i = 0; i < dim; ++i)
508 if (i != elevationDirection) {
509 totalCurvesInPatch *= dimSize[i];
510 indicesOther[counter++] = dimSize[i];
511 }
512
513 auto newAndOldCurve = [&elevationDirection, indicesOther, &oldCPs, &newCPs](auto i) {
514 const auto mI =
515 MultiDimensionalNet<dim - 1, int>::template directToMultiIndex<std::array<int, dim - 1>>(indicesOther, i);
516 std::array<int, dim> multiIndex{};
517 for (int c = 0, j = 0; j < dim; ++j)
518 if (j != elevationDirection)
519 multiIndex[j] = mI[c++];
520
521 auto oldCurve = [&oldCPs, &elevationDirection, multiIndex](int i) mutable {
522 multiIndex[elevationDirection] = i;
523 return oldCPs.get(multiIndex);
524 };
525
526 auto newCurve = [&newCPs, &elevationDirection, multiIndex](int i) mutable -> auto& {
527 multiIndex[elevationDirection] = i;
528 return newCPs.get(multiIndex);
529 };
530 return std::make_pair(oldCurve, newCurve);
531 };
532
533 for (auto [oldCurve, newCurve] :
534 std::ranges::iota_view(0, totalCurvesInPatch) | std::views::transform(newAndOldCurve)) {
535 ua = U[0];
536 int kind = ph + 1;
537 int r = -1;
538 int a = p;
539 int b = p + 1;
540 int cind = 1;
541
542 newCurve(0) = oldCurve(0);
543 for (int i = 0; i <= p; ++i)
544 bpts[i] = oldCurve(i);
545
546 std::vector<ScalarType> alphas(p - 1);
547 while (b < m) {
548 const int bold = b;
549 while (b < m && FloatCmp::eq(U[b], U[b + 1]))
550 ++b;
551 const int mul = b - bold + 1;
552 const auto ub = U[b];
553 const int oldr = r;
554 r = p - mul;
555 const int lbz = (oldr > 0) ? ((oldr + 2) / 2) : 1; // Insert knot u(b) r times
556 const int rbz = (r > 0) ? (ph - (r + 1) / 2) : ph;
557 if (r > 0) // Insert knot to get Bezier segment
558 {
559 const auto numer = ub - ua;
560 for (int k = p; k > mul; --k)
561 alphas[k - mul - 1] = numer / (U[a + k] - ua);
562 for (int j = 1; j <= r; ++j) {
563 const int s = mul + j;
564 for (int k = p; k >= s; --k)
565 bpts[k] = alphas[k - s] * bpts[k] + (1.0 - alphas[k - s]) * bpts[k - 1];
566 nextbpts[r - j] = bpts[p];
567 }
568 } // End of insert knot
569
570 for (int i = lbz; i <= ph; ++i) { // Degree elevate Bezier
571 ebpts[i].setZero();
572 for (int j = std::max(0, i - t); j <= std::min(p, i); ++j)
573 ebpts[i] += bezalfs[i][j] * bpts[j];
574 } // End of degree elevating Bezier
575 if (oldr > 1) { // Must remove knot u = U[a] oldr times
576 int first = kind - 2;
577 int last = kind;
578 const ScalarType den = ub - ua;
579 const ScalarType bet = (ub - Uh[kind - 1]) / den;
580 for (int tr = 1; tr < oldr; ++tr) { // Knot removal loop
581 int i = first;
582 int j = last;
583 int kj = j - kind + 1;
584 while (j - i > tr) {
585 if (i < cind) {
586 const ScalarType alf = (ub - Uh[i]) / (ua - Uh[i]);
587 newCurve(i) = alf * newCurve(i) + (1.0 - alf) * newCurve(i - 1);
588 }
589 if (j >= lbz) {
590 const auto factor = (j - tr <= kind - ph + oldr) ? (ub - Uh[j - tr]) / den : bet;
591 ebpts[kj] = factor * ebpts[kj] + (1.0 - factor) * ebpts[kj + 1];
592 }
593 ++i;
594 --j;
595 --kj;
596 }
597 --first;
598 ++last;
599 }
600 } // End of removing knot, u= U[a]
601 if (a != p)
602 kind += ph - oldr;
603 for (int j = lbz; j <= rbz; ++j, ++cind) // Load ctrl pts into Qw
604 newCurve(cind) = ebpts[j];
605 if (b < m) { // Set up for next pass through loop
606 std::ranges::copy(nextbpts | std::views::take(r), bpts.begin());
607 for (int j = r; j <= p; ++j)
608 bpts[j] = oldCurve(b - p + j);
609 a = b;
610 ++b;
611 ua = ub;
612 }
613 } // End of while-loop (b<m)
614 }
615 std::ranges::transform(newData.controlPoints.directGetAll(), newData.controlPoints.directGetAll().begin(),
616 [](auto& cp) -> ControlPointType { return {.p = cp.p / cp.w, .w = cp.w}; });
617 return newData;
618}

631template <std::integral auto dim, std::integral auto dimworld, typename ScalarType>
632auto knotRefinement(const NURBSPatchData<dim, dimworld, ScalarType>& oldData, const std::vector<double>& newKnots,
633 const int refinementDirection) {
634 if (newKnots.empty())
635 return oldData;
636 assert(std::ranges::is_sorted(newKnots) &&
637 "You passed a non-sorted vector of new knots. Sort it first or check if you passed the correct vector.");
638 using NurbsPatchData = NURBSPatchData<dim, dimworld, ScalarType>;
639 using ControlPoint = typename NurbsPatchData::ControlPointType;
640 using namespace std::ranges;
641 std::array<std::vector<double>, dim> newKnotsArray;
642 std::array<int, dim - 1> otherDirections;
643 for (int counter = 0, i = 0; i < dim; ++i) {
644 if (i == refinementDirection)
645 continue;
646 newKnotsArray[i] = oldData.knotSpans[i];
647 otherDirections[counter++] = i;
648 }
649
650 typename NurbsPatchData::ControlPointNetType newCPv(oldData.controlPoints.strideSizes());
651
652 auto oldCPv = oldData.controlPoints;
653 auto oldKnotVec = oldData.knotSpans[refinementDirection];
654
655 auto& newKnotVec = newKnotsArray[refinementDirection];
656
657 // Combine old and new knot vectors
658 newKnotVec.resize(oldKnotVec.size() + newKnots.size());
659 merge(oldKnotVec, newKnots, std::begin(newKnotVec));
660
661 const auto newKSize = newKnots.size();
662 std::array<int, dim> numberOfCPperDir = oldCPv.strideSizes();
663 numberOfCPperDir[refinementDirection] += newKSize;
664
665 newCPv.resize(numberOfCPperDir);
666
667 auto scaleCPWithW = [](const auto& cp) -> ControlPoint { return {.p = cp.w * cp.p, .w = cp.w}; };
668
669 const int p = oldData.degree[refinementDirection];
670 const int a = findSpan(p, newKnots.front(), oldKnotVec);
671 const int b = findSpan(p, newKnots.back(), oldKnotVec);
672
673 auto newSurfI = newCPv.hyperSurfBegin(otherDirections);
674 auto oldSurfI = oldCPv.hyperSurfBegin(otherDirections);
675
676 for (auto hSOld = oldSurfI, hSNew = newSurfI; hSOld != oldCPv.hyperSurfBegin(otherDirections) + (a - p + 1);
677 ++hSOld, ++hSNew)
678 std::ranges::transform(*hSOld, (*hSNew).begin(), scaleCPWithW);
679 for (auto hSOld = oldSurfI + b, hSNew = newSurfI + b; hSOld != oldCPv.hyperSurfEnd(otherDirections); ++hSOld, ++hSNew)
680 std::ranges::transform(*hSOld, (*hSNew).begin(), scaleCPWithW);
681
682 newSurfI = newCPv.hyperSurfEnd(otherDirections) - 1;
683 oldSurfI = oldCPv.hyperSurfEnd(otherDirections) - 1;
684
685 auto currentNewKnot = newKnotVec.end();
686 auto currentOldKnot = oldKnotVec.end();
687
688 for (const auto& currentAdditionalKnot : reverse_view(newKnots)) {
689 while (currentAdditionalKnot <= *(currentOldKnot - 1) &&
690 std::distance(oldKnotVec.begin(), currentOldKnot) > a + 1) {
691 std::ranges::transform(*oldSurfI, (*newSurfI).begin(), scaleCPWithW);
692 --currentNewKnot, --currentOldKnot;
693 --newSurfI, --oldSurfI;
694 }
695
696 std::ranges::copy(*(newSurfI + 1), (*newSurfI).begin());
697 ++newSurfI;
698 currentOldKnot -= p;
699 for ([[maybe_unused]] int _ : iota_view{0, p}) {
700 ++newSurfI;
701
702 auto alpha = *currentNewKnot - currentAdditionalKnot;
703 using std::abs;
704 if (abs(alpha) < 1e-7)
705 std::ranges::copy(*newSurfI, (*(newSurfI - 1)).begin());
706 else {
707 alpha /= (*currentNewKnot - *(currentOldKnot));
708 std::ranges::transform(*newSurfI, *(newSurfI - 1), (*(newSurfI - 1)).begin(),
709 [&alpha](auto& cp, auto& cpL) { return alpha * cpL + cp * (1.0 - alpha); });
710 }
711 ++currentOldKnot, ++currentNewKnot;
712 }
713 currentNewKnot -= p + 1;
714 newSurfI -= p + 2;
715 }
716
717 std::ranges::transform(newCPv.directGetAll(), newCPv.directGetAll().begin(),
