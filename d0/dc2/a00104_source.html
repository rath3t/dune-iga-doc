58 const IdType& subId(const IdType& elementId, int localSubIndex, int codim) const {
59 if (codim == 0)

80 int outsideIntersectionIndex(const IdType& insideElementId, const IdType& outsideElementId, int indexInInside) const {
81 const IdType& insideSubId = subId(insideElementId, indexInInside, 1);
82
83 for (const auto i : Dune::range(subIds(outsideElementId, 1))) {
84 const IdType& outsideSubId = subId(outsideElementId, i, 1);
85 if (outsideSubId == insideSubId) {
86 return i;
87 }
88 }
89 DUNE_THROW(GridError, "outsideIntersectionIndex not successful");
90 }

142 template <int cc>
143 auto subIndexFromId(const IdType& id, int i, int codim, int lvl) const {
144 assert(codim > 0);
145 if constexpr (cc == 0) {
146 if (codim == 1)
147 return idToEdgeInfoMap.at(globalEdgesIdOfElementsMap_.at(id)[i]).indexInLvlStorage;
148 else if (codim == 2)
149 return idToVertexInfoMap[lvl].at(globalVerticesIdOfElementsMap.at(id)[i]).indexInLvlStorage;
150 __builtin_unreachable();
151 } else if constexpr (cc == 1) {
152 if (codim == 1)
153 return idToEdgeInfoMap.at(id).indexInLvlStorage;
154 else if (codim == 2) // get index of vertex from edge id
155 return idToVertexInfoMap[lvl].at(globalVertexIdOfEdgesMap_.at(id)[i]).indexInLvlStorage;
156 __builtin_unreachable();
157 } else {
158 assert(codim == 2 and i == 0);
159 return idToVertexInfoMap[lvl].at(id).indexInLvlStorage;
160 }

174 GeoTypes types(int codim, int level) const {
175 if (codim == 0) {
176 if (not Preferences::getInstance().reconstructTrimmedLocalGeometry())
177 return GeoTypes{GeometryTypes::cube(gridDim)};
178
179 if (numberOfTrimmedElements[level] == 0)
180 return GeoTypes{GeometryTypes::cube(gridDim)};
181
182 if (numberOfUnTrimmedElements[level] == 0)
183 return GeoTypes{GeometryTypes::none(gridDim)};
184
185 return GeoTypes{GeometryTypes::cube(gridDim), GeometryTypes::none(gridDim)};
186 }
187 return GeoTypes{GeometryTypes::cube(gridDim - codim)};
188 }

207 std::size_t size(GeometryType type, int lvl) const {
208 if (type == GeometryTypes::quadrilateral) {
209 if (not Preferences::getInstance().reportTrimmedElementGeometryTypeAsNone())
210 return numberOfTrimmedElements[lvl] + numberOfUnTrimmedElements[lvl];
211 return numberOfUnTrimmedElements[lvl];
212 }
213 if (type == GeometryTypes::none(gridDim))
214 return numberOfTrimmedElements[lvl];
215 if (type == GeometryTypes::line)
216 return std::get<1>(entityImps_[lvl]).size();
217 if (type == GeometryTypes::vertex)
218 return std::get<2>(entityImps_[lvl]).size();
219 return 0;
220 }

222 std::size_t sizeOfInfos(int codim, int lvl) const {
223 if (codim == 2)
224 return idToVertexInfoMap[lvl].size();
225
226 auto count = [=](auto&& range) -> std::size_t {
227 return std::ranges::count_if(range, [=](const auto& entityInfoPair) { return entityInfoPair.second.lvl == lvl; });
228 };
229 if (codim == 0)
230 return count(idToElementInfoMap);
231 if (codim == 1)
232 return count(idToEdgeInfoMap);
233 __builtin_unreachable();
234 }
