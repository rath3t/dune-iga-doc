25 static auto createIntegrationRule(const PatchElement& element, int quadratureOrder,

27 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) {
28 if (not element.impl().getLocalEntity().isTrimmed()) {

35private:
36 static auto makeQuadratureRule(const std::vector<Element>& elements, int quadratureOrder,
37 const QuadratureType::Enum qt) {
38 assert(qt == QuadratureType::GaussLegendre);
39
40 Dune::QuadratureRule<double, dim> vector{};
41
42 for (auto& subElement : elements) {
43 const auto& rule = Dune::QuadratureRules<double, dim>::rule(subElement.type(), quadratureOrder, qt);
44 for (auto ip : rule) {
45 auto globalInSpan = subElement.global(ip.position());
46 vector.emplace_back(globalInSpan, ip.weight() * subElement.integrationElement(ip.position()));
47 }
48 }
49 return vector;
50 }
51};
