364 QuadratureRule<double, dim> getQuadratureRule(const std::optional<int>& p_order = std::nullopt,
365 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) const
366 requires(not Trimmer::isAlwaysTrivial and dimension == 2)
367 {
368 auto degree = patchGrid_->patchGeometry(this->level()).degree();
369 int order = p_order.value_or(dimension * *std::ranges::max_element(degree));
370 if (not isTrimmed())
371 return QuadratureRules<double, dimension>::rule(this->type(), order, qt);
372
373 return patchGrid_->integrationRule()(localEntity_, order, qt);
374 }
