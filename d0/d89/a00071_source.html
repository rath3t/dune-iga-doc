22template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
23class TrimmedLocalGeometryImpl<2, coorddim, GridImp, localGeometryTag>
24{
25public:
26 using ctype = typename GridImp::ctype;
27
28 static constexpr int mydimension = 2;
29 using Trimmer = typename GridImp::Trimmer;
30
31 static constexpr int coorddimension = coorddim;
32 static constexpr int codim = coorddimension - mydimension;
33 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
34 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
35
36 using HostGeometry = typename Trimmer::TrimmerTraits::template Codim<0>::HostParameterSpaceGridEntity::Geometry;
37 using TrimData = typename Trimmer::ElementTrimData;
38
39 using LocalCoordinate = FieldVector<ctype, mydimension>;
40 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
41 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
42 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
43 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
44 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
45 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
46 using Volume = ctype;
47
48 // type of the LocalView of the patch geometry
49 using GeometryLocalView = typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
50 ctype>::template GeometryLocalView<codim, Trimmer>;
51
53 TrimmedLocalGeometryImpl() = default;
54 explicit TrimmedLocalGeometryImpl(const HostGeometry& hostGeometry, const TrimData& trimData)
55 : hostGeometry_{hostGeometry},
56 trimData_{trimData} {}
57 bool operator==(const TrimmedLocalGeometryImpl& b) const {
58 return b.hostGeometry_ == this->hostGeometry_;
59 }
60
63 [[nodiscard]] GeometryType type() const {
64 return GeometryTypes::none(mydimension);
65 }
66
67 // return whether we have an affine mapping (true for straight lines)
68 [[nodiscard]] bool affine() const {
69 return true;
70 }
71
72 // return the number of corners of this element. Corners are numbered 0...n-1
73 [[nodiscard]] int corners() const {
74 return trimData_.size(2);
75 }
76
77 // @todo write a test for that
78 GlobalCoordinate center() const {
79 // average of corners
80 GlobalCoordinate c{0, 0};
81 for (auto i : Dune::range(corners()))
82 c += corner(i);
83
84 return c / corners();
85 }
86
87 // access to coordinates of corners. Index is the number of the corner
88 GlobalCoordinate corner(int i) const {
89 auto vData = trimData_.vertex(i);
90 if (vData.isHost)
91 return hostGeometry_.corner(Transformations::mapToDune(2, vData.idx));
92
93 return vData.geometry.value();
94 }
95
98 GlobalCoordinate global(const LocalCoordinate& local) const {
99 return hostGeometry_.global(local);
100 }
101
104 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
105 return hostGeometry_.jacobianTransposed(local);
106 }
107
110 LocalCoordinate local(const GlobalCoordinate& global) const {
111 return hostGeometry_.local(global);
112 }
113
114 // Returns true if the point is in the current element
115 // @todo
116 bool checkInside(const LocalCoordinate& local) const {
117 return true;
118 }
119
120 // @todo
121 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
122 return hostGeometry_.volume();
123 }
124
125 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
126 return hostGeometry_.jacobianInverseTransposed(local);
127 }
128
129private:
130 HostGeometry hostGeometry_;
131 TrimData trimData_;
132};

134template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
135class TrimmedLocalGeometryImpl<1, coorddim, GridImp, localGeometryTag>
136{
137public:
138 using ctype = typename GridImp::ctype;
139
140 static constexpr int mydimension = 1;
141 using Trimmer = typename GridImp::Trimmer;
142
143 static constexpr int coorddimension = coorddim;
144 static constexpr int codim = coorddimension - mydimension;
145 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
146 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
147 using LocalCoordinate = FieldVector<ctype, mydimension>;
148 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
149 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
150 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
151 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
152 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
153 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
154 using Volume = ctype;
155
156 // type of the LocalView of the patch geometry
157 using GeometryLocalView = typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
158 ctype>::template GeometryLocalView<codim, Trimmer>;
159
161 TrimmedLocalGeometryImpl() = default;
162 explicit TrimmedLocalGeometryImpl(const GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>& patchGeometry)
163 : patchGeometry{patchGeometry} {}
164
165 bool operator==(const TrimmedLocalGeometryImpl& b) const {
166 return Dune::FloatCmp::eq(b.patchGeometry.corner(0) == this->patchGeometry.corner(0)) and
167 Dune::FloatCmp::eq(b.patchGeometry.corner(1) == this->patchGeometry.corner(1));
168 };
169
172 [[nodiscard]] GeometryType type() const {
173 return GeometryTypes::cube(mydimension);
174 }
175
176 // return whether we have an affine mapping (true for straight lines)
177 [[nodiscard]] bool affine() const {
178 // handy if the trims are straight lines for other algorithms
179 if constexpr (codim == 0)
180 return true;
181 else
182 return patchGeometry.degree()[0] == 1;
183 }
184
185 // return the number of corners of this element. Corners are numbered 0...n-1
186 [[nodiscard]] int corners() const {
187 return patchGeometry.corners();
188 }
189
190 GlobalCoordinate center() const {
191 return patchGeometry.center();
192 }
193
194 // access to coordinates of corners. Index is the number of the corner
195 GlobalCoordinate corner(int i) const {
196 return patchGeometry.corner(i);
197 }
198
201 GlobalCoordinate global(const LocalCoordinate& local) const {
202 return patchGeometry.global(local);
203 }
204
207 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
208 return patchGeometry.jacobianTransposed(local);
209 }
210
213 LocalCoordinate local(const GlobalCoordinate& global) const {
214 return patchGeometry.local(global);
215 }
216
217 // Returns true if the point is in the current element
218 // @todo
219 bool checkInside(const LocalCoordinate& local) const {
220 return true;
221 }
222
223 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
224 return patchGeometry.volume();
225 }
226
227 // The Jacobian matrix of the mapping from the reference element to this element
228 // @todo not yet implemented
229 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
230 return patchGeometry.jacobianInverseTransposed(local);
231 }
232
233private:
234 GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype> patchGeometry;
235};
