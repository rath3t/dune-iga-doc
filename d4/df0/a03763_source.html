593 std::vector<ElementTrimData> trimElements(std::optional<int> level_ = std::nullopt) const {
594 int level = level_.value_or(maxLevel());
595 std::vector<ElementTrimData> elementTrimDatas;
596 auto gv = parameterSpaceGrid_->levelGridView(level);
597 for (const auto& ele : elements(gv)) {
598 if (trimData_.has_value())
599 elementTrimDatas.emplace_back(trimElement(ele, trimData_.value()));
600 else
601 elementTrimDatas.emplace_back(ElementTrimFlag::full, ele);
602 }
603 return elementTrimDatas;
604 }
