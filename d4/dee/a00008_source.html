82template <typename ScalarType, int dim, size_t dim2>
83requires(dim == dim2)
84bool clampToBoundaryAndCheckIfIsAtAllBoundaries(FieldVector<ScalarType, dim>& x,
85 const std::array<Domain<ScalarType>, dim2>& domain = {}) {
86 int breakDueToBoundaryCounter = 0;
87
88 for (int j = 0; j < dim; ++j) {
89 x[j] = std::clamp(x[j], domain[j].left(), domain[j].right());
90 if (x[j] == domain[j].left() or x[j] == domain[j].right())
91 ++breakDueToBoundaryCounter;
92 }
93 if (dim == breakDueToBoundaryCounter)
94 return true;
95 return false;
96}

107template <std::floating_point T>
108auto linspace(T a, T b, unsigned int N) {
109 const T inc = (b - a) / static_cast<T>(N - 1);
110 auto val = [=](int i) -> T {
111 if (i != 0 and i != N - 1)
112 return a + i * inc;
113 if (i == N - 1)
114 return b;
115 return a;
116 };
117
118 return std::views::iota(0u, N) | std::views::transform(val);
119}
