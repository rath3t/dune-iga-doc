33 static auto createSimplicies(const ParameterSpaceElement& element, const Parameters& parameters = Parameters{})
34 -> std::tuple<std::vector<Element>, std::vector<Point>, std::vector<Index>> {
35 std::vector<Point> vertices{};
36 std::vector<Element> elements{};
37
38 auto& trimData = element.trimData();
39 auto hostEntity = element.getHostEntity();
40
41 for (const auto& edgeInfo : trimData.edges()) {
42 if (edgeInfo.isTrimmed) {
43 auto localGeometry = GeometryKernel::transformToSpan(edgeInfo.geometry.value(), hostEntity.geometry());
44 splitBoundary(localGeometry, parameters, vertices);
45 } else {
46 auto refElement = referenceElement(hostEntity);
47 switch (edgeInfo.idx) {
48 case 0:
49 vertices.push_back(refElement.template geometry<2>(0).center());
50 break;
51 case 1:
52 vertices.push_back(refElement.template geometry<2>(1).center());
53 break;
54 case 2:
55 vertices.push_back(refElement.template geometry<2>(3).center());
56 break;
57 case 3:
58 vertices.push_back(refElement.template geometry<2>(2).center());
59 break;
60 default:
61 assert(edgeInfo.idx < 4);
62 __builtin_unreachable();
63 }
64 }
65 }
66
67 auto indices = triangulate(vertices);
68 for (auto it = indices.begin(); it < indices.end(); it += 3)
69 elements.emplace_back(GeometryTypes::triangle, std::vector<FieldVector<double, dim>>{
70 vertices[*it], vertices[*(it + 1)], vertices[*(it + 2)]});
71
72 return std::make_tuple(elements, vertices, indices);
73 }

77 static void splitBoundary(const auto& localGeometry, const Parameters& parameters, std::vector<Point>& points) {
78 if (localGeometry.affine()) {
79 points.push_back(localGeometry.global({0.0}));
80 return;
81 }
82
83 if (not parameters.useAdaptiveDivisions()) {
84 points.reserve(parameters.boundaryDivisions);
85 for (auto local : Utilities::linspace(0.0, 1.0, parameters.boundaryDivisions))
86 points.push_back(localGeometry.global({local}));
87 } else {
88 // Guess initial amount of segments (a linear segment of lenght 1 should be devided by 1 divions)
89 double curveLength = computeCurveLength(localGeometry);
90 auto segments = static_cast<unsigned int>(std::ceil(1 * curveLength * localGeometry.degree()[0]));
91
92 bool converged = false;
93 std::vector<Point> newPoints;
94 while (not converged) {
95 std::tie(converged, newPoints) =
96 checkForConvergence(localGeometry, segments, curveLength, parameters.targetAccuracy);
97 ++segments;
98 if (converged)
99 points.insert(points.end(), newPoints.begin(), newPoints.end());
100 }
101 }
102 // points.push_back(localGeometry.global({1.0}));
103
104 // Remove last element to avoid duplication
105 points.pop_back();
106 }

108 static auto triangulate(const std::vector<Point>& points) -> std::vector<Index> {
109 std::vector<std::vector<Point>> polygonInput;
110 polygonInput.push_back(points);
111
112 return mapbox::earcut<Index>(polygonInput);
113 }

115 // Function to calculate the distance between two points
116 static auto checkForConvergence(const auto& localGeometry, unsigned int numSegments, double curveLength,
117 double targetAccuracy) -> std::pair<bool, std::vector<Point>> {
118 auto [totalDistance, points] = getPointsAndDistance(localGeometry, numSegments);
119 auto converged = std::abs((totalDistance - curveLength) / curveLength) < targetAccuracy;
120 return std::make_pair(converged, points);
121 }
122 static auto computeCurveLength(const auto& localGeometry) -> double {
123 unsigned int numSegments = localGeometry.degree()[0] * 25;
124
125 return std::get<0>(getPointsAndDistance(localGeometry, numSegments));
126 }
127
128 static auto getPointsAndDistance(const auto& localGeometry, unsigned int numSegments)
129 -> std::pair<double, std::vector<Point>> {
130 auto distance = [](const Point& p1, const Point& p2) {
131 return std::sqrt(Dune::power(p2[0] - p1[0], 2) + Dune::power(p2[1] - p1[1], 2));
132 };
133
134 std::vector<Point> points;
135 for (auto local : Utilities::linspace(0.0, 1.0, numSegments + 1))
136 points.push_back(localGeometry.global({local}));
137
138 double totalDistance = 0.0;
139 for (auto i : Dune::range(1ul, points.size()))
140 totalDistance += distance(points[i - 1], points[i]);
141
142 return std::make_pair(totalDistance, points);
143 }
144};
