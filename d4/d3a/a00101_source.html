30 static auto createIntegrationRule(const PatchElement& element, int quadratureOrder,
31 const Parameters& parameters = Parameters{},
32 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) {
33 if (not element.impl().getLocalEntity().isTrimmed()) {
34 return QuadratureRules<double, dim>::rule(element.type(), quadratureOrder, qt);
35 }
36
37 auto [elements, vertices, indices] = createSimplicies(element, parameters);
38 return makeQuadratureRule(elements, quadratureOrder, qt);
39 }
40
41 static auto createSimplicies(const PatchElement& element, const Parameters& parameters = Parameters{})
42 -> std::tuple<std::vector<Element>, std::vector<Point>, std::vector<Index>> {
43 std::vector<Point> vertices{};
44 std::vector<Element> elements{};
45
46 auto& trimData = element.impl().getLocalEntity().trimData();
47 auto hostEntity = element.impl().getLocalEntity().getHostEntity();
48
49 for (const auto& edgeInfo : trimData.edges()) {
50 if (edgeInfo.isTrimmed) {
51 auto localGeometry = GeometryKernel::transformToSpan(edgeInfo.geometry.value(), hostEntity.geometry());
52 std::vector<Point> pointsT = splitBoundary(localGeometry, parameters);
53 vertices.insert(vertices.end(), pointsT.begin(), pointsT.end());
54 } else {
55 auto refElement = referenceElement(hostEntity);
56 switch (edgeInfo.idx) {
57 case 0:
58 vertices.push_back(refElement.template geometry<2>(0).center());
59 break;
60 case 1:
61 vertices.push_back(refElement.template geometry<2>(1).center());
62 break;
63 case 2:
64 vertices.push_back(refElement.template geometry<2>(3).center());
65 break;
66 case 3:
67 vertices.push_back(refElement.template geometry<2>(2).center());
68 break;
69 default:
70 assert(edgeInfo.idx < 4);
71 __builtin_unreachable();
72 }
73 }
74 }
75
76 auto indices = triangulate(vertices);
77 for (auto it = indices.begin(); it < indices.end(); it += 3)
78 elements.emplace_back(GeometryTypes::triangle, std::vector<FieldVector<double, dim>>{
79 vertices[*it], vertices[*(it + 1)], vertices[*(it + 2)]});
80
81 return std::make_tuple(elements, vertices, indices);
82 }
83
84private:
85 static std::vector<Point> splitBoundary(const auto& localGeometry, const Parameters& parameters) {
86 std::vector<Point> points;
87
88 if (localGeometry.affine()) {
89 points.push_back(localGeometry.global({0.0}));
90 return points;
91 }
92
93 points.reserve(parameters.maxBoundaryDivisions);
94
95 // todo at the moment only boundaryDivisions
96 for (auto local : Utilities::linspace(0.0, 1.0, parameters.maxBoundaryDivisions))
97 points.push_back(localGeometry.global({local}));
98
99 // Remove last element to avoid duplication
100 points.pop_back();
101 return points;
102 }
103 static auto triangulate(const std::vector<Point>& points) -> std::vector<Index> {
104 std::vector<std::vector<Point>> polygonInput;
105 polygonInput.push_back(points);
106
107 return mapbox::earcut<Index>(polygonInput);
108 }
109 static auto makeQuadratureRule(const std::vector<Element>& elements, int quadratureOrder,
110 const QuadratureType::Enum qt) {
111 assert(qt == QuadratureType::GaussLegendre);
112
113 Dune::QuadratureRule<double, dim> vector{};
114
115 for (auto& subElement : elements) {
116 const auto& rule = Dune::QuadratureRules<double, dim>::rule(subElement.type(), quadratureOrder, qt);
117 for (auto ip : rule) {
118 auto globalInSpan = subElement.global(ip.position());
119 vector.emplace_back(globalInSpan, ip.weight() * subElement.integrationElement(ip.position()));
120 }
121 }
122 return vector;
123 }
124};
