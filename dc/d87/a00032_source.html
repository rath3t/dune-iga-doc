26 template <int dim>
27 auto& voigtIndices() {
28 if constexpr (dim == 1) {
29 static std::array<IndexPair, dim*(dim + 1) / 2> voigt1D = {
30 {0, 0}
31 };
32 return voigt1D;
33 } else if constexpr (dim == 2) {
34 static std::array<IndexPair, dim*(dim + 1) / 2> voigt2D = {
35 {{0, 0}, {1, 1}, {0, 1}}
36 };
37 return voigt2D;
38 } else if constexpr (dim == 3) {
39 static std::array<IndexPair, dim*(dim + 1) / 2> voigt3D = {
40 {{0, 0}, {1, 1}, {2, 2}, {1, 2}, {0, 2}, {0, 1}}
41 };
42 return voigt3D;

86 // if we have codim==0, then the Jacobian in the parameter space of the grid entity itself is a DiagonalMatrix,
87 // and
88 // Coordinates in a single knot span differ from coordinates on the B-spline patch
89 // by an affine transformation. This transformation is stored in the diagonal entries.
90 // If trimming is disabled the Jacobian in the parameter space of subentities (edges or surfaces) is a
91 // DiagonalMatrixBlock but is treated as a FieldMatrix, since is also just cubes and an axis-aligned geometry, if
92 // trimming is enabled the Jacobian of the parameterspace for subentities is a potentially fully populated
93 // FieldMatrix, Think about an arbitrary curve in the axis-aligned cube parameter grid
94 // ------->
95 // `____________________C_______D_______`
96 // ::````````````````::``````````'|````::
97 // :: :: `| ::
98 // :: ^F:: `| ::
99 // :: | :: `| ::
100 // :: | :: `| ::
101 // :: |E:: `| ::
102 // :: :: .| x ::
103 // :: :: _: ::
104 // ::................::..........|:....::
105 // ::''''''''''''''''::''''''''''/'''''::
106 // :: :: _:B ::
107 // :: :: .| ::
108 // :: :: `::A ::
109 // :: '::::||::::::' ::
110 // :: ::` :: x ::
111 // :: |' :: ::
112 // :: | x :: ::
113 // ::````````|```````::````````````````::
114 // `____________________________________`
115 // at point A the tangent of the curve (the JacobianInParameterSpace) points into the oblique direction (to B)
116 // for the axis-aligned geometry these tangents are from E to F or from C to D, depending on the orientation of
117 // the subentity, which are aligned with the axis
118 using JacobianTransposedInParameterSpace = typename ParameterSpaceGeometry::JacobianTransposed;
119 using GlobalInParameterSpace = typename ParameterSpaceGeometry::GlobalCoordinate;

337 Hessian hessian(const LocalCoordinate& local) const {
338 checkState();
339 const auto ouInPatch = globalInParameterSpace(local);
340
341 const auto [p, dfdg, dfdgdg] =
342 patchGeometry_->zeroFirstAndSecondDerivativeOfPositionImpl(ouInPatch, nurbsLocalView_, localControlPointNet);
343
344 const JacobianTransposedInParameterSpace dgdt = jacobianTransposedInParameterSpace(local);
345 static_assert(JacobianTransposedInParameterSpace::rows == mydimension);
346 static_assert(JacobianTransposedInParameterSpace::cols == gridDimension);
347 // dgdt (1x2, curve on surface), (2x2 surface in surface), (2x3 surface in 3D), (3x3 volume in 3D)
348 FieldMatrix<ctype, patchNumberOfSecondDerivatives, numberOfSecondDerivatives> dgdtSquared(0);
349 for (int patchIndex = 0; auto [patchRow, patchCol] : Impl::voigtIndices<gridDimension>()) {
350 for (int myIndex = 0; auto [myRow, myCol] : Impl::voigtIndices<mydimension>()) {
351 if constexpr (not std::is_same_v<JacobianTransposedInParameterSpace, DiagonalMatrix<ctype, gridDimension>>)
352 dgdtSquared[patchIndex][myIndex] =
353 (dgdt[myRow][patchRow] * dgdt[myCol][patchCol] + dgdt[myCol][patchRow] * dgdt[myRow][patchCol]) *
354 ((patchRow == patchCol) ? 0.5 : 1.0);
355 else {
356 const auto dgdtmyRowpatchRow = (myRow == patchRow) ? dgdt[myRow][patchRow] : 0;
357 const auto dgdtmyColpatchCol = (myCol == patchCol) ? dgdt[myCol][patchCol] : 0;
358 const auto dgdtmyColpatchRow = (myCol == patchRow) ? dgdt[myCol][patchRow] : 0;
359 const auto dgdtmyRowpatchCol = (myRow == patchCol) ? dgdt[myRow][patchCol] : 0;
360 dgdtSquared[patchIndex][myIndex] =
361 (dgdtmyRowpatchRow * dgdtmyColpatchCol + dgdtmyColpatchRow * dgdtmyRowpatchCol) *
362 ((patchRow == patchCol) ? 0.5 : 1.0);
363 }
364 ++myIndex;
365 }
366 ++patchIndex;
367 }
368
369 // h = mydimension * (mydimension + 1) / 2 X worlddimension = 1x2 (curve in 2D), 1x3 (curve in 3D), (3x2 surface
370 // in 2D), (3x3 surface in 3D), (6x3 volume in 3D)
371 //
372 // dfdgdg = gridDimension*(gridDimension + 1) / 2 X worlddimension
373 // = 3x2 (curve on surface flat),3x3 (curve on surface in 3D), 6x3 (curve in volume), 6x3 (surface in
374 // volume),(3x3 surface in surface in 3D)
375 //
376 // dgdtSquared = gridDimension * (gridDimension + 1) / 2 X mydimension * (mydimension +
377 // 1) / 2 = 3x1 (curve on surface), 6x1 (curve in volume), 6x3 (surface in volume), 3x3 ( surface in surface),
378 // 6x6 ( volume in volume)
379 Hessian h = transposedView(dgdtSquared) * dfdgdg;
380
381 /* if trimming is enabled the parameter space geometry is potentially non-linear,
382 * the resutling Hessian has another contribution due to chain-rule, namely the second derivative of g */
383 if constexpr (not Trimmer::template isLocalGeometryLinear<codim>) {
384 // const auto dgdtdt = parameterSpaceGeometry->hessian(ouInPatch);
385 //
386 // assert(TrimmerType::template isLocalGeometryLinear<
387 // codim> && "This can not be checked yet. Check if this works with trimming and then remove
388 // assert");
389 // h += transpose(transposedView(dfdg) * dgdtdt);
390 }
391 return h;
392 }
