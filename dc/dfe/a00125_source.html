13 auto sameCorner(const auto& corner1, const auto& corner2, double precision = 1e-10) -> bool {
14 return std::ranges::all_of(Dune::range(2),
15 [&](auto i) -> bool { return Dune::FloatCmp::eq(corner1[i], corner2[i], precision); });
16 };
17
18 bool isSameEdgeGeometry(const auto& geo1, const auto& geo2) {
19 return sameCorner(geo1.corner(0), geo2.corner(0)) and sameCorner(geo1.corner(1), geo2.corner(1)) ||
20 sameCorner(geo1.corner(0), geo2.corner(1)) and sameCorner(geo1.corner(1), geo2.corner(0)) ||
21 sameCorner(geo1.corner(1), geo2.corner(0)) and sameCorner(geo1.corner(0), geo2.corner(1));
22 }

32template <int dim, int dimworld, typename ScalarType>
33auto ParameterSpaceImpl<dim, dimworld, ScalarType>::idForTrimmedVertex(const FieldVector<double, 2>& vertex)
34 -> GlobalIdType {
35 using PersistentIndexType = typename ParameterSpaceTraits::PersistentIndexType;
36
37 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
38 auto& vertexIdContainer = entityContainer_.trimmedVertexIds_.back();
39 bool alreadyThere = false;
40 PersistentIndexType foundIndex{};
41
42 for (auto& [vertexId, pos] : vertexIdContainer) {
43 if (Util::sameCorner(vertex, pos)) {
44 alreadyThere = true;
45 foundIndex = vertexId.id;
46 }
47 }
48 if (not alreadyThere)
49 foundIndex = globalIdSet_->newFreeIndex();
50
51 GlobalIdType vertexId{.entityIdType = GlobalIdType::EntityIdType::newId, .id = foundIndex};
52 if (not alreadyThere)
53 vertexIdContainer.emplace(vertexId, vertex);
54 return vertexId;
55}

57template <int dim, int dimworld, typename ScalarType>
58auto ParameterSpaceImpl<dim, dimworld, ScalarType>::idForTrimmedHostEdge(
59 typename ParameterSpaceTraits::PersistentIndexType hostEdgeId,
60 const typename ElementTrimData::EdgeInfo& trimmedEdge) -> GlobalIdType {
61 using HostEdge = typename ParameterSpaceTraits::template Codim<1>::HostParameterSpaceGridEntity;
62 using PersistentIndexType = typename ParameterSpaceTraits::PersistentIndexType;
63
64 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
65
66 // We have to check weather the edgeId is already there
67 bool alreadyThere = false;
68 PersistentIndexType foundIndex{};
69 for (auto& [edgeId, edgeInfo] : entityContainer_.idToEdgeInfoMap) {
70 if (edgeId.hostId.has_value() and edgeInfo.id.hostId.value() == hostEdgeId) {
71 // We have the same hostId, this doesn't necessarily mean that its the same edge
72 auto edgeGeo = edgeInfo.edgeGeometries.front().geometry;
73 if (Util::isSameEdgeGeometry(edgeGeo, trimmedEdge.geometry.value())) {
74 alreadyThere = true;
75 foundIndex = edgeId.id;
76 break;
77 }
78 }
79 }
80 if (not alreadyThere)
81 foundIndex = globalIdSet_->newFreeIndex();
82 GlobalIdType edgeId = {
83 .entityIdType = GlobalIdType::EntityIdType::newId, .id = foundIndex, .hostId = std::make_optional(hostEdgeId)};
84 return edgeId;
85}

87template <int dim, int dimworld, typename ScalarType>
88void ParameterSpaceImpl<dim, dimworld, ScalarType>::createAndSaveElementInfo(
89 const std::tuple<unsigned int, unsigned int, int>& indices, const HostEntity<0>& ele, bool trimmed) {
90 const unsigned int unTrimmedElementIndex = std::get<0>(indices);
91 const unsigned int trimmedElementIndex = std::get<1>(indices);
92 const int newLevel = std::get<2>(indices);
93
94 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
95 auto& indexSet = parameterSpaceGrid_->levelGridView(newLevel).indexSet();
96
97 std::optional<GlobalIdType> fatherId{};
98 if (ele.hasFather())
99 fatherId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = globalIdSetParameterSpace.id(ele.father())};
100
101 GlobalIdType elementId = makeElementID(ele);
102 auto hostIndexInLvl = untrimmedParameterSpaceGrid_->levelGridView(newLevel).indexSet().index(
103 parameterSpaceGrid_->template getHostEntity<0>(ele));
104
105 EntityInfo<0> elementInfo = {
106 .indexInLvlStorage = trimmedElementIndex + unTrimmedElementIndex,
107 .unTrimmedIndexInLvl = unTrimmedElementIndex,
108 .trimmedIndexInLvl = trimmedElementIndex,
109 .hostIndexInLvl = hostIndexInLvl,
110 .lvl = newLevel,
111 .trimmed = trimmed,
112 .id = elementId,
113 .hostSeed = ele.seed(),
114 .fatherId = fatherId,
115 };
116 assert(trimmedElementIndex + unTrimmedElementIndex == indexSet.index(ele));
117
118 entityContainer_.idToElementInfoMap.insert({elementId, elementInfo});
119
120 // If we have a father we have to add us as his son, this can be faster, we can store in decendantIds,
121 // the indexInLvlStorage and lvl, which would provide faster access
122 if (fatherId.has_value()) {
123 entityContainer_.idToElementInfoMap.at(fatherId.value()).decendantIds.push_back(elementId);
124 entityContainer_.template entity<0>(fatherId.value(), newLevel - 1).entityInfo_.decendantIds.push_back(elementId);
125 }
126}

128template <int dim, int dimworld, typename ScalarType>
129void ParameterSpaceImpl<dim, dimworld, ScalarType>::collectElementEdges(int level, const HostEntity<0>& ele,
130 const ElementTrimData& eleTrimData) {
131 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
132 GlobalIdType elementId = makeElementID(ele);
133
134 auto& elementEdgeIndices = entityContainer_.globalEdgesIdOfElementsMap_[elementId];
135 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
136
137 // Define some lambdas (can be moved to separate functions later)
138 auto addHostEdge = [&](const unsigned int localEdgeIndex) {
139 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
140 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostEdgeId};
141 elementEdgeIndices.emplace_back(edgeId);
142
143 if (entityContainer_.idToEdgeInfoMap.contains(edgeId))
144 return;
145
146 auto edge = ele.template subEntity<1>(localEdgeIndex);
147 EntityInfo<1> edgeInfo{.indexInLvlStorage = entityContainer_.edgeCount.back()++,
148 .lvl = level,
149 .trimmed = false,
150 .id = edgeId,
151 .hostSeed = edge.seed()};
152 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
153
154 // store vertex ids of edges, for subIndex method of indexSet
155 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
156
157 const auto& cube = ReferenceElements<ctype, mydimension>::cube();
158 for (auto vertexLocalIndexWRTElement : cube.subEntities(localEdgeIndex, 1, 2)) {
159 auto hostVertexId = globalIdSetParameterSpace.subId(ele, vertexLocalIndexWRTElement, 2);
160 GlobalIdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
161 edgeVertexIndices.emplace_back(vertexId);
162 }
163 };
164
165 auto addTrimmedHostEdge = [&](int localEdgeIndex, const typename ElementTrimData::EdgeInfo& edgeOfTrimmedElement) {
166 // To get neighborhood information we save the original hostId
167 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
168 GlobalIdType edgeId = idForTrimmedHostEdge(hostEdgeId, edgeOfTrimmedElement);
169 elementEdgeIndices.emplace_back(edgeId);
170
171 if (entityContainer_.idToEdgeInfoMap.contains(edgeId)) {
172 entityContainer_.idToEdgeInfoMap.at(edgeId).edgeGeometries.emplace_back(indexSet.index(ele),
173 edgeOfTrimmedElement.geometry.value());
174 return;
175 }
176
177 auto edge = ele.template subEntity<1>(localEdgeIndex);
178 EntityInfo<1> edgeInfo{.indexInLvlStorage = entityContainer_.edgeCount.back()++,
179 .lvl = level,
180 .trimmed = true,
181 .id = edgeId,
182 .hostSeed = edge.seed(),
183 .trimInfo = edgeOfTrimmedElement};
184 edgeInfo.edgeGeometries.emplace_back(indexSet.index(ele), edgeOfTrimmedElement.geometry.value());
185 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
186
187 // store vertex ids of edges, for subIndex method of indexSet
188 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
189
190 const auto& cube = ReferenceElements<ctype, mydimension>::cube();
191 auto subEntityRange = cube.subEntities(localEdgeIndex, 1, 2);
192 std::vector<std::size_t> subEntityVector{subEntityRange.begin(), subEntityRange.end()};
193
194 // We have one vertex that belongs to the hostgrid, and one new one
195 if (edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::HostNew or
196 edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::NewHost) {
197 std::size_t vertexLocalIndexWRTElement =
198 edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::HostNew ? subEntityVector[0]
199 : subEntityVector[1];
200 auto hostVertexId = globalIdSetParameterSpace.subId(ele, vertexLocalIndexWRTElement, 2);
201 edgeVertexIndices.emplace_back(
202 GlobalIdType{.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId});
203
204 // Now the new one
205 auto edgeGeo = edgeOfTrimmedElement.geometry.value();
206 auto newVertex = edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::HostNew
207 ? edgeGeo.corner(1)
208 : edgeGeo.corner(0);
209 edgeVertexIndices.emplace_back(idForTrimmedVertex(newVertex));
210 } else {
211 auto edgeGeo = edgeOfTrimmedElement.geometry.value();
212 edgeVertexIndices.emplace_back(idForTrimmedVertex(edgeGeo.corner(0)));
213 edgeVertexIndices.emplace_back(idForTrimmedVertex(edgeGeo.corner(1)));
214 }
215 };
216
217 auto addTrimmedEdge = [&](const typename ElementTrimData::EdgeInfo& edgeOfTrimmedElement) {
218 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::newId, .id = globalIdSet_->newFreeIndex()};
219 elementEdgeIndices.emplace_back(edgeId);
220
221 EntityInfo<1> edgeInfo{.indexInLvlStorage = entityContainer_.edgeCount.back()++,
222 .lvl = level,
223 .trimmed = true,
224 .id = edgeId,
225 .trimInfo = edgeOfTrimmedElement};
226 edgeInfo.edgeGeometries.emplace_back(indexSet.index(ele), edgeOfTrimmedElement.geometry.value());
227 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
228
229 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
230 for (auto c : Dune::range(2)) {
231 auto vertexId = idForTrimmedVertex(edgeOfTrimmedElement.geometry.value().corner(c));
232 edgeVertexIndices.emplace_back(vertexId);
233 }
234 };
235
236 if (eleTrimData.flag() == ElementTrimFlag::full or not Preferences::getInstance().reconstructTrimmedLocalGeometry()) {
237 for (auto localEdgeIndex : Dune::range(ele.subEntities(1))) {
238 addHostEdge(localEdgeIndex);
239 }
240 } else /* trimmed */ {
241 for (const auto& edgeOfTrimmedElement : eleTrimData.edges()) {
242 if (edgeOfTrimmedElement.isHost and not edgeOfTrimmedElement.isTrimmed) {
243 addHostEdge(Transformations::mapToDune(1, edgeOfTrimmedElement.idx));
244 } else if (edgeOfTrimmedElement.isHost and edgeOfTrimmedElement.isTrimmed) {
245 // This is a host Edge which is partially trimmed
246 addTrimmedHostEdge(Transformations::mapToDune(1, edgeOfTrimmedElement.idx), edgeOfTrimmedElement);
247 } else /* new Edge*/ {
248 addTrimmedEdge(edgeOfTrimmedElement);
249 }
250 }
251 }

254template <int dim, int dimworld, typename ScalarType>
255void ParameterSpaceImpl<dim, dimworld, ScalarType>::collectElementVertices(int level, const HostEntity<0>& ele,
256 const ElementTrimData& eleTrimData) {
257 using VertexGeo = typename ParameterSpaceTraits::template Codim<2>::TrimmedParameterSpaceGeometry::PatchGeometry;
258
259 GlobalIdType elementId = makeElementID(ele);
260 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
261
262 auto& elementVertexIndices = entityContainer_.globalVerticesIdOfElementsMap[elementId];
263 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
264
265 auto addHostVertex = [&](unsigned int localVertexIndex) {
266 auto hostVertexId = globalIdSetParameterSpace.subId(ele, localVertexIndex, 2);
267 GlobalIdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
268 elementVertexIndices.emplace_back(vertexId);
269
270 if (entityContainer_.idToVertexInfoMap.back().contains(vertexId)) {
271 return;
272 }
273
274 auto vertex = ele.template subEntity<2>(localVertexIndex);
275 EntityInfo<2> vertexInfo{.indexInLvlStorage = entityContainer_.vertexCount.back()++,
276 .lvl = level,
277 .trimmed = false,
278 .id = vertexId,
279 .hostSeed = vertex.seed()};
280 entityContainer_.idToVertexInfoMap.back().insert({vertexId, vertexInfo});
281 };
282
283 auto addNewVertex = [&](const typename ElementTrimData::VertexInfo& vertex) {
284 GlobalIdType vertexId = idForTrimmedVertex(vertex.geometry.value());
285 elementVertexIndices.emplace_back(vertexId);
286
287 if (entityContainer_.idToVertexInfoMap.back().contains(vertexId)) {
288 return;
289 }
290
291 EntityInfo<2> vertexInfo{.indexInLvlStorage = entityContainer_.vertexCount.back()++,
292 .lvl = level,
293 .trimmed = true,
294 .id = vertexId,
295 .trimInfo = vertex};
296 entityContainer_.idToVertexInfoMap.back().insert({vertexId, vertexInfo});
297 };
298
299 if (eleTrimData.flag() == ElementTrimFlag::full or not Preferences::getInstance().reconstructTrimmedLocalGeometry()) {
300 // Save eleVertices in local order
301 for (auto localVertexIndex : Dune::range(ele.subEntities(2))) {
302 addHostVertex(localVertexIndex);
303 }
304
305 } else /* trimmed */ {
306 for (auto& vertexInfo : eleTrimData.vertices()) {
307 if (vertexInfo.isHost) {
308 addHostVertex(Transformations::mapToDune(2, vertexInfo.idx));
309 } else /* new */ {
310 addNewVertex(vertexInfo);
311 }
312 }
313 }

316template <int dim, int dimworld, typename ScalarType>
317void ParameterSpaceImpl<dim, dimworld, ScalarType>::createElements(int level,
318 const std::vector<ElementTrimData>& trimDatas) {
319 using ElementEntity = typename ParameterSpaceTraits::template Codim<0>::ParameterSpaceGridEntity;
320
321 auto& elementMap = entityContainer_.idToElementInfoMap;
322 auto& elementContainer = std::get<0>(entityContainer_.entityImps_.back());
323
324 elementContainer.resize(entityContainer_.numberOfTrimmedElements.back() +
325 entityContainer_.numberOfUnTrimmedElements.back());
326 for (auto& [eleId, eleInfo] :
327 elementMap | std::ranges::views::filter([level](const auto& pair) { return pair.second.lvl == level; })) {
328 auto ele = parameterSpaceGrid_->entity(eleInfo.hostSeed);
329 if (not eleInfo.trimmed) {
330 elementContainer[eleInfo.indexInLvlStorage] = ElementEntity{grid_, ele, eleInfo};
331 } else {
332 elementContainer[eleInfo.indexInLvlStorage] =
333 ElementEntity{grid_, ele, eleInfo, trimDatas[eleInfo.hostIndexInLvl]};
334 }
335 }

338template <int dim, int dimworld, typename ScalarType>
339void ParameterSpaceImpl<dim, dimworld, ScalarType>::createSubEntities(int level) {
340 using EdgeEntity = typename ParameterSpaceTraits::template Codim<1>::ParameterSpaceGridEntity;
341 using VertexEntity = typename ParameterSpaceTraits::template Codim<2>::ParameterSpaceGridEntity;
342
343 // auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
344 // auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
345
346 auto& vertexMap = entityContainer_.idToVertexInfoMap.back();
347 auto& edgeMap = entityContainer_.idToEdgeInfoMap;
348
349 auto& vertexContainer = std::get<2>(entityContainer_.entityImps_.back());
350 auto& edgeContainer = std::get<1>(entityContainer_.entityImps_.back());
351
352 // we are resizing the containers, so we can add the entities in the indexSet order, the index is obtained from
353 // the infos (also this is a bit more efficient as pushing back all the time)
354
355 vertexContainer.resize(entityContainer_.vertexCount.back());
356 for (auto& [vertexId, vertexInfo] : vertexMap) {
357 if (not vertexInfo.trimmed or not Preferences::getInstance().reconstructTrimmedLocalGeometry()) {
358 auto vertex = parameterSpaceGrid_->entity(vertexInfo.hostSeed);
359 vertexContainer[vertexInfo.indexInLvlStorage] = VertexEntity{grid_, vertex, vertexInfo};
360 } else {
361 vertexContainer[vertexInfo.indexInLvlStorage] = VertexEntity(grid_, vertexInfo);
362 }
363 }
364
365 edgeContainer.resize(entityContainer_.edgeCount.back());
366 for (auto& [edgeId, edgeInfo] :
367 edgeMap | std::ranges::views::filter([level](const auto& pair) { return pair.second.lvl == level; })) {
368 if (not edgeInfo.trimmed or not Preferences::getInstance().reconstructTrimmedLocalGeometry()) {
369 auto edge = parameterSpaceGrid_->entity(edgeInfo.hostSeed);
370 edgeContainer[edgeInfo.indexInLvlStorage] = EdgeEntity{grid_, edge, edgeInfo};
371 } else {
372 edgeContainer[edgeInfo.indexInLvlStorage] = EdgeEntity{grid_, edgeInfo};
373 }
374 }
