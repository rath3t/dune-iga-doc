13 auto sameCorner(const auto& corner1, const auto& corner2, double precision = 1e-10) -> bool {
14 return std::ranges::all_of(Dune::range(2),
15 [&](auto i) -> bool { return Dune::FloatCmp::eq(corner1[i], corner2[i], precision); });
16 };
17
18 bool isSameEdgeGeometry(const auto& geo1, const auto& geo2) {
19 return sameCorner(geo1.corner(0), geo2.corner(0)) and sameCorner(geo1.corner(1), geo2.corner(1)) ||
20 sameCorner(geo1.corner(0), geo2.corner(1)) and sameCorner(geo1.corner(1), geo2.corner(0)) ||
21 sameCorner(geo1.corner(1), geo2.corner(0)) and sameCorner(geo1.corner(0), geo2.corner(1));
22 }

32template <int dim, int dimworld, typename ScalarType>
33auto TrimmerImpl<dim, dimworld, ScalarType>::idForTrimmedVertex(const FieldVector<double, 2>& vertex) -> GlobalIdType {
34 using PersistentIndexType = typename TrimmerTraits::PersistentIndexType;
35
36 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
37 auto& vertexIdContainer = entityContainer_.trimmedVertexIds_.back();
38 bool alreadyThere = false;
39 PersistentIndexType foundIndex{};
40
41 for (auto& [vertexId, pos] : vertexIdContainer) {
42 if (Util::sameCorner(vertex, pos)) {
43 alreadyThere = true;
44 foundIndex = vertexId.id;
45 }
46 }
47 if (not alreadyThere)
48 foundIndex = globalIdSet_->newFreeIndex();
49
50 GlobalIdType vertexId{.entityIdType = GlobalIdType::EntityIdType::newId, .id = foundIndex};
51 if (not alreadyThere)
52 vertexIdContainer.emplace(vertexId, vertex);
53 return vertexId;
54}

56template <int dim, int dimworld, typename ScalarType>
57auto TrimmerImpl<dim, dimworld, ScalarType>::idForTrimmedHostEdge(
58 typename TrimmerTraits::PersistentIndexType hostEdgeId,
59 const typename ElementTrimData::EdgeInfo& trimmedEdge) -> GlobalIdType {
60 using HostEdge = typename TrimmerTraits::template Codim<1>::HostParameterSpaceGridEntity;
61 using PersistentIndexType = typename TrimmerTraits::PersistentIndexType;
62
63 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
64
65 // We have to check weather the edgeId is already there
66 bool alreadyThere = false;
67 PersistentIndexType foundIndex{};
68 for (auto& [edgeId, edgeInfo] : entityContainer_.idToEdgeInfoMap) {
69 if (edgeId.hostId.has_value() and edgeInfo.id.hostId.value() == hostEdgeId) {
70 // We have the same hostId, this doens't neccesarily mean that its the same edge
71 auto edgeGeo = edgeInfo.edgeGeometries.front().geometry;
72 if (Util::isSameEdgeGeometry(edgeGeo, trimmedEdge.geometry.value())) {
73 alreadyThere = true;
74 foundIndex = edgeId.id;
75 break;
76 }
77 }
78 }
79 if (not alreadyThere)
80 foundIndex = globalIdSet_->newFreeIndex();
81 GlobalIdType edgeId = {
82 .entityIdType = GlobalIdType::EntityIdType::newId, .id = foundIndex, .hostId = std::make_optional(hostEdgeId)};
83 return edgeId;
84}

86template <int dim, int dimworld, typename ScalarType>
87void TrimmerImpl<dim, dimworld, ScalarType>::createAndSaveElementInfo(
88 const std::tuple<unsigned int, unsigned int, int>& indices, const HostEntity<0>& ele, bool trimmed) {
89 const unsigned int unTrimmedElementIndex = std::get<0>(indices);
90 const unsigned int trimmedElementIndex = std::get<1>(indices);
91 const int newLevel = std::get<2>(indices);
92
93 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
94 auto& indexSet = parameterSpaceGrid_->levelGridView(newLevel).indexSet();
95
96 std::optional<GlobalIdType> fatherId{};
97 if (ele.hasFather())
98 fatherId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = globalIdSetParameterSpace.id(ele.father())};
99
100 GlobalIdType elementId = makeElementID(ele);
101 auto hostIndexInLvl = untrimmedParameterSpaceGrid_->levelGridView(newLevel).indexSet().index(
102 parameterSpaceGrid_->template getHostEntity<0>(ele));
103
104 EntityInfo<0> elementInfo = {
105 .indexInLvlStorage = trimmedElementIndex + unTrimmedElementIndex,
106 .unTrimmedIndexInLvl = unTrimmedElementIndex,
107 .trimmedIndexInLvl = trimmedElementIndex,
108 .hostIndexInLvl = hostIndexInLvl,
109 .lvl = newLevel,
110 .trimmed = trimmed,
111 .id = elementId,
112 .hostSeed = ele.seed(),
113 .fatherId = fatherId,
114 };
115 assert(trimmedElementIndex + unTrimmedElementIndex == indexSet.index(ele));
116
117 entityContainer_.idToElementInfoMap.insert({elementId, elementInfo});
118
119 // If we have a father we have to add us as his son, this can be faster, we can store in decendantIds,
120 // the indexInLvlStorage and lvl, which would provide faster access
121 if (fatherId.has_value()) {
122 entityContainer_.idToElementInfoMap.at(fatherId.value()).decendantIds.push_back(elementId);
123 entityContainer_.template entity<0>(fatherId.value(), newLevel - 1).entityInfo_.decendantIds.push_back(elementId);
124 }
125}

127template <int dim, int dimworld, typename ScalarType>
128void TrimmerImpl<dim, dimworld, ScalarType>::collectElementEdges(int level, const HostEntity<0>& ele,
129 const ElementTrimData& eleTrimData) {
130 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
131 GlobalIdType elementId = makeElementID(ele);
132
133 auto& elementEdgeIndices = entityContainer_.globalEdgesIdOfElementsMap_[elementId];
134 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
135
136 // Define some lambdas (can be moved to seperate functions later)
137 auto addHostEdge = [&](const unsigned int localEdgeIndex) {
138 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
139 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostEdgeId};
140 elementEdgeIndices.emplace_back(edgeId);
141
142 if (entityContainer_.idToEdgeInfoMap.contains(edgeId))
143 return;
144
145 auto edge = ele.template subEntity<1>(localEdgeIndex);
146 EntityInfo<1> edgeInfo{.indexInLvlStorage = entityContainer_.edgeCount.back()++,
147 .lvl = level,
148 .trimmed = false,
149 .id = edgeId,
150 .hostSeed = edge.seed()};
151 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
152
153 // store vertex ids of edges, for subIndex method of indexSet
154 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
155
156 const auto& cube = ReferenceElements<ctype, mydimension>::cube();
157 for (auto vertexLocalIndexWRTElement : cube.subEntities(localEdgeIndex, 1, 2)) {
158 auto hostVertexId = globalIdSetParameterSpace.subId(ele, vertexLocalIndexWRTElement, 2);
159 GlobalIdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
160 edgeVertexIndices.emplace_back(vertexId);
161 }
162 };
163
164 auto addTrimmedHostEdge = [&](int localEdgeIndex, const typename ElementTrimData::EdgeInfo& edgeOfTrimmedElement) {
165 // To get neighborhoud information we save the original hostId
166 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
167 GlobalIdType edgeId = idForTrimmedHostEdge(hostEdgeId, edgeOfTrimmedElement);
168 elementEdgeIndices.emplace_back(edgeId);
169
170 if (entityContainer_.idToEdgeInfoMap.contains(edgeId)) {
171 entityContainer_.idToEdgeInfoMap.at(edgeId).edgeGeometries.emplace_back(indexSet.index(ele),
172 edgeOfTrimmedElement.geometry.value());
173 return;
174 }
175
176 auto edge = ele.template subEntity<1>(localEdgeIndex);
177 EntityInfo<1> edgeInfo{.indexInLvlStorage = entityContainer_.edgeCount.back()++,
178 .lvl = level,
179 .trimmed = true,
180 .id = edgeId,
181 .hostSeed = edge.seed(),
182 .trimInfo = edgeOfTrimmedElement};
183 edgeInfo.edgeGeometries.emplace_back(indexSet.index(ele), edgeOfTrimmedElement.geometry.value());
184 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
185
186 // store vertex ids of edges, for subIndex method of indexSet
187 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
188
189 const auto& cube = ReferenceElements<ctype, mydimension>::cube();
190 auto subEntityRange = cube.subEntities(localEdgeIndex, 1, 2);
191 std::vector<std::size_t> subEntityVector{subEntityRange.begin(), subEntityRange.end()};
192
193 // We have one vertex that belongs to the hostgrid, and one new one
194 if (edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::HostNew or
195 edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::NewHost) {
196 std::size_t vertexLocalIndexWRTElement =
197 edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::HostNew ? subEntityVector[0]
198 : subEntityVector[1];
199 auto hostVertexId = globalIdSetParameterSpace.subId(ele, vertexLocalIndexWRTElement, 2);
200 edgeVertexIndices.emplace_back(
201 GlobalIdType{.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId});
202
203 // Now the new one
204 auto edgeGeo = edgeOfTrimmedElement.geometry.value();
205 auto newVertex = edgeOfTrimmedElement.direction == ElementTrimData::TrimmedHostEdgeDirection::HostNew
206 ? edgeGeo.corner(1)
207 : edgeGeo.corner(0);
208 edgeVertexIndices.emplace_back(idForTrimmedVertex(newVertex));
209 } else {
210 auto edgeGeo = edgeOfTrimmedElement.geometry.value();
211 edgeVertexIndices.emplace_back(idForTrimmedVertex(edgeGeo.corner(0)));
212 edgeVertexIndices.emplace_back(idForTrimmedVertex(edgeGeo.corner(1)));
213 }
214 };
215
216 auto addTrimmedEdge = [&](const typename ElementTrimData::EdgeInfo& edgeOfTrimmedElement) {
217 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::newId, .id = globalIdSet_->newFreeIndex()};
218 elementEdgeIndices.emplace_back(edgeId);
219
220 EntityInfo<1> edgeInfo{.indexInLvlStorage = entityContainer_.edgeCount.back()++,
221 .lvl = level,
222 .trimmed = true,
223 .id = edgeId,
224 .trimInfo = edgeOfTrimmedElement};
225 edgeInfo.edgeGeometries.emplace_back(indexSet.index(ele), edgeOfTrimmedElement.geometry.value());
226 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
227
228 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
229 for (auto c : Dune::range(2)) {
230 auto vertexId = idForTrimmedVertex(edgeOfTrimmedElement.geometry.value().corner(c));
231 edgeVertexIndices.emplace_back(vertexId);
232 }
233 };
234
235 if (eleTrimData.flag() == ElementTrimFlag::full) {
236 for (auto localEdgeIndex : Dune::range(ele.subEntities(1))) {
237 addHostEdge(localEdgeIndex);
238 }
239 } else /* trimmed */ {
240 for (const auto& edgeOfTrimmedElement : eleTrimData.edges()) {
241 if (edgeOfTrimmedElement.isHost and not edgeOfTrimmedElement.isTrimmed) {
242 addHostEdge(Transformations::mapToDune(1, edgeOfTrimmedElement.idx));
243 } else if (edgeOfTrimmedElement.isHost and edgeOfTrimmedElement.isTrimmed) {
244 // This is a host Edge which is partially trimmed
245 addTrimmedHostEdge(Transformations::mapToDune(1, edgeOfTrimmedElement.idx), edgeOfTrimmedElement);
246 } else /* new Edge*/ {
247 addTrimmedEdge(edgeOfTrimmedElement);
248 }
249 }
250 }

253template <int dim, int dimworld, typename ScalarType>
254void TrimmerImpl<dim, dimworld, ScalarType>::collectElementVertices(int level, const HostEntity<0>& ele,
255 const ElementTrimData& eleTrimData) {
256 using VertexGeo = typename TrimmerTraits::template Codim<2>::TrimmedParameterSpaceGeometry::PatchGeometry;
257
258 GlobalIdType elementId = makeElementID(ele);
259 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
260
261 auto& elementVertexIndices = entityContainer_.globalVerticesIdOfElementsMap[elementId];
262 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
263
264 auto addHostVertex = [&](unsigned int localVertexIndex) {
265 auto hostVertexId = globalIdSetParameterSpace.subId(ele, localVertexIndex, 2);
266 GlobalIdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
267 elementVertexIndices.emplace_back(vertexId);
268
269 if (entityContainer_.idToVertexInfoMap.back().contains(vertexId)) {
270 return;
271 }
272
273 auto vertex = ele.template subEntity<2>(localVertexIndex);
274 EntityInfo<2> vertexInfo{.indexInLvlStorage = entityContainer_.vertexCount.back()++,
275 .lvl = level,
276 .trimmed = false,
277 .id = vertexId,
278 .hostSeed = vertex.seed()};
279 entityContainer_.idToVertexInfoMap.back().insert({vertexId, vertexInfo});
280 };
281
282 auto addNewVertex = [&](const typename ElementTrimData::VertexInfo& vertex) {
283 GlobalIdType vertexId = idForTrimmedVertex(vertex.geometry.value());
284 elementVertexIndices.emplace_back(vertexId);
285
286 if (entityContainer_.idToVertexInfoMap.back().contains(vertexId)) {
287 return;
288 }
289
290 EntityInfo<2> vertexInfo{.indexInLvlStorage = entityContainer_.vertexCount.back()++,
291 .lvl = level,
292 .trimmed = true,
293 .id = vertexId,
294 .trimInfo = vertex};
295 entityContainer_.idToVertexInfoMap.back().insert({vertexId, vertexInfo});
296 };
297
298 if (eleTrimData.flag() == ElementTrimFlag::full) {
299 // Save eleVertices in local order
300 for (auto localVertexIndex : Dune::range(ele.subEntities(2))) {
301 addHostVertex(localVertexIndex);
302 }
303
304 } else /* trimmed */ {
305 for (auto& vertexInfo : eleTrimData.vertices()) {
306 if (vertexInfo.isHost) {
307 addHostVertex(Transformations::mapToDune(2, vertexInfo.idx));
308 } else /* new */ {
309 addNewVertex(vertexInfo);
310 }
311 }
312 }

315template <int dim, int dimworld, typename ScalarType>
316void TrimmerImpl<dim, dimworld, ScalarType>::createElements(int level, const std::vector<ElementTrimData>& trimDatas) {
317 using ElementEntity = typename TrimmerTraits::template Codim<0>::ParameterSpaceGridEntity;
318
319 auto& elementMap = entityContainer_.idToElementInfoMap;
320 auto& elementContainer = std::get<0>(entityContainer_.entityImps_.back());
321
322 elementContainer.resize(entityContainer_.numberOfTrimmedElements.back() +
323 entityContainer_.numberOfUnTrimmedElements.back());
324 for (auto& [eleId, eleInfo] :
325 elementMap | std::ranges::views::filter([level](const auto& pair) { return pair.second.lvl == level; })) {
326 auto ele = parameterSpaceGrid_->entity(eleInfo.hostSeed);
327 if (not eleInfo.trimmed) {
328 elementContainer[eleInfo.indexInLvlStorage] = ElementEntity{grid_, ele, eleInfo};
329 } else {
330 elementContainer[eleInfo.indexInLvlStorage] =
331 ElementEntity{grid_, ele, eleInfo, trimDatas[eleInfo.hostIndexInLvl]};
332 }
333 }

336template <int dim, int dimworld, typename ScalarType>
337void TrimmerImpl<dim, dimworld, ScalarType>::createSubEntities(int level) {
338 using EdgeEntity = typename TrimmerTraits::template Codim<1>::ParameterSpaceGridEntity;
339 using VertexEntity = typename TrimmerTraits::template Codim<2>::ParameterSpaceGridEntity;
340
341 // auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
342 // auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
343
344 auto& vertexMap = entityContainer_.idToVertexInfoMap.back();
345 auto& edgeMap = entityContainer_.idToEdgeInfoMap;
346
347 auto& vertexContainer = std::get<2>(entityContainer_.entityImps_.back());
348 auto& edgeContainer = std::get<1>(entityContainer_.entityImps_.back());
349
350 // we are resizing the containers, so we can add the entities in the indexSet order, the index is obtained from
351 // the infos (also this is a bit more efficient as pushing back all the time)
352
353 vertexContainer.resize(entityContainer_.vertexCount.back());
354 for (auto& [vertexId, vertexInfo] : vertexMap) {
355 if (not vertexInfo.trimmed) {
356 auto vertex = parameterSpaceGrid_->entity(vertexInfo.hostSeed);
357 vertexContainer[vertexInfo.indexInLvlStorage] = VertexEntity{grid_, vertex, vertexInfo};
358 } else {
359 vertexContainer[vertexInfo.indexInLvlStorage] = VertexEntity(grid_, vertexInfo);
360 }
361 }
362
363 edgeContainer.resize(entityContainer_.edgeCount.back());
364 for (auto& [edgeId, edgeInfo] :
365 edgeMap | std::ranges::views::filter([level](const auto& pair) { return pair.second.lvl == level; })) {
366 if (not edgeInfo.trimmed) {
367 auto edge = parameterSpaceGrid_->entity(edgeInfo.hostSeed);
368 edgeContainer[edgeInfo.indexInLvlStorage] = EdgeEntity{grid_, edge, edgeInfo};
369 } else {
370 edgeContainer[edgeInfo.indexInLvlStorage] = EdgeEntity{grid_, edgeInfo};
371 }
372 }
