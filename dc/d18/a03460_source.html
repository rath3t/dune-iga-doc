46 template <class GridImp, IntersectionType type_>
47 struct TrimmedIntersectionImpl
48 {
49 constexpr static int dim = GridImp::dimension;
50 constexpr static int mydim = GridImp::dimension - 1;
51 constexpr static int dimworld = GridImp::dimension;
52 using ctype = typename GridImp::ctype;
53 using IdType = typename GridImp::GridFamily::TrimmerTraits::GlobalIdSetId;
54 using NormalVector = FieldVector<ctype, dim>;
55 using LocalCoordinate = FieldVector<ctype, mydim>;
56 using EdgeInfo = typename GridImp::Trimmer::TrimmerTraits::template Codim<1>::EntityInfo;
57
58 using IntersectionGeometry = GeometryKernel::NURBSPatch<mydim, dim, ctype>;
59 using HostLeafIntersection = IntersectionTraits::HostIntersection<GridImp, type_>;
60 using ParameterSpaceGridEntity = IntersectionTraits::ParameterSpaceGridEntity<GridImp>;
61 using LocalGeometry = IntersectionTraits::LocalGeometry<GridImp>;
62 using Geometry = IntersectionTraits::Geometry<GridImp>;
63 using TrimmedLocalGeometry = typename GridImp::GridFamily::TrimmerTraits::template Codim<1>::TrimmedLocalGeometry;
64 using TrimmedGeometry =
65 typename GridImp::GridFamily::TrimmerTraits::template Codim<1>::TrimmedParameterSpaceGeometry;
66
67 TrimmedIntersectionImpl() = default;
68
69 TrimmedIntersectionImpl(const GridImp* patchGrid, const IdType& insideElementId, const EdgeInfo& edgeInfo,
70 int indexInInside)
71 : patchGrid_(patchGrid),
72 edgeInfo_(edgeInfo),
73 insideElementId_(insideElementId),
74 geo_(edgeInfo_.geometryForIdx(
75 patchGrid_->trimmer().entityContainer_.idToElementInfoMap.at(insideElementId_).indexInLvlStorage)),
76 indexInInside_(indexInInside) {
77 assert(geo_.domain()[0].isUnitDomain());
78 }
79
80 ParameterSpaceGridEntity inside() const {
81 return patchGrid_->trimmer().entityContainer_.template entity<0>(insideElementId_, level());
82 }
83
84 ParameterSpaceGridEntity outside() const {
85 DUNE_THROW(GridError, "No Outside Entities for trimmed Intersections");
86 }
87
88 bool boundary() const {
89 return true;
90 }
91
92 bool neighbor() const {
93 return false;
94 }
95
96 size_t boundarySegmentIndex() const {
97 return 0;
98 }
99
100 // A boundary has to be conforming
101 bool conforming() const {
102 return true;
103 }
104
105 GeometryType type() const {
106 return GeometryTypes::none(mydim);
107 }
108
109 LocalGeometry geometryInInside() const {
110 auto localGeo = GeometryKernel::transformToSpan(geo_, inside().getHostEntity().geometry());
111 return LocalGeometry(TrimmedLocalGeometry(localGeo));
112 }
113
114 LocalGeometry geometryInOutside() const {
115 DUNE_THROW(GridError, "No Outside Entities for trimmed Intersections");
116 }
117
118 Geometry geometry() const {
119 return Geometry(TrimmedGeometry(geo_));
120 }
121
122 int indexInInside() const {
123 return indexInInside_;
124 }
125
126 int indexInOutside() const {
127 DUNE_THROW(GridError, "No Outside Entities for trimmed Intersections");
128 }
129
130 NormalVector outerNormal(const LocalCoordinate& local) const {
131 auto jacobian = geo_.jacobianTransposed(local)[0];
132 return {jacobian[1], jacobian[0]};
133 }
134
135 NormalVector integrationOuterNormal(const LocalCoordinate& local) const {
136 return unitOuterNormal(local) * geo_.integrationElement(local);
137 }
138
139 NormalVector unitOuterNormal(const LocalCoordinate& local) const {
140 auto N = outerNormal(local);
141 return N / N.two_norm();
142 }
143
144 NormalVector centerUnitOuterNormal() const {
145 return unitOuterNormal(0.5);
146 }
147
148 bool equals(const TrimmedIntersectionImpl& other) const {
149 return geo_ == other.geo_ and indexInInside_ == other.indexInInside_;
150 }
151
152 private:
153 int level() const {
154 if constexpr (type_ == IntersectionType::Leaf)
155 return patchGrid_->maxLevel();
156 else
157 return edgeInfo_.lvl;
158 }
159
160 const GridImp* patchGrid_{};
161 EdgeInfo edgeInfo_;
162 IdType insideElementId_;
163 IntersectionGeometry geo_;
164 int indexInInside_;
165 };
