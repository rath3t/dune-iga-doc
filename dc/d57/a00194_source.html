73template <class Grid>
74requires(IsSpecializationTwoNonTypesTemplateAndType<Dune::IGA::PatchGrid, Grid>::value)
75inline static std::shared_ptr<Grid> reader(const pybind11::dict& dict) {
76 std::string file_path;
77 Dune::Python::IGA::Reader reader = IGA::Reader::json;
78 bool trim = true;
79 std::array<int, 2> degreeElevate = {0, 0};
80 std::array<int, 2> preKnot = {0, 0};
81 std::array<int, 2> postKnot = {0, 0};
82 if (dict.contains("reader"))
83 reader = dict["reader"].cast<Dune::Python::IGA::Reader>();
84
85 switch (reader) {
86 case IGA::Reader::json: {
87 if (dict.contains("file_path"))
88 file_path = dict["file_path"].cast<std::string>();
89 else
90 DUNE_THROW(Dune::IOError, "No field in dict with name file_path. Unable to read grid");
91 if (dict.contains("trim"))
92 trim = dict["trim"].cast<bool>();
93 if (dict.contains("degree_elevate"))
94 degreeElevate = dict["degree_elevate"].cast<std::array<int, 2>>();
95 if (dict.contains("pre_knot_refine"))
96 preKnot = dict["pre_knot_refine"].cast<std::array<int, 2>>();
97 if (dict.contains("post_knot_refinement"))
98 postKnot = dict["post_knot_refinement"].cast<std::array<int, 2>>();
99
100 using ScalarType = typename Grid::ctype;
101 using GridFactory = Dune::GridFactory<Grid>;
102
103 auto gridFactory = GridFactory();
104 gridFactory.insertJson(file_path, trim, preKnot, degreeElevate, postKnot);
105
106 return gridFactory.createGrid();
107 }
108 default:
109 DUNE_THROW(Dune::NotImplemented, "Your requested reader is not implemented");
110 }
111}
