19 auto ordersFromDegrees(const std::array<int, dim>& degree) {
20 std::array<int, dim> order;
21 std::ranges::transform(degree, order.begin(), [](const auto& p) { return p + 1; });
22 return order;

25 template <std::integral auto dim>
26 void createPartialSubDerivativPermutations(const FieldVector<int, dim>& v, std::vector<FieldVector<int, dim>>& perm) {
27 perm.resize(0);
28 for (int i = 1; i < std::pow(2, v.size()); ++i) {
29 FieldVector<int, dim> x{};
30 for (int j = 0; j < v.size(); ++j)
31 if ((i & (1 << j)) != 0) {
32 x[j] = v[j];
33 if (v[j] == 0)
34 goto outer;
35 }
36 perm.push_back(x);
37 outer:;
38 }
39 }

41 template <Concept::Vector VectorType>
42 int binom(const VectorType& n, const VectorType& k) {
43 return std::inner_product(n.begin(), n.end(), k.begin(), 1, std::multiplies{},
44 [](auto& ni, auto& ki) { return Dune::binomial(ni, ki); });

63template <std::floating_point ScalarType, std::integral auto dim>
64auto generateRefinedKnots(const std::array<std::vector<ScalarType>, dim>& knotSpans, const int dir,
65 const int refinementLevel) {
66 const int newKnotsSizeForEachSpan = Dune::power(2, refinementLevel);
67 auto unique_Knots = knotSpans;
68 std::vector<double> additionalKnots;
69 auto& unique_KnotPerDim = unique_Knots[dir];
70 unique_KnotPerDim.erase(std::ranges::begin(std::ranges::unique(unique_KnotPerDim)), std::end(unique_KnotPerDim));
71
72 for (int i = 0; i < unique_KnotPerDim.size() - 1; ++i) {
73 const double spanLength = unique_KnotPerDim[i + 1] - unique_KnotPerDim[i];
74 const double increment = spanLength / newKnotsSizeForEachSpan;
75 for (int j = 1; j < newKnotsSizeForEachSpan; ++j) {
76 additionalKnots.emplace_back(unique_KnotPerDim[i] + increment * j);
77 }
78 }
79 return additionalKnots;
80}

94template <std::integral auto dim, std::integral auto dim2, typename NetValueType>
95requires((std::floating_point<NetValueType> || Concept::Vector<NetValueType> ||
96 is_instantiation_of<ControlPoint, NetValueType>::value) and
97 (dim == dim2))
98auto netOfSpan(std::array<int, dim> subNetStart, const std::array<int, dim>& degree,
99 const MultiDimensionalNet<dim2, NetValueType>& net) {
100 std::array<int, dim> order = Impl::ordersFromDegrees(degree);
101 for (std::size_t i = 0; i < dim; ++i)
102 subNetStart[i] -= degree[i];
103 return net.subNet(subNetStart, order);
104}

307 static auto basisFunctions(const FieldVector<ScalarType, dim>& u,
308 const std::array<std::vector<ScalarType>, dim>& knots, const std::array<int, dim>& degree,
309 const MultiDimensionalNet<dim, ScalarType>& weights, std::array<int, dim> spIndex) {
310 std::array<DynamicVectorType, (size_t)dim> bSplines;
311
312 for (std::size_t i = 0; i < dim; ++i)
313 bSplines[i] = BsplineBasis<ScalarType>::basisFunctions(u[i], knots[i], degree[i], spIndex[i]);
314
315 auto Nnet = MultiDimensionalNet<dim, ScalarType>(bSplines);
316 // @todo Alex this should also be cached when called with a localView
317 const auto subNetWeights = netOfSpan(spIndex, degree, weights);
318
319 const ScalarType invSumWeight = dot(Nnet, subNetWeights);
320 Nnet *= subNetWeights;
321 Nnet /= invSumWeight;
322 return Nnet;
323 }

335 static auto basisFunctionDerivatives(const FieldVector<ScalarType, dim>& u,
336 const std::array<std::vector<ScalarType>, dim>& knots,
337 const std::array<int, dim>& degree,
338 const MultiDimensionalNet<dim, double>& weights, const int derivativeOrder,
339 std::optional<std::array<int, dim>> spIndex = std::nullopt) {
340 std::array<DynamicMatrixType, dim> bSplineDerivatives;
341 std::array<int, dim> spanIndex = spIndex ? spIndex.value() : findSpan(degree, u, knots);
342
343 for (int i = 0; i < dim; ++i)
344 bSplineDerivatives[i] =
345 BsplineBasis<ScalarType>::basisFunctionDerivatives(u[i], knots[i], degree[i], derivativeOrder, spanIndex[i]);
346
347 std::array<std::vector<ScalarType>, dim> dimArrayOfVectors;
348 FieldVector<int, dim> dimSize(derivativeOrder + 1);
349 MultiDimensionalNet<dim, MultiDimensionalNet<dim, ScalarType>> netOfDerivativeNets(dimSize);
350
351 for (int j = 0; auto& derivNet : netOfDerivativeNets.directGetAll()) {
352 auto multiIndex = netOfDerivativeNets.directToMultiIndex(j++);
353 for (int i = 0; i < multiIndex.size(); ++i)
354 dimArrayOfVectors[i] = bSplineDerivatives[i][multiIndex[i]].container();
355 derivNet = MultiDimensionalNet<dim, ScalarType>(dimArrayOfVectors);
356 }
357
358 MultiDimensionalNet<dim, MultiDimensionalNet<dim, ScalarType>> R = netOfDerivativeNets;
359 MultiDimensionalNet<dim, ScalarType> netsOfWeightfunctions(dimSize);
360 const auto subNetWeights = netOfSpan(spanIndex, degree, weights);
361
362 for (int j = 0; j < R.size(); ++j) {
363 R.directGet(j) *= subNetWeights;
364 netsOfWeightfunctions.directGet(j) = dot(netOfDerivativeNets.directGet(j), subNetWeights);
365 }
366
367 std::vector<FieldVector<int, dim>> perms;
368 // The following is inefficient, it computes too much derivative directions and recycles nothing
369 for (int j = 0; j < R.size(); ++j) {
370 const auto derivOrders = R.template directToMultiIndex<FieldVector<int, dim>>(j);
371 Impl::createPartialSubDerivativPermutations(derivOrders, perms);
372
373 for (const auto& perm : perms) {
374 const MultiDimensionalNetIndex<dim> kNet(perm + FieldVector<int, dim>(1));
375 auto startMultiIndex = perm;
376 std::ranges::transform(startMultiIndex, startMultiIndex.begin(), [](const auto& v) { return (v != 0); });
377 for (int kk = kNet.index(startMultiIndex); kk < kNet.size(); ++kk) {
378 const auto multik = kNet.template directToMultiIndex<FieldVector<int, dim>>(kk);
379 const ScalarType fac = (Impl::binom(perm, multik) * netsOfWeightfunctions.get(multik));
380 const auto& Rdirect_d = R.get(derivOrders - multik);
381 auto& Rdirect = R.directGet(j);
382 for (int i = 0; i < R.directGet(j).size(); ++i)
383 Rdirect.directGet(i) -= fac * Rdirect_d.directGet(i); // generalized Piegl Tiller (4.20)
384 }
385 }
386 R.directGet(j) /= netsOfWeightfunctions.directGet(0);
387 }
388 return R;
389 }

440template <std::integral auto dim, std::integral auto dimworld, typename ScalarType>
441auto degreeElevate(const NURBSPatchData<dim, dimworld, ScalarType>& oldData, const int elevationDirection,
442 const int elevationFactor) {
443 using DynamicMatrixType = DynamicMatrix<ScalarType>;
444 using NURBSPatchData = NURBSPatchData<dim, dimworld, ScalarType>;
445 using ControlPointNetType = typename NURBSPatchData::ControlPointNetType;
446 using ControlPointType = typename NURBSPatchData::ControlPointType;
447 const int t = elevationFactor;
448 const int p = oldData.degree[elevationDirection];
449 const int m = oldData.controlPoints.strideSizes()[elevationDirection] + p;
450 const int ph = p + t;
451 const int ph2 = ph / 2;
452 /* Compute Bezier degree elevation coefficients */
453 DynamicMatrixType bezalfs(p + t + 1, p + 1);
454 bezalfs[0][0] = bezalfs[ph][p] = 1.0;
455 for (int i = 1; i <= ph2; ++i) {
456 const ScalarType inv = 1.0 / binomial(ph, i);
457 for (int j = std::max(0, i - t); j <= std::min(p, i); ++j)
458 bezalfs[i][j] = inv * binomial(p, j) * binomial(t, i - j);
459 }
460 for (int i = ph2 + 1; i <= ph - 1; ++i)
461 for (int j = std::max(0, i - t); j <= std::min(p, i); ++j)
462 bezalfs[i][j] = bezalfs[ph - i][p - j];
463
464 ControlPointNetType newCPv(oldData.controlPoints.strideSizes());
465
466 auto oldCPs = oldData.controlPoints;
467
468 auto scaleCPWithW = [](const auto& cp) -> ControlPointType { return {.p = cp.w * cp.p, .w = cp.w}; };
469 std::ranges::transform(oldCPs.directGetAll(), oldCPs.directGetAll().begin(), scaleCPWithW);
470
471 std::vector<ControlPointType> bpts(p + 1), ebpts(ph + 1), nextbpts(p - 1);
472
473 NURBSPatchData newData;
474 newData.degree = oldData.degree;
475 newData.degree[elevationDirection] += t;
476 const auto& U = oldData.knotSpans[elevationDirection];
477 double ua = U[0];
478 std::vector<ScalarType> Uh;
479 for (int j = 0, i = 0; i < U.size(); ++i) // insert knot t times for each unique knot
480 if (j < U.size()) {
481 do {
482 Uh.push_back(U[j]);
483 ++j;
484 } while (j != U.size() && FloatCmp::eq(U[j - 1], U[j]));
485 Uh.insert(Uh.end(), t, U[j - 1]);
486 if (j == U.size())
487 break;
488 }
489
490 for (int i = 0; i < dim; ++i)
491 if (i == elevationDirection)
492 newData.knotSpans[i] = Uh;
493 else
494 newData.knotSpans[i] = oldData.knotSpans[i];
495
496 auto dimSize = oldData.controlPoints.strideSizes();
497 dimSize[elevationDirection] = Uh.size() - ph - 1;
498 newData.controlPoints = MultiDimensionalNet<dim, ControlPointType>(dimSize);
499 auto& newCPs = newData.controlPoints;
500
501 int totalCurvesInPatch = 1;
502 std::array<int, dim - 1> indicesOther;
503 for (int counter = 0, i = 0; i < dim; ++i)
504 if (i != elevationDirection) {
505 totalCurvesInPatch *= dimSize[i];
506 indicesOther[counter++] = dimSize[i];
507 }
508
509 auto newAndOldCurve = [&elevationDirection, indicesOther, &oldCPs, &newCPs](auto i) {
510 const auto mI =
511 MultiDimensionalNet<dim - 1, int>::template directToMultiIndex<std::array<int, dim - 1>>(indicesOther, i);
512 std::array<int, dim> multiIndex{};
513 for (int c = 0, j = 0; j < dim; ++j)
514 if (j != elevationDirection)
515 multiIndex[j] = mI[c++];
516
517 auto oldCurve = [&oldCPs, &elevationDirection, multiIndex](int i) mutable {
518 multiIndex[elevationDirection] = i;
519 return oldCPs.get(multiIndex);
520 };
521
522 auto newCurve = [&newCPs, &elevationDirection, multiIndex](int i) mutable -> auto& {
523 multiIndex[elevationDirection] = i;
524 return newCPs.get(multiIndex);
525 };
526 return std::make_pair(oldCurve, newCurve);
527 };
528
529 for (auto [oldCurve, newCurve] :
530 std::ranges::iota_view(0, totalCurvesInPatch) | std::views::transform(newAndOldCurve)) {
531 ua = U[0];
532 int kind = ph + 1;
533 int r = -1;
534 int a = p;
535 int b = p + 1;
536 int cind = 1;
537
538 newCurve(0) = oldCurve(0);
539 for (int i = 0; i <= p; ++i)
540 bpts[i] = oldCurve(i);
541
542 std::vector<ScalarType> alphas(p - 1);
543 while (b < m) {
544 const int bold = b;
545 while (b < m && FloatCmp::eq(U[b], U[b + 1]))
546 ++b;
547 const int mul = b - bold + 1;
548 const auto ub = U[b];
549 const int oldr = r;
550 r = p - mul;
551 const int lbz = (oldr > 0) ? ((oldr + 2) / 2) : 1; // Insert knot u(b) r times
552 const int rbz = (r > 0) ? (ph - (r + 1) / 2) : ph;
553 if (r > 0) // Insert knot to get Bezier segment
554 {
555 const auto numer = ub - ua;
556 for (int k = p; k > mul; --k)
557 alphas[k - mul - 1] = numer / (U[a + k] - ua);
558 for (int j = 1; j <= r; ++j) {
559 const int s = mul + j;
560 for (int k = p; k >= s; --k)
561 bpts[k] = alphas[k - s] * bpts[k] + (1.0 - alphas[k - s]) * bpts[k - 1];
562 nextbpts[r - j] = bpts[p];
563 }
564 } // End of insert knot
565
566 for (int i = lbz; i <= ph; ++i) { // Degree elevate Bezier
567 ebpts[i].setZero();
568 for (int j = std::max(0, i - t); j <= std::min(p, i); ++j)
569 ebpts[i] += bezalfs[i][j] * bpts[j];
570 } // End of degree elevating Bezier
571 if (oldr > 1) { // Must remove knot u = U[a] oldr times
572 int first = kind - 2;
573 int last = kind;
574 const ScalarType den = ub - ua;
575 const ScalarType bet = (ub - Uh[kind - 1]) / den;
576 for (int tr = 1; tr < oldr; ++tr) { // Knot removal loop
577 int i = first;
578 int j = last;
579 int kj = j - kind + 1;
580 while (j - i > tr) {
581 if (i < cind) {
582 const ScalarType alf = (ub - Uh[i]) / (ua - Uh[i]);
583 newCurve(i) = alf * newCurve(i) + (1.0 - alf) * newCurve(i - 1);
584 }
585 if (j >= lbz) {
586 const auto factor = (j - tr <= kind - ph + oldr) ? (ub - Uh[j - tr]) / den : bet;
587 ebpts[kj] = factor * ebpts[kj] + (1.0 - factor) * ebpts[kj + 1];
588 }
589 ++i;
590 --j;
591 --kj;
592 }
593 --first;
594 ++last;
595 }
596 } // End of removing knot, u= U[a]
597 if (a != p)
598 kind += ph - oldr;
599 for (int j = lbz; j <= rbz; ++j, ++cind) // Load ctrl pts into Qw
600 newCurve(cind) = ebpts[j];
601 if (b < m) { // Set up for next pass through loop
602 std::ranges::copy(nextbpts | std::views::take(r), bpts.begin());
603 for (int j = r; j <= p; ++j)
604 bpts[j] = oldCurve(b - p + j);
605 a = b;
606 ++b;
607 ua = ub;
608 }
609 } // End of while-loop (b<m)
610 }
611 std::ranges::transform(newData.controlPoints.directGetAll(), newData.controlPoints.directGetAll().begin(),
612 [](auto& cp) -> ControlPointType { return {.p = cp.p / cp.w, .w = cp.w}; });
613 return newData;
614}

627template <std::integral auto dim, std::integral auto dimworld, typename ScalarType>
628auto knotRefinement(const NURBSPatchData<dim, dimworld, ScalarType>& oldData, const std::vector<double>& newKnots,
629 const int refinementDirection) {
630 if (newKnots.empty())
631 return oldData;
632 assert(std::ranges::is_sorted(newKnots) &&
633 "You passed a non-sorted vector of new knots. Sort it first or check if you passed the correct vector.");
634 using NurbsPatchData = NURBSPatchData<dim, dimworld, ScalarType>;
635 using ControlPoint = typename NurbsPatchData::ControlPointType;
636 using namespace std::ranges;
637 std::array<std::vector<double>, dim> newKnotsArray;
638 std::array<int, dim - 1> otherDirections;
639 for (int counter = 0, i = 0; i < dim; ++i) {
640 if (i == refinementDirection)
641 continue;
642 newKnotsArray[i] = oldData.knotSpans[i];
643 otherDirections[counter++] = i;
644 }
645
646 typename NurbsPatchData::ControlPointNetType newCPv(oldData.controlPoints.strideSizes());
647
648 auto oldCPv = oldData.controlPoints;
649 auto oldKnotVec = oldData.knotSpans[refinementDirection];
650
651 auto& newKnotVec = newKnotsArray[refinementDirection];
652
653 // Combine old and new knot vectors
654 newKnotVec.resize(oldKnotVec.size() + newKnots.size());
655 merge(oldKnotVec, newKnots, std::begin(newKnotVec));
656
657 const auto newKSize = newKnots.size();
658 std::array<int, dim> numberOfCPperDir = oldCPv.strideSizes();
659 numberOfCPperDir[refinementDirection] += newKSize;
660
661 newCPv.resize(numberOfCPperDir);
662
663 auto scaleCPWithW = [](const auto& cp) -> ControlPoint { return {.p = cp.w * cp.p, .w = cp.w}; };
664
665 const int p = oldData.degree[refinementDirection];
666 const int a = findSpan(p, newKnots.front(), oldKnotVec);
667 const int b = findSpan(p, newKnots.back(), oldKnotVec);
668
669 auto newSurfI = newCPv.hyperSurfBegin(otherDirections);
670 auto oldSurfI = oldCPv.hyperSurfBegin(otherDirections);
671
672 for (auto hSOld = oldSurfI, hSNew = newSurfI; hSOld != oldCPv.hyperSurfBegin(otherDirections) + (a - p + 1);
673 ++hSOld, ++hSNew)
674 std::ranges::transform(*hSOld, (*hSNew).begin(), scaleCPWithW);
675 for (auto hSOld = oldSurfI + b, hSNew = newSurfI + b; hSOld != oldCPv.hyperSurfEnd(otherDirections); ++hSOld, ++hSNew)
676 std::ranges::transform(*hSOld, (*hSNew).begin(), scaleCPWithW);
677
678 newSurfI = newCPv.hyperSurfEnd(otherDirections) - 1;
679 oldSurfI = oldCPv.hyperSurfEnd(otherDirections) - 1;
680
681 auto currentNewKnot = newKnotVec.end();
682 auto currentOldKnot = oldKnotVec.end();
683
684 for (const auto& currentAdditionalKnot : reverse_view(newKnots)) {
685 while (currentAdditionalKnot <= *(currentOldKnot - 1) &&
686 std::distance(oldKnotVec.begin(), currentOldKnot) > a + 1) {
687 std::ranges::transform(*oldSurfI, (*newSurfI).begin(), scaleCPWithW);
688 --currentNewKnot, --currentOldKnot;
689 --newSurfI, --oldSurfI;
690 }
691
692 std::ranges::copy(*(newSurfI + 1), (*newSurfI).begin());
693 ++newSurfI;
694 currentOldKnot -= p;
695 for ([[maybe_unused]] int _ : iota_view{0, p}) {
696 ++newSurfI;
697
698 auto alpha = *currentNewKnot - currentAdditionalKnot;
699 using std::abs;
700 if (abs(alpha) < 1e-7)
701 std::ranges::copy(*newSurfI, (*(newSurfI - 1)).begin());
702 else {
703 alpha /= (*currentNewKnot - *(currentOldKnot));
704 std::ranges::transform(*newSurfI, *(newSurfI - 1), (*(newSurfI - 1)).begin(),
705 [&alpha](auto& cp, auto& cpL) { return alpha * cpL + cp * (1.0 - alpha); });
706 }
707 ++currentOldKnot, ++currentNewKnot;
708 }
709 currentNewKnot -= p + 1;
710 newSurfI -= p + 2;
711 }
712
713 std::ranges::transform(newCPv.directGetAll(), newCPv.directGetAll().begin(),
714 [](auto& cp) -> ControlPoint { return {.p = cp.p / cp.w, .w = cp.w}; });
715
