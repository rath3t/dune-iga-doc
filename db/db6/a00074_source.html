23template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
24class TrimmedLocalGeometryImpl<2, coorddim, GridImp, localGeometryTag>
25{
26public:
27 using ctype = typename GridImp::ctype;
28
29 static constexpr int mydimension = 2;
30 using ParameterSpace = typename GridImp::ParameterSpace;
31 using SimplexGeneratorImpl = SimplexGenerator<GridImp>;
32
33 static constexpr int coorddimension = coorddim;
34 static constexpr int codim = coorddimension - mydimension;
35 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
36 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
37
38 using HostGeometry =
39 typename ParameterSpace::ParameterSpaceTraits::template Codim<0>::HostParameterSpaceGridEntity::Geometry;
40 using TrimData = typename ParameterSpace::ElementTrimData;
41
42 using ParameterSpaceElement = typename GridImp::ParameterSpace::template Codim<0>::ParameterSpaceGridEntity;
43
44 using LocalCoordinate = FieldVector<ctype, mydimension>;
45 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
46 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
47 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
48 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
49 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
50 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
51 using Volume = ctype;
52
53 // type of the LocalView of the patch geometry
54 using GeometryLocalView =
55 typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
56 ctype>::template GeometryLocalView<codim, ParameterSpace>;
57
58 struct Triangulation
59 {
60 std::vector<typename SimplexGeneratorImpl::Element> elements{};
61 std::vector<typename SimplexGeneratorImpl::Point> points{};
62 std::vector<typename SimplexGeneratorImpl::Index> indices{};
63 };
64
66 TrimmedLocalGeometryImpl() = delete;
67 explicit TrimmedLocalGeometryImpl(const HostGeometry& hostGeometry, const ParameterSpaceElement& element)
68 : hostGeometry_{hostGeometry},
69 trimData_{element.trimData()},
70 element_(element) {
71 std::tie(triangulation_.elements, triangulation_.points, triangulation_.indices) =
72 SimplexGeneratorImpl::createSimplicies(element, {.boundaryDivisions = 5, .targetAccuracy = 2});
73 assert(triangulation_.elements.size() > 0);
74 }
75
76 bool operator==(const TrimmedLocalGeometryImpl& b) const {
77 return b.hostGeometry_ == this->hostGeometry_;
78 }
79
82 [[nodiscard]] GeometryType type() const {
83 if (Preferences::getInstance().reportTrimmedElementGeometryTypeAsNone())
84 return GeometryTypes::none(mydimension);
85 return GeometryTypes::cube(mydimension);
86 }
87
88 // return whether we have an affine mapping (true for straight lines)
89 [[nodiscard]] bool affine() const {
90 return true;
91 }
92
93 // return the number of corners of this element. Corners are numbered 0...n-1
94 [[nodiscard]] int corners() const {
95 return trimData_.size(2);
96 }
97
98 GlobalCoordinate center() const {
99 return global(GeometryKernel::centerOfMass(triangulation_.elements));
100 }
101
102 // access to coordinates of corners. Index is the number of the corner
103 GlobalCoordinate corner(int i) const {
104 auto vData = trimData_.vertex(i);
105 if (vData.isHost)
106 return hostGeometry_.corner(Transformations::mapToDune(2, vData.idx));
107
108 return vData.geometry.value();
109 }
110
113 GlobalCoordinate global(const LocalCoordinate& local) const {
114 return hostGeometry_.global(local);
115 }
116
119 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
120 return hostGeometry_.jacobianTransposed(local);
121 }
122
125 LocalCoordinate local(const GlobalCoordinate& global) const {
126 return hostGeometry_.local(global);
127 }
128
129 // Returns true if the point is in the current element
130 // TODO
131 bool checkInside(const LocalCoordinate& local) const {
132 return true;
133 }
134
135 // TODO
136 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
137 return hostGeometry_.volume();
138 }
139
140 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
141 return hostGeometry_.jacobianInverseTransposed(local);
142 }
143
144 auto getQuadratureRule(const std::optional<int>& p_order = std::nullopt,
145 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) const {
146 return element_.getQuadratureRule(p_order, qt);
147 }
148
149 bool isTrimmed() const {
150 return true;
151 }
152
153private:
154 HostGeometry hostGeometry_;
155 TrimData trimData_;
156 ParameterSpaceElement element_; // This can not be a ptr as the entity could live shorter than the geo, maybe only
157 // store a quadrature rule
158 Triangulation triangulation_;
159};

161template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
162class TrimmedLocalGeometryImpl<1, coorddim, GridImp, localGeometryTag>
163{
164public:
165 using ctype = typename GridImp::ctype;
166
167 static constexpr int mydimension = 1;
168 using ParameterSpace = typename GridImp::ParameterSpace;
169
170 static constexpr int coorddimension = coorddim;
171 static constexpr int codim = coorddimension - mydimension;
172 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
173 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
174 using LocalCoordinate = FieldVector<ctype, mydimension>;
175 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
176 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
177 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
178 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
179 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
180 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
181 using Volume = ctype;
182
183 // type of the LocalView of the patch geometry
184 using GeometryLocalView =
185 typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
186 ctype>::template GeometryLocalView<codim, ParameterSpace>;
187
189 TrimmedLocalGeometryImpl() = default;
190 explicit TrimmedLocalGeometryImpl(const GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>& patchGeometry)
191 : patchGeometry{patchGeometry} {}
192
193 bool operator==(const TrimmedLocalGeometryImpl& b) const {
194 return Dune::FloatCmp::eq(b.patchGeometry.corner(0) == this->patchGeometry.corner(0)) and
195 Dune::FloatCmp::eq(b.patchGeometry.corner(1) == this->patchGeometry.corner(1));
196 };
197
200 [[nodiscard]] GeometryType type() const {
201 return GeometryTypes::cube(mydimension);
202 }
203
204 // return whether we have an affine mapping (true for straight lines)
205 [[nodiscard]] bool affine() const {
206 // handy if the trims are straight lines for other algorithms
207 if constexpr (codim == 0)
208 return true;
209 else
210 return patchGeometry.degree()[0] == 1;
211 }
212
213 // return the number of corners of this element. Corners are numbered 0...n-1
214 [[nodiscard]] int corners() const {
215 return patchGeometry.corners();
216 }
217
218 GlobalCoordinate center() const {
219 return patchGeometry.center();
220 }
221
222 // access to coordinates of corners. Index is the number of the corner
223 GlobalCoordinate corner(int i) const {
224 return patchGeometry.corner(i);
225 }
226
229 GlobalCoordinate global(const LocalCoordinate& local) const {
230 return patchGeometry.global(local);
231 }
232
235 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
236 return patchGeometry.jacobianTransposed(local);
237 }
238
241 LocalCoordinate local(const GlobalCoordinate& global) const {
242 return patchGeometry.local(global);
243 }
244
245 // Returns true if the point is in the current element
246 // TODO
247 bool checkInside(const LocalCoordinate& local) const {
248 return true;
249 }
250
251 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
252 return patchGeometry.volume();
253 }
254
255 // The Jacobian matrix of the mapping from the reference element to this element
256 // TODO not yet implemented
257 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
258 return patchGeometry.jacobianInverseTransposed(local);
259 }
260
261private:
262 GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype> patchGeometry;
263};
