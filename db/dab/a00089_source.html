23template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
24class TrimmedLocalGeometryImpl<2, coorddim, GridImp, localGeometryTag>
25{
26public:
27 using ctype = typename GridImp::ctype;
28
29 static constexpr int mydimension = 2;
30 using Trimmer = typename GridImp::Trimmer;
31
32 static constexpr int coorddimension = coorddim;
33 static constexpr int codim = coorddimension - mydimension;
34 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
35 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
36
37 using HostGeometry = typename Trimmer::TrimmerTraits::template Codim<0>::HostParameterSpaceGridEntity::Geometry;
38 using TrimData = typename Trimmer::ElementTrimData;
39
40 using LocalCoordinate = FieldVector<ctype, mydimension>;
41 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
42 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
43 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
44 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
45 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
46 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
47 using Volume = ctype;
48
49 // type of the LocalView of the patch geometry
50 using GeometryLocalView = typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
51 ctype>::template GeometryLocalView<codim, Trimmer>;
52
54 TrimmedLocalGeometryImpl() = default;
55 explicit TrimmedLocalGeometryImpl(const HostGeometry& hostGeometry, const TrimData& trimData)
56 : hostGeometry_{hostGeometry},
57 trimData_{trimData} {}
58 bool operator==(const TrimmedLocalGeometryImpl& b) const {
59 return b.hostGeometry_ == this->hostGeometry_;
60 }
61
64 [[nodiscard]] GeometryType type() const {
65 return GeometryTypes::none(mydimension);
66 }
67
68 // return whether we have an affine mapping (true for straight lines)
69 [[nodiscard]] bool affine() const {
70 return true;
71 }
72
73 // return the number of corners of this element. Corners are numbered 0...n-1
74 [[nodiscard]] int corners() const {
75 return trimData_.size(2);
76 }
77
78 GlobalCoordinate center() const {
79 return hostGeometry_.center();
80 }
81
82 // access to coordinates of corners. Index is the number of the corner
83 GlobalCoordinate corner(int i) const {
84 auto vData = trimData_.vertex(i);
85 if (vData.isHost) {
86 return hostGeometry_.corner(Transformations::mapToDune(2, vData.idx));
87 }
88
89 return vData.geometry.value();
90 }
91
94 GlobalCoordinate global(const LocalCoordinate& local) const {
95 return hostGeometry_.global(local);
96 }
97
100 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
101 return hostGeometry_.jacobianTransposed(local);
102 }
103
106 LocalCoordinate local(const GlobalCoordinate& global) const {
107 return hostGeometry_.local(global);
108 }
109
110 // Returns true if the point is in the current element
111 // @todo
112 bool checkInside(const LocalCoordinate& local) const {
113 return true;
114 }
115
116 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
117 return hostGeometry_.volume();
118 }
119
120 // The Jacobian matrix of the mapping from the reference element to this element
121 // @todo not yet implemented
122 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
123 DUNE_THROW(Dune::NotImplemented, "jacobianInverseTransposed() not yet implemented");
124 return JacobianInverseTransposed{};
125 }
126
127private:
128 HostGeometry hostGeometry_;
129 TrimData trimData_;
130};

132template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
133class TrimmedLocalGeometryImpl<1, coorddim, GridImp, localGeometryTag>
134{
135public:

161 : patchGeometry{patchGeometry} {}
162
163 bool operator==(const TrimmedLocalGeometryImpl& b) const {
164 return Dune::FloatCmp::eq(b.patchGeometry.corner(0) == this->patchGeometry.corner(0)) and
165 Dune::FloatCmp::eq(b.patchGeometry.corner(1) == this->patchGeometry.corner(1));
166 };

171 return GeometryTypes::none(mydimension);
172 }
173
174 // return whether we have an affine mapping (true for straight lines)
175 [[nodiscard]] bool affine() const {
176 // handy if the trims are straight lines for other algorithms
177 if constexpr (codim == 0)
178 return true;
179 else
180 return patchGeometry.degree()[0] == 1;
181 }
182
183 // return the number of corners of this element. Corners are numbered 0...n-1
184 [[nodiscard]] int corners() const {
185 return patchGeometry.corners();
186 }
187
188 GlobalCoordinate center() const {
189 return patchGeometry.center();
190 }
191
192 // access to coordinates of corners. Index is the number of the corner
193 GlobalCoordinate corner(int i) const {
194 return patchGeometry.corner(i);
195 }
196
199 GlobalCoordinate global(const LocalCoordinate& local) const {
200 return patchGeometry.global(local);
201 }
202
205 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
206 return patchGeometry.jacobianTransposed(local);
207 }
208
211 LocalCoordinate local(const GlobalCoordinate& global) const {
212 return patchGeometry.local(global);
213 }
214
215 // Returns true if the point is in the current element
216 // @todo
217 bool checkInside(const LocalCoordinate& local) const {
218 return true;
219 }
220

222 return patchGeometry.volume();

224

228 DUNE_THROW(Dune::NotImplemented, "jacobianInverseTransposed() not yet implemented");
229 return JacobianInverseTransposed{};
230 }
231
232private:
233 GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype> patchGeometry;
