126 /* this is for testing purposes only */
127 void drawResult(const std::string& filename, bool inParameterSpace, bool newFig = true) {
128 if (flag_ == ElementTrimFlag::empty)
129 return;
130 auto eleGeometry = hostEntity_.geometry();
131 auto lowerLeftCorner = inParameterSpace ? eleGeometry.corner(0) : Dune::FieldVector<double, 2>({0, 0});
132 auto upperRightCorner = eleGeometry.corner(3);
133 Dune::DiagonalMatrix<double, 2> scaling({inParameterSpace ? 1.0 / (upperRightCorner[0] - lowerLeftCorner[0]) : 1,
134 inParameterSpace ? 1.0 / (upperRightCorner[1] - lowerLeftCorner[1]) : 1});
135
136 assert(vertices_.size() == edges_.size());
137
138 if (newFig)
139 auto figure = matplot::figure(true);
140 matplot::hold("on");
141
142 matplot::text(eleGeometry.center()[0], eleGeometry.center()[1], std::to_string(edges_.size()))
143 ->font("Arial")
144 .font_size(18);
145
146 constexpr std::array idxLookUp = {0, 1, 3, 2};
147
148 const double scale = eleGeometry.volume() / 5;
149
150 auto plotEllipse = [&](const Vertex& v) {
151 const auto w = 0.025 * scale;
152 const auto c = matplot::ellipse(v[0] - (w / 2), v[1] - (w / 2), w, w);
153 c->color("blue");
154 };
155
156 auto plotLine = [&](std::vector<Vertex>& vs, bool thin = false) {
157 std::vector<double> x;
158 std::ranges::transform(vs, std::back_inserter(x), [](auto& v) { return v[0]; });
159
160 std::vector<double> y;
161 std::ranges::transform(vs, std::back_inserter(y), [](auto& v) { return v[1]; });
162 if (thin)
163 matplot::plot(x, y)->line_width(0.5).color("grey");
164 else
165 matplot::plot(x, y)->line_width(2).color("red");
166
167 // Annotate
168 if (thin)
169 return;
170 matplot::text(x.front(), y.front(), "S")->font("Arial").color("blue").font_size(18);
171 };
172
173 for (auto c : edgeLookUp) {
174 Dune::FieldVector<double, 2> firstVertexPos = eleGeometry.corner(c[0]) - lowerLeftCorner;
175 Dune::FieldVector<double, 2> secondVertexPos = eleGeometry.corner(c[1]) - lowerLeftCorner;
176 scaling.mv(firstVertexPos, firstVertexPos);
177 scaling.mv(secondVertexPos, secondVertexPos);
178 std::vector<Vertex> vs{firstVertexPos, secondVertexPos};
179 plotLine(vs, true);
180 }
181
182 for (auto& v : vertices_) {
183 if (v.geometry.has_value()) {
184 Dune::FieldVector<double, 2> pos = (v.geometry.value() - lowerLeftCorner);
185 scaling.mv(pos, pos);
186 plotEllipse(pos);
187 } else {
188 Dune::FieldVector<double, 2> pos = eleGeometry.corner(idxLookUp[v.idx]) - lowerLeftCorner;
189 scaling.mv(pos, pos);
190 plotEllipse(pos);
191 }
192 }
193
194 for (auto& curve : edges_) {
195 if (curve.geometry.has_value()) {
196 std::vector<Vertex> vs;
197 for (double u : Utilities::linspace(curve.geometry.value().domain().front(), 40)) {
198 Dune::FieldVector<double, 2> pos = (curve.geometry.value().global(u) - lowerLeftCorner);
199 scaling.mv(pos, pos);
200 vs.push_back(pos);
201 }
202 plotLine(vs);
203 } else {
204 auto cornerIdx = edgeLookUp[curve.idx];
205 Dune::FieldVector<double, 2> firstVertexPos = (eleGeometry.corner(cornerIdx.front()) - lowerLeftCorner);
206 Dune::FieldVector<double, 2> secondVertexPos = (eleGeometry.corner(cornerIdx.back()) - lowerLeftCorner);
207 scaling.mv(firstVertexPos, firstVertexPos);
208 scaling.mv(secondVertexPos, secondVertexPos);
209 std::vector<Vertex> vs{firstVertexPos, secondVertexPos};
210 plotLine(vs);
211 }
212 }
213
214 matplot::axis(matplot::equal);
215 static int elleC = 0;
216 if (newFig)
217 matplot::save(filename + std::to_string(elleC++) + (inParameterSpace ? "inParameterSpace" : ""), "gif");
218 }

274 // We are nudging the vertex slightly inward so that findSpan is guaranteed to succeed
275 auto nudgedVertex(const Vertex& v, int edgeIdx) {
276 if (edgeIdx == -1)
277 return v;

292private:
293 auto sampleCurves() {
294 auto eleGeo = hostEntity_.geometry();
295 auto lowerLeftCorner = eleGeo.corner(0);
296 auto upperRightCorner = eleGeo.corner(3);
297 Dune::DiagonalMatrix<double, 2> scaling(
298 {1.0 / (upperRightCorner[0] - lowerLeftCorner[0]), 1.0 / (upperRightCorner[1] - lowerLeftCorner[1])});
299
300 for (auto& edge : edges_) {
301 if (edge.geometry.has_value()) {
302 for (const auto v : Utilities::linspace(edge.geometry->domain()[0], 100)) {
303 auto fV = edge.geometry->global({v}) - lowerLeftCorner;
304 scaling.mv(fV, fV);
305 path_.emplace_back(fV[0], fV[1]);
306 }
307 } else {
308 auto cornerIdx = edgeLookUp[edge.idx];
309
310 auto firstVertex = eleGeo.corner(cornerIdx.front()) - lowerLeftCorner;
311 auto secondVertex = eleGeo.corner(cornerIdx.back()) - lowerLeftCorner;
312 scaling.mv(firstVertex, firstVertex);
313 scaling.mv(secondVertex, secondVertex);
314
315 path_.emplace_back(firstVertex[0], firstVertex[1]);
316 path_.emplace_back(secondVertex[0], secondVertex[1]);
317 }
318 }
319 }

327 int newEdgeCounter_ = 4;
