34 static auto createIntegrationRule(const PatchElement& element, int quadratureOrder,
35 const Parameters& parameters = Parameters{},
36 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) {
37 if (not element.impl().getLocalEntity().isTrimmed()) {
38 return QuadratureRules<double, dim>::rule(element.type(), quadratureOrder, qt);
39 }
40
41 auto [elements, vertices, indices] = createSimplicies(element, parameters);
42 return makeQuadratureRule(elements, quadratureOrder, qt);
43 }

45 static auto createSimplicies(const PatchElement& element, const Parameters& parameters = Parameters{})
46 -> std::tuple<std::vector<Element>, std::vector<Point>, std::vector<Index>> {
47 std::vector<Point> vertices{};
48 std::vector<Element> elements{};
49
50 auto& trimData = element.impl().getLocalEntity().trimData();
51 auto hostEntity = element.impl().getLocalEntity().getHostEntity();
52
53 for (const auto& edgeInfo : trimData.edges()) {
54 if (edgeInfo.isTrimmed) {
55 auto localGeometry = GeometryKernel::transformToSpan(edgeInfo.geometry.value(), hostEntity.geometry());
56 splitBoundary(localGeometry, parameters, vertices);
57 } else {
58 auto refElement = referenceElement(hostEntity);
59 switch (edgeInfo.idx) {
60 case 0:
61 vertices.push_back(refElement.template geometry<2>(0).center());
62 break;
63 case 1:
64 vertices.push_back(refElement.template geometry<2>(1).center());
65 break;
66 case 2:
67 vertices.push_back(refElement.template geometry<2>(3).center());
68 break;
69 case 3:
70 vertices.push_back(refElement.template geometry<2>(2).center());
71 break;
72 default:
73 assert(edgeInfo.idx < 4);
74 __builtin_unreachable();
75 }
76 }
77 }
78
79 auto indices = triangulate(vertices);
80 for (auto it = indices.begin(); it < indices.end(); it += 3)
81 elements.emplace_back(GeometryTypes::triangle, std::vector<FieldVector<double, dim>>{
82 vertices[*it], vertices[*(it + 1)], vertices[*(it + 2)]});
83
84 return std::make_tuple(elements, vertices, indices);
85 }

87private:
88 static void splitBoundary(const auto& localGeometry, const Parameters& parameters, std::vector<Point>& points) {
89
90 if (localGeometry.affine()) {
91 points.push_back(localGeometry.global({0.0}));
92 return;
93 }
94
95 if (not parameters.useAdaptiveDivisions()) {
96 points.reserve(parameters.boundaryDivisions);
97 for (auto local : Utilities::linspace(0.0, 1.0, parameters.boundaryDivisions))
98 points.push_back(localGeometry.global({local}));
99 } else {
100 // Guess initial amount of segments (a quadratic segment of lenght 1 should be devided by 5 divions)
101 double curveLength = localGeometry.curveLength();
102 auto initialSegments = static_cast<unsigned int>(std::ceil(5 * curveLength * localGeometry.degree()[0]));
103
104 std::vector<Point> initialPoints;
105 for (const auto i : Dune::range(initialSegments)) {
106 double t = static_cast<double>(i) / initialSegments;
107 initialPoints.push_back(localGeometry.global({t}));
108 }
109
110 points.push_back(initialPoints.front());
111
112 for (size_t i_ = 0; i_ < initialPoints.size() - 1; ++i_) {
113 const auto i = static_cast<double>(i_);
114 refineSegments(points, localGeometry, i * (1.0 / initialSegments), (i + 1) * (1.0 / initialSegments),
115 parameters.targetAccuracy);
116 }
117 }
118 points.push_back(localGeometry.global({1.0}));
119
120 // Remove last element to avoid duplication
121 points.pop_back();
122 }

124 static auto triangulate(const std::vector<Point>& points) -> std::vector<Index> {
125 std::vector<std::vector<Point>> polygonInput;
126 polygonInput.push_back(points);
127
128 return mapbox::earcut<Index>(polygonInput);
129 }
130 static auto makeQuadratureRule(const std::vector<Element>& elements, int quadratureOrder,
131 const QuadratureType::Enum qt) {
132 assert(qt == QuadratureType::GaussLegendre);
133
134 Dune::QuadratureRule<double, dim> vector{};
135
136 for (auto& subElement : elements) {
137 const auto& rule = Dune::QuadratureRules<double, dim>::rule(subElement.type(), quadratureOrder, qt);
138 for (auto ip : rule) {
139 auto globalInSpan = subElement.global(ip.position());
140 vector.emplace_back(globalInSpan, ip.weight() * subElement.integrationElement(ip.position()));
141 }
142 }
143 return vector;
144 }

146 // Function to calculate the distance between two points
147 static double distance(const Point& p1, const Point& p2) {
148 return std::sqrt(Dune::power(p2[0] - p1[0], 2) + Dune::power(p2[1] - p1[1], 2));
149 }
150
151 static void refineSegments(std::vector<Point>& points, const auto& localGeometry, double t_start, double t_end,
152 double tolerance) {
153 Point p1 = localGeometry.global({t_start});
154 Point p2 = localGeometry.global({t_end});
155
156 // Calculate the midpoint of the parameter
157 double t_mid = (t_start + t_end) / 2;
158 Point p_mid = localGeometry.global({t_mid});
159
160 // Calculate the perpendicular distance from the midpoint to the line segment
161 double numerator =
162 std::abs((p2[1] - p1[1]) * p_mid[0] - (p2[0] - p1[0]) * p_mid[1] + p2[0] * p1[1] - p2[1] * p1[0]);
163 double denominator = distance(p1, p2);
164 double dist_to_segment = numerator / denominator;
165
166 if (dist_to_segment > tolerance) {
167 // If the distance is greater than the tolerance, subdivide recursively
168 refineSegments(points, localGeometry, t_start, t_mid, tolerance);
169 refineSegments(points, localGeometry, t_mid, t_end, tolerance);
170 } else {
171 points.push_back(p2); // Include the endpoint of the segment
172 }
173 }
174};
