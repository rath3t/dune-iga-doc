376 template <typename IntegrationRuleGenerator = DefaultTrim::SimplexIntegrationRuleGenerator<const GridImp>>
377 Dune::QuadratureRule<double, dim> getQuadratureRule(
378 const std::optional<int>& p_order = std::nullopt,
379 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) const
380 requires(not Trimmer::isAlwaysTrivial and dimension == 2)
381 {
382 auto degree = patchGrid_->patchGeometry(this->level()).degree();
383 int order = p_order.value_or(dimension * *std::ranges::max_element(degree));
384 if (not isTrimmed())
385 return Dune::QuadratureRules<double, dimension>::rule(this->type(), order, qt);
386
387 const auto parameters = typename IntegrationRuleGenerator::Parameters{
388 .boundaryDivisions = Preferences::getInstance().boundaryDivisions(),
389 .targetAccuracy = Preferences::getInstance().targetAccuracy()};
390
391 return IntegrationRuleGenerator::createIntegrationRule(*this, order, parameters);
392 }
