22template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
23class TrimmedLocalGeometryImpl<2, coorddim, GridImp, localGeometryTag>
24{
25public:
26 using ctype = typename GridImp::ctype;
27
28 static constexpr int mydimension = 2;
29 using ParameterSpace = typename GridImp::ParameterSpace;
30
31 static constexpr int coorddimension = coorddim;
32 static constexpr int codim = coorddimension - mydimension;
33 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
34 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
35
36 using HostGeometry =
37 typename ParameterSpace::ParameterSpaceTraits::template Codim<0>::HostParameterSpaceGridEntity::Geometry;
38 using TrimData = typename ParameterSpace::ElementTrimData;
39
40 using ParameterSpaceElement = typename GridImp::ParameterSpace::template Codim<0>::ParameterSpaceGridEntity;
41
42 using LocalCoordinate = FieldVector<ctype, mydimension>;
43 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
44 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
45 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
46 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
47 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
48 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
49 using Volume = ctype;
50
51 // type of the LocalView of the patch geometry
52 using GeometryLocalView =
53 typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
54 ctype>::template GeometryLocalView<codim, ParameterSpace>;
55
57 TrimmedLocalGeometryImpl() = default;
58 explicit TrimmedLocalGeometryImpl(const HostGeometry& hostGeometry, const ParameterSpaceElement* element)
59 : hostGeometry_{hostGeometry},
60 trimData_{element->trimData()},
61 element_(element) {}
62
63 bool operator==(const TrimmedLocalGeometryImpl& b) const {
64 return b.hostGeometry_ == this->hostGeometry_;
65 }
66
69 [[nodiscard]] GeometryType type() const {
70 if (Preferences::getInstance().reportTrimmedElementGeometryTypeAsNone())
71 return GeometryTypes::none(mydimension);
72 return GeometryTypes::cube(mydimension);
73 }
74
75 // return whether we have an affine mapping (true for straight lines)
76 [[nodiscard]] bool affine() const {
77 return true;
78 }
79
80 // return the number of corners of this element. Corners are numbered 0...n-1
81 [[nodiscard]] int corners() const {
82 return trimData_.size(2);
83 }
84
85 // @todo write a test for that
86 GlobalCoordinate center() const {
87 // average of corners
88 GlobalCoordinate c{0, 0};
89 for (auto i : Dune::range(corners()))
90 c += corner(i);
91
92 return c / corners();
93 }
94
95 // access to coordinates of corners. Index is the number of the corner
96 GlobalCoordinate corner(int i) const {
97 auto vData = trimData_.vertex(i);
98 if (vData.isHost)
99 return hostGeometry_.corner(Transformations::mapToDune(2, vData.idx));
100
101 return vData.geometry.value();
102 }
103
106 GlobalCoordinate global(const LocalCoordinate& local) const {
107 return hostGeometry_.global(local);
108 }
109
112 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
113 return hostGeometry_.jacobianTransposed(local);
114 }
115
118 LocalCoordinate local(const GlobalCoordinate& global) const {
119 return hostGeometry_.local(global);
120 }
121
122 // Returns true if the point is in the current element
123 // @todo
124 bool checkInside(const LocalCoordinate& local) const {
125 return true;
126 }
127
128 // @todo // Use integration rule and area integration
129 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
130 return hostGeometry_.volume();
131 }
132
133 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
134 return hostGeometry_.jacobianInverseTransposed(local);
135 }
136
137 auto getQuadratureRule(const std::optional<int>& p_order = std::nullopt,
138 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) const {
139 return element_->getQuadratureRule(p_order, qt);
140 }
141
142 bool isTrimmed() const {
143 return true;
144 }
145
146private:
147 HostGeometry hostGeometry_;
148 TrimData trimData_;
149 const ParameterSpaceElement* element_;
150};

152template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
153class TrimmedLocalGeometryImpl<1, coorddim, GridImp, localGeometryTag>
154{
155public:
156 using ctype = typename GridImp::ctype;
157
158 static constexpr int mydimension = 1;
159 using ParameterSpace = typename GridImp::ParameterSpace;
160
161 static constexpr int coorddimension = coorddim;
162 static constexpr int codim = coorddimension - mydimension;
163 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
164 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
165 using LocalCoordinate = FieldVector<ctype, mydimension>;
166 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
167 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
168 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
169 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
170 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
171 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
172 using Volume = ctype;
173
174 // type of the LocalView of the patch geometry
175 using GeometryLocalView =
176 typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
177 ctype>::template GeometryLocalView<codim, ParameterSpace>;
178
180 TrimmedLocalGeometryImpl() = default;
181 explicit TrimmedLocalGeometryImpl(const GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>& patchGeometry)
182 : patchGeometry{patchGeometry} {}
183
184 bool operator==(const TrimmedLocalGeometryImpl& b) const {
185 return Dune::FloatCmp::eq(b.patchGeometry.corner(0) == this->patchGeometry.corner(0)) and
186 Dune::FloatCmp::eq(b.patchGeometry.corner(1) == this->patchGeometry.corner(1));
187 };
188
191 [[nodiscard]] GeometryType type() const {
192 return GeometryTypes::cube(mydimension);
193 }
194
195 // return whether we have an affine mapping (true for straight lines)
196 [[nodiscard]] bool affine() const {
197 // handy if the trims are straight lines for other algorithms
198 if constexpr (codim == 0)
199 return true;
200 else
201 return patchGeometry.degree()[0] == 1;
202 }
203
204 // return the number of corners of this element. Corners are numbered 0...n-1
205 [[nodiscard]] int corners() const {
206 return patchGeometry.corners();
207 }
208
209 GlobalCoordinate center() const {
210 return patchGeometry.center();
211 }
212
213 // access to coordinates of corners. Index is the number of the corner
214 GlobalCoordinate corner(int i) const {
215 return patchGeometry.corner(i);
216 }
217
220 GlobalCoordinate global(const LocalCoordinate& local) const {
221 return patchGeometry.global(local);
222 }
223
226 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
227 return patchGeometry.jacobianTransposed(local);
228 }
229
232 LocalCoordinate local(const GlobalCoordinate& global) const {
233 return patchGeometry.local(global);
234 }
235
236 // Returns true if the point is in the current element
237 // @todo
238 bool checkInside(const LocalCoordinate& local) const {
239 return true;
240 }
241
242 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
243 return patchGeometry.volume();
244 }
245
246 // The Jacobian matrix of the mapping from the reference element to this element
247 // @todo not yet implemented
248 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
249 return patchGeometry.jacobianInverseTransposed(local);
250 }
251
252private:
253 GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype> patchGeometry;
254};
