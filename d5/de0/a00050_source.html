14 if (p1[0] < p2[0])

23template <typename UnstructuredGrid, typename PatchGrid>
24std::unique_ptr<UnstructuredGrid> createUnstructuredGridImpl(PatchGrid* patchGrid) {
25 auto gridView = patchGrid->leafGridView();
26 auto& idSet = patchGrid->globalIdSet();
27
28 IGARefinedGeometries geometries(gridView, 0, 0);
29
30 std::set<FieldVector<double, 2>, Impl::Comparator> vertices;
31
32 for (const auto& element : elements(gridView)) {
33 auto id = idSet.id(element);
34 auto geometry = element.geometry();
35
36 std::ranges::transform(geometries.getVertices(id), std::inserter(vertices, vertices.begin()),
37 [&](const auto& v) { return geometry.global(v); });
38 }
39
40 auto gridFactory = Dune::GridFactory<UnstructuredGrid>();
41
42 for (const auto& v : vertices) {
43 gridFactory.insertVertex(v);
44 }
45
46 // Reconstruct grid
47 for (const auto& element : elements(gridView)) {
48 auto geometry = element.geometry();
49 auto id = idSet.id(element);
50
51 auto gt = geometries.geometryType(id);
52 unsigned int nSubI = gt == GeometryTypes::simplex(2) ? 3 : 4;
53
54 auto& eleVertices = geometries.getVertices(id);
55
56 for (auto subEleIdx : std::views::iota(0u, geometries.nElements(id))) {
57 std::vector<unsigned int> elementVertices;
58
59 for (auto subEntityIndex : std::views::iota(0u, nSubI)) {
60 auto localVertexIdx = geometries.vertexSubIndex(id, subEleIdx, subEntityIndex);
61 Dune::FieldVector<double, 2> vertex = geometry.global(eleVertices[localVertexIdx]);
62
63 // Find Idx
64 auto it = vertices.find(vertex);
65 assert(it != vertices.end());
66
67 elementVertices.push_back(std::distance(vertices.begin(), it));
68 }
69 gridFactory.insertElement(gt, elementVertices);
70 }
71 }
72
73 return gridFactory.createGrid();
74}
