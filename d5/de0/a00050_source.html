15template <int dim, int dimworld, typename ScalarType>
16auto TrimmerImpl<dim, dimworld, ScalarType>::trimElement(const YASPEntity<0>& element,
17 const PatchTrimData& patchTrimData) -> ElementTrimData {
18 auto geo = element.geometry();
19
20 static constexpr int numberOfCorners = 4;
21 std::array<FieldVector<double, 2>, numberOfCorners> corners;
22 for (const auto i : Dune::range(numberOfCorners))
23 corners[i] = geo.corner(Util::vertexIndexMapping[i]);
24
25 auto [flag, result] = Util::clipElementRectangle(element, patchTrimData);
26
27 ElementTrimData elementTrimData(flag, element);
28 if (flag != ElementTrimFlag::trimmed)
29 return elementTrimData;
30
31 auto nextEntity = [&](const int i) { return (i + 1) % result.vertices_.size(); };
32 auto getTrimmingCurveIdx = [&](auto& vV) -> auto { return patchTrimData.getIndices(vV.zValue()); };
33
34 auto throwGridError = [] {
35 DUNE_THROW(Dune::GridError, "TrimElement wasn't successfull. Could not connect element trim curves");
36 };
37
38 auto assertPoint = [&](const Clipper2Lib::PointD& ip, const FieldVector<ScalarType, dim>& curvePt,
39 double prec = 0.1) {
40 if (not(FloatCmp::eq(ip.x, curvePt[0], prec) and FloatCmp::eq(ip.y, curvePt[1], prec))) {
41 std::cout << "Found ClipperPoint " << ip << " does not coincide with determined point on Curve " << curvePt
42 << std::endl;
43 throwGridError();
44 }
45 };
46
47 using namespace Util;
48
49 // Major todo: Create fallback to straight line if algorithm is not able to find correct Trimming CurveIdx
50
51 // State
52 std::vector<FieldVector<ScalarType, dim>> foundVertices;
53 Impl::CurveLoopIndexEncoder::IndexResult currentCurveIdx = {std::numeric_limits<size_t>::infinity(),
54 std::numeric_limits<size_t>::infinity(),
55 std::numeric_limits<size_t>::infinity()};
56 double currentT = std::numeric_limits<double>::infinity();
57
58 for (const auto i : std::views::iota(0u, result.vertices_.size())) {
59 auto vV1 = result.vertices_[i];
60 auto vV2 = result.vertices_[nextEntity(i)];
61
62 auto pt1 = vV1.pt;
63 auto pt2 = vV2.pt;
64
65 // First case edge is completly untrimmed
66 if (vV1.isHost() and vV2.isHost()) {
67 elementTrimData.addEdge(Util::giveEdgeIdx(vV1.hostId(), vV2.hostId()));
68 foundVertices.push_back({pt2.x, pt2.y});
69 }
70
71 // Second case edge begins on a hostVertex and ends on a newVertex
72 else if (vV1.isHost() and vV2.isNew()) {
73 currentCurveIdx = getTrimmingCurveIdx(vV2);
74 auto [tParam, curvePoint] =
75 Util::callFindIntersection(patchTrimData.getCurve(currentCurveIdx), vV2.edgeId(), pt2, corners);
76 assertPoint(pt2, curvePoint);
77
78 FieldVector<ScalarType, dim> p =
79 foundVertices.empty() ? FieldVector<ScalarType, dim>{pt1.x, pt1.y} : foundVertices.back();
80 auto trimmedEdge = Util::createHostGeometry<TrimmingCurve>(p, curvePoint);
81 elementTrimData.addEdgeHostNew(vV2.edgeId(), trimmedEdge, curvePoint);
82 foundVertices.push_back(curvePoint);
83
84 currentT = tParam;
85 }
86 // Third case newVertex - newVertex
87 else if (vV1.isNew() && vV2.isNew()) {
88 // If there is no Vertex yet found add search for vV1 and add it to the foundVertices, also set currentCurveIdx
89 if (foundVertices.empty()) {
90 currentCurveIdx = getTrimmingCurveIdx(vV1);
91 auto [currentT, curvePoint] =
92 Util::callFindIntersection(patchTrimData.getCurve(currentCurveIdx), vV1.edgeId(), pt1, corners);
93 assertPoint(pt1, curvePoint);
94
95 foundVertices.push_back(curvePoint);
96 }
97 if (getTrimmingCurveIdx(vV2).curve != currentCurveIdx.curve and
98 getTrimmingCurveIdx(vV2).loop != currentCurveIdx.loop)
99 throwGridError();
100
101 auto [tParam, curvePoint] =
102 Util::callFindIntersection(patchTrimData.getCurve(currentCurveIdx), vV2.edgeId(), pt2, corners);
103 assertPoint(pt2, curvePoint);
104
105 // If currentT > tParam it can have 2 reasons. 1) the trim id only on one side of the element, and it has to
106 // connect back or 2) if the trimming curve consist only of one curve where it has the same front and back
107 // Controlpoint, so sometimes the wrong tParam (front or back) gets determined
108 if (currentT > tParam) {
109 bool success = false;
110 if (currentCurveIdx.loop > 0 && patchTrimData.loops()[currentCurveIdx.loop].size() == 1) {
111 auto curve = patchTrimData.getCurve(currentCurveIdx);
112 if (curve.isConnectedAtBoundary(0)) {
113 auto elementTrimmingCurve = Util::createTrimmingCurveSlice(curve, currentT, curve.domain()[0].back());
114 elementTrimData.addEdgeNewNew(elementTrimmingCurve, curvePoint);
115 currentT = std::numeric_limits<double>::infinity();
116 success = true;
117 }
118 } else if (vV1.edgeId() == vV2.edgeId()) {
119 auto trimmedEdge = Util::createHostGeometry<TrimmingCurve>(foundVertices.back(), curvePoint);
120 elementTrimData.addEdgeNewNewOnHost(vV2.edgeId(), trimmedEdge, curvePoint);
121 currentT = tParam;
122 success = true;
123 }
124 if (not success)
125 throwGridError();
126 } else {
127 auto elementTrimmingCurve =
128 Util::createTrimmingCurveSlice(patchTrimData.getCurve(currentCurveIdx), currentT, tParam);
129 elementTrimData.addEdgeNewNew(elementTrimmingCurve, curvePoint);
130 currentT = std::numeric_limits<double>::infinity();
131 }
132
133 foundVertices.push_back(curvePoint);
134 }
135
136 // Fourth case edge begins on a newVertex and ends in a HostVertex
137 else if (vV1.isNew() and vV2.isHost()) {
138 auto v2 = Dune::FieldVector<ScalarType, dim>{pt2.x, pt2.y};
139
140 FieldVector<ScalarType, dim> p;
141 if (foundVertices.empty())
142 std::tie(std::ignore, p) =
143 Util::callFindIntersection(patchTrimData.getCurve(getTrimmingCurveIdx(vV1)), vV1.edgeId(), pt1, corners);
144 else
145 p = foundVertices.back();
146
147 auto trimmedEdge = Util::createHostGeometry<TrimmingCurve>(p, v2);
148 elementTrimData.addEdgeNewHost(vV1.edgeId(), trimmedEdge, vV2.hostId());
149
150 foundVertices.push_back(v2);
151 }
152 // Additional cases to cover inside vertices
153 else if (vV1.isNew() and vV2.isInside()) {
154 if (foundVertices.empty()) {
155 currentCurveIdx = getTrimmingCurveIdx(vV1);
156 FieldVector<ScalarType, dim> curvePoint;
157 std::tie(currentT, curvePoint) =
158 Util::callFindIntersection(patchTrimData.getCurve(currentCurveIdx), vV1.edgeId(), pt1, corners);
159 assertPoint(pt1, curvePoint);
160
161 foundVertices.push_back(curvePoint);
162 }
163 const auto& curve = patchTrimData.loops()[vV2.loop()].curves()[vV2.formerCurve()];
164 double tParam = curve.domain().front().back();
165 auto elementTrimmingCurve = Util::createTrimmingCurveSlice(curve, currentT, tParam);
166 auto curvePoint = curve.corner(1);
167 elementTrimData.addEdgeNewNew(elementTrimmingCurve, curvePoint);
168 foundVertices.push_back(curvePoint);
169 } else if (vV1.isInside() and vV2.isNew()) {
170 const auto& curve = patchTrimData.loops()[vV1.loop()].curves()[vV1.subsequentCurve()];
171 currentT = curve.domain().front().front();
172 auto [tParam, curvePoint] = Util::callFindIntersection(curve, vV2.edgeId(), pt2, corners);
173 assertPoint(pt2, curvePoint);
174
175 auto elementTrimmingCurve = Util::createTrimmingCurveSlice(curve, currentT, tParam);
176 elementTrimData.addEdgeNewNew(elementTrimmingCurve, curvePoint);
177 foundVertices.push_back(curvePoint);
178 } else {
179 throwGridError();
180 }
181 }
182 elementTrimData.finalize();
183 return std::move(elementTrimData);
184}
