135 [[nodiscard]] double volume() const {
136 auto dom = domain();
137 double volParamterSpace = 1.0;
138 for (int i = 0; i < mydimension; ++i)
139 volParamterSpace *= dom[i].volume();
140 const auto rule = Dune::QuadratureRules<ctype, mydimension>::rule(
141 this->type(), mydimension * (*std::ranges::max_element(patchData_.degree)));
142 ctype vol = 0.0;
143 const std::array<Utilities::Domain<double>, mydimension> input{};
144 for (auto& gp : rule) {
145 vol += integrationElement(Utilities::mapToRange(gp.position(), input, domain())) * gp.weight() * volParamterSpace;
146 }
147 return vol;
148 }
152 [[nodiscard]] double curveLength() const
153 requires(mydimension == 1)
154 {
155 if (affine())
156 return (corner(1) - corner(0)).two_norm();
157
158 auto dom = domain();
159 double volParamterSpace = dom[0].volume();
160
161 const auto rule =
162 Dune::QuadratureRules<ctype, mydimension>::rule(this->type(), 4 * *std::ranges::max_element(patchData_.degree));
163 ctype vol = 0.0;
164 const std::array<Utilities::Domain<double>, mydimension> input{};
165 for (auto& gp : rule) {
166 vol += integrationElement(Utilities::mapToRange(gp.position(), input, domain())) * gp.weight() * volParamterSpace;
167 }
168 return vol;
169 }

388 ConnectionResult isConnectedAtBoundary(int boundary, double tol = 1e-8) const {
389 const auto& cps = patchData_.controlPoints;
390 if constexpr (mydimension == 1) {
391 if (auto cp = cps.directGetAll(); FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(
392 cp.front().p, cp.back().p, tol))
393 return ConnectionResult(true, !boundary);
394 else
395 return ConnectionResult(false, -1);
396 } else if constexpr (mydimension == 2) {
397 auto edgeRange = [&cps](int b) {
398 switch (b) {
399 case 0:
400 return cps.hyperSurfFront({1});
401 case 1:
402 return cps.hyperSurfBack({1});
403 case 2:
404 return cps.hyperSurfFront({0});
405 case 3:
406 return cps.hyperSurfBack({0});
407 default:
408 assert("No valid boundary index");
409 }
410 __builtin_unreachable();
411 };
412
413 auto theEdgeToCheck = edgeRange(boundary);
414 for (auto i : Dune::range(4)) {
415 if (boundary == i)
416 continue;
417 if (std::ranges::equal(theEdgeToCheck, edgeRange(i), [tol](auto f, auto s) {
418 return FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(f.p, s.p, tol);
419 }))
420 return ConnectionResult(true, i);
421 }
422 return ConnectionResult(false, -1);
423 } else if constexpr (mydimension == 3) {
424 auto edgeRange = [&cps](int b) {
425 switch (b) {
426 case 0:
427 return cps.hyperSurfFront({1, 2});
428 case 1:
429 return cps.hyperSurfBack({1, 2});
430 case 2:
431 return cps.hyperSurfFront({0, 1});
432 case 3:
433 return cps.hyperSurfBack({0, 1});
434 case 4:
435 return cps.hyperSurfFront({0, 2});
436 case 5:
437 return cps.hyperSurfBack({0, 2});
438 default:
439 assert("No valid boundary index");
440 }
441 __builtin_unreachable();
442 };
443
444 auto theEdgeToCheck = edgeRange(boundary);
445 for (auto i : Dune::range(6)) {
446 if (boundary == i)
447 continue;
448 if (std::ranges::equal(theEdgeToCheck, edgeRange(i), [tol](auto f, auto s) {
449 return FloatCmp::eq<FieldVector<ctype, coorddimension>, FloatCmp::CmpStyle::absolute>(f.p, s.p, tol);
450 }))
451 return ConnectionResult(true, i);
452 }
453 return ConnectionResult(false, -1);
454 } else
455 DUNE_THROW(Dune::NotImplemented,
456 "Higher dimensions not implemented yet. But should be straight-forward to do so, somebody just has "
457 "to come up with a resenable indexing of hyper surfaces.");
458 }

465 /* @brief Calculate NURBS and control point net for a given local coordinate. */
466 auto calculateNurbsAndControlPointNet(const LocalCoordinate& u) const {
467 auto subNetStart = Splines::findSpan(patchData_.degree, u, patchData_.knotSpans);
468 auto cpCoordinateNet = Splines::netOfSpan(subNetStart, patchData_.degree,
469 Splines::extractControlCoordinates(patchData_.controlPoints));
470 auto nurbsLocalView = nurbs_.localView();
471 nurbsLocalView.bind(subNetStart);
472 return std::make_tuple(nurbsLocalView, cpCoordinateNet, subNetStart);
473 }
474
475 /* @brief Calculate zero first and second derivatives of the position for a given local coordinate. */
476 static auto zeroFirstAndSecondDerivativeOfPositionImpl(const LocalCoordinate& u, const NurbsLocalView& nurbsLocalView,
477 const ControlPointCoordinateNetType& localControlPointNet) {
478 return std::make_tuple(GeometryKernel::position(u, nurbsLocalView, localControlPointNet),
479 GeometryKernel::jacobianTransposed(u, nurbsLocalView, localControlPointNet),
480 GeometryKernel::hessian(u, nurbsLocalView, localControlPointNet));
481 }
