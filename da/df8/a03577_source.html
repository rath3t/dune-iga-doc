354 // compute the grid indices and ids
355 void update(GridImp* grid) {
356 grid_ = grid;
357 localIdSet_->update();
358
359 globalIdSet_->update();
360
361 // //////////////////////////////////////////
362 // Create the index sets
363 // //////////////////////////////////////////
364 for (int i = levelIndexSets_.size(); i <= maxLevel(); i++) {
365 auto p = std::make_unique<LevelIndexSet>();
366 levelIndexSets_.emplace_back(std::move(p));
367 }
368
369 for (int i = 0; i <= maxLevel(); i++)
370 if (levelIndexSets_[i])
371 levelIndexSets_[i]->update(*grid_, i);
372
373 leafIndexSet_->update(*grid_);
374 }
