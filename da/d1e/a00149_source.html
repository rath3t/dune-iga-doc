22template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
23class TrimmedLocalGeometryImpl<2, coorddim, GridImp, localGeometryTag>
24{
25public:
26 using ctype = typename GridImp::ctype;
27
28 static constexpr int mydimension = 2;
29 using Trimmer = typename GridImp::Trimmer;
30
31 static constexpr int coorddimension = coorddim;
32 static constexpr int codim = coorddimension - mydimension;
33 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
34 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
35
36 using HostGeometry = typename Trimmer::TrimmerTraits::template Codim<0>::HostParameterSpaceGridEntity::Geometry;
37 using TrimData = typename Trimmer::ElementTrimData;
38
39 using ParameterSpaceElement = typename GridImp::Trimmer::template Codim<0>::ParameterSpaceGridEntity;
40
41 using LocalCoordinate = FieldVector<ctype, mydimension>;
42 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
43 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
44 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
45 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
46 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
47 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
48 using Volume = ctype;
49
50 // type of the LocalView of the patch geometry
51 using GeometryLocalView = typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
52 ctype>::template GeometryLocalView<codim, Trimmer>;
53
55 TrimmedLocalGeometryImpl() = default;
56 explicit TrimmedLocalGeometryImpl(const HostGeometry& hostGeometry, const ParameterSpaceElement* element)
57 : hostGeometry_{hostGeometry},
58 trimData_{element->trimData()},
59 element_(element) {}
60
61 bool operator==(const TrimmedLocalGeometryImpl& b) const {
62 return b.hostGeometry_ == this->hostGeometry_;
63 }
64
67 [[nodiscard]] GeometryType type() const {
68 if (Preferences::getInstance().reportTrimmedElementGeometryTypeAsNone())
69 return GeometryTypes::none(mydimension);
70 return GeometryTypes::cube(mydimension);
71 }
72
73 // return whether we have an affine mapping (true for straight lines)
74 [[nodiscard]] bool affine() const {
75 return true;
76 }
77
78 // TODO Write something to get the integration Rule for this element (maybe give a pointer to parameterspace element
79 // instead of trimData)
80
81 // return the number of corners of this element. Corners are numbered 0...n-1
82 [[nodiscard]] int corners() const {
83 return trimData_.size(2);
84 }
85
86 // @todo write a test for that
87 GlobalCoordinate center() const {
88 // average of corners
89 GlobalCoordinate c{0, 0};
90 for (auto i : Dune::range(corners()))
91 c += corner(i);
92
93 return c / corners();
94 }
95
96 // access to coordinates of corners. Index is the number of the corner
97 GlobalCoordinate corner(int i) const {
98 auto vData = trimData_.vertex(i);
99 if (vData.isHost)
100 return hostGeometry_.corner(Transformations::mapToDune(2, vData.idx));
101
102 return vData.geometry.value();
103 }
104
107 GlobalCoordinate global(const LocalCoordinate& local) const {
108 return hostGeometry_.global(local);
109 }
110
113 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
114 return hostGeometry_.jacobianTransposed(local);
115 }
116
119 LocalCoordinate local(const GlobalCoordinate& global) const {
120 return hostGeometry_.local(global);
121 }
122
123 // Returns true if the point is in the current element
124 // @todo
125 bool checkInside(const LocalCoordinate& local) const {
126 return true;
127 }
128
129 // @todo
130 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
131 return hostGeometry_.volume();
132 }
133
134 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
135 return hostGeometry_.jacobianInverseTransposed(local);
136 }
137
138 auto getQuadratureRule(const std::optional<int>& p_order = std::nullopt,
139 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) const {
140 return element_->getQuadratureRule(p_order, qt);
141 }
142
143private:
144 HostGeometry hostGeometry_;
145 TrimData trimData_;
146 const ParameterSpaceElement* element_;
147};

149template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
150class TrimmedLocalGeometryImpl<1, coorddim, GridImp, localGeometryTag>
151{
152public:
153 using ctype = typename GridImp::ctype;
154
155 static constexpr int mydimension = 1;
156 using Trimmer = typename GridImp::Trimmer;
157
158 static constexpr int coorddimension = coorddim;
159 static constexpr int codim = coorddimension - mydimension;
160 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
161 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
162 using LocalCoordinate = FieldVector<ctype, mydimension>;
163 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
164 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
165 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
166 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
167 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
168 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
169 using Volume = ctype;
170
171 // type of the LocalView of the patch geometry
172 using GeometryLocalView = typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
173 ctype>::template GeometryLocalView<codim, Trimmer>;
174
176 TrimmedLocalGeometryImpl() = default;
177 explicit TrimmedLocalGeometryImpl(const GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>& patchGeometry)
178 : patchGeometry{patchGeometry} {}
179
180 bool operator==(const TrimmedLocalGeometryImpl& b) const {
181 return Dune::FloatCmp::eq(b.patchGeometry.corner(0) == this->patchGeometry.corner(0)) and
182 Dune::FloatCmp::eq(b.patchGeometry.corner(1) == this->patchGeometry.corner(1));
183 };
184
187 [[nodiscard]] GeometryType type() const {
188 return GeometryTypes::cube(mydimension);
189 }
190
191 // return whether we have an affine mapping (true for straight lines)
192 [[nodiscard]] bool affine() const {
193 // handy if the trims are straight lines for other algorithms
194 if constexpr (codim == 0)
195 return true;
196 else
197 return patchGeometry.degree()[0] == 1;
198 }
199
200 // return the number of corners of this element. Corners are numbered 0...n-1
201 [[nodiscard]] int corners() const {
202 return patchGeometry.corners();
203 }
204
205 GlobalCoordinate center() const {
206 return patchGeometry.center();
207 }
208
209 // access to coordinates of corners. Index is the number of the corner
210 GlobalCoordinate corner(int i) const {
211 return patchGeometry.corner(i);
212 }
213
216 GlobalCoordinate global(const LocalCoordinate& local) const {
217 return patchGeometry.global(local);
218 }
219
222 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
223 return patchGeometry.jacobianTransposed(local);
224 }
225
228 LocalCoordinate local(const GlobalCoordinate& global) const {
229 return patchGeometry.local(global);
230 }
231
232 // Returns true if the point is in the current element
233 // @todo
234 bool checkInside(const LocalCoordinate& local) const {
235 return true;
236 }
237
238 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
239 return patchGeometry.volume();
240 }
241
242 // The Jacobian matrix of the mapping from the reference element to this element
243 // @todo not yet implemented
244 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
245 return patchGeometry.jacobianInverseTransposed(local);
246 }
247
248private:
249 GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype> patchGeometry;
250};
