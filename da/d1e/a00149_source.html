30template <int dim, int dimworld, typename ScalarType>
31void TrimmerImpl<dim, dimworld, ScalarType>::createAndSaveElementInfo(const std::tuple<unsigned int, unsigned int, int>& indices,
32 const HostEntity<0>& ele, bool trimmed) {
33 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
34
35 std::optional<GlobalIdType> fatherId{};
36 if (ele.hasFather())
37 fatherId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = globalIdSetParameterSpace.id(ele.father())};
38
39 GlobalIdType elementId = makeElementID(ele);
40
41 const unsigned int unTrimmedElementIndex = std::get<0>(indices);
42 const unsigned int trimmedElementIndex = std::get<1>(indices);
43 const int newLevel = std::get<2>(indices);
44 EntityInfo<0> elementInfo = {
45 .indexInLvlStorage = trimmedElementIndex + unTrimmedElementIndex,
46 .unTrimmedIndexInLvl = unTrimmedElementIndex,
47 .lvl = newLevel,
48 .stemFromTrim = trimmed,
49 .id = elementId,
50 .hostSeed = ele.seed(),
51 .fatherId = fatherId,
52 };
53
54 entityContainer_.idToElementInfoMap.insert({elementId, elementInfo});
55
56 // If we have a father we have to add us as his son, this can be faster, we can store in decendantIds,
57 // the indexInLvlStorage and lvl, which would provide faster access
58 if (fatherId.has_value()) {
59 entityContainer_.idToElementInfoMap.at(fatherId.value()).decendantIds.push_back(elementId);
60 entityContainer_.template entity<0>(fatherId.value(), newLevel - 1).entityInfo_.decendantIds.push_back(elementId);
61 }
62 std::get<0>(entityContainer_.entityImps_.back()).emplace_back(grid_, ele, elementInfo);
63}

65template <int dim, int dimworld, typename ScalarType>
66void TrimmerImpl<dim, dimworld, ScalarType>::collectElementEdges(int level, const HostEntity<0>& ele,
67 const ElementTrimData& eleTrimData) {
68 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
69 GlobalIdType elementId = makeElementID(ele);
70
71 auto& elementEdgeIndices = entityContainer_.globalEdgesIdOfElementsMap_[elementId];
72 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
73
74 // Define some lambdas (can be moved to seperate functions later)
75 auto addHostEdge = [&](const unsigned int localEdgeIndex) {
76 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
77 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostEdgeId};
78 elementEdgeIndices.emplace_back(edgeId);
79
80 if (entityContainer_.idToEdgeInfoMap.contains(edgeId))
81 return;
82
83 auto edge = ele.template subEntity<1>(localEdgeIndex);
84 EntityInfo<1> edgeInfo{
85 .indexInLvlStorage = indexSet.index(edge), .lvl = level, .stemFromTrim = false, .id = edgeId, .hostSeed = edge.seed()};
86 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
87
88 // store vertex ids of edges, for subIndex method of indexSet
89 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
90
91 if (edgeVertexIndices.size() < 2) {
92 const auto& cube = ReferenceElements<ctype, mydimension>::cube();
93 for (auto vertexLocalIndexWRTElement : cube.subEntities(localEdgeIndex, 1, 2)) {
94 auto hostVertexId = globalIdSetParameterSpace.subId(ele, vertexLocalIndexWRTElement, 2);
95 GlobalIdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
96 edgeVertexIndices.emplace_back(vertexId);
97 }
98 }
99 };

101 auto addTrimmedHostEdge = [&](int localEdgeIndex, const typename ElementTrimData::EdgeInfo& edgeOfTrimmedElement) {
102 // To get neighborhoud information we save the original hostId
103 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
104 GlobalIdType edgeId = idForTrimmedHostEdge(hostEdgeId, edgeOfTrimmedElement);
105
106 // \todo We have to somehow check if this edge is already present in the idSet
107 // Check for hostId, then for coordinates
108 // We have to get a unique .id
109
110 // GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::newId,
111 // .id = globalIdSet_->newFreeIndex(),
112 // .hostId = std::make_optional(hostEdgeId)};
113
114 };
115
116 if (eleTrimData.flag() == ElementTrimFlag::full) {
117 for (auto localEdgeIndex : Dune::range(ele.subEntities(1))) {
118 addHostEdge(localEdgeIndex);
119 }
120 } else /* trimmed */ {
121 for (auto edgeOfTrimmedElement : eleTrimData.edges()) {
122 if (edgeOfTrimmedElement.isHost and not edgeOfTrimmedElement.isTrimmed) {
123 // This is a host Edge with full length (do some as above)
124 int localEdgeIndex = Util::edgeIndexMapping[edgeOfTrimmedElement.idx];
125 addHostEdge(localEdgeIndex);
126 } else if (edgeOfTrimmedElement.isHost and edgeOfTrimmedElement.isTrimmed) {
127 // This is a host Edge which is partially trimmed
128 auto localEdgeIndex = Util::edgeIndexMapping[edgeOfTrimmedElement.idx];
129 addTrimmedHostEdge(localEdgeIndex, edgeOfTrimmedElement);
130 }
131 }
132 }

162template <int dim, int dimworld, typename ScalarType>
163void TrimmerImpl<dim, dimworld, ScalarType>::createSubEntities(int level) {
164 using EdgeEntity = typename TrimmerTraits::template Codim<1>::ParameterSpaceGridEntity;
165 using VertexEntity = typename TrimmerTraits::template Codim<2>::ParameterSpaceGridEntity;
166
167 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
168 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
169 auto& vertexMap = entityContainer_.idToVertexInfoMap.back();
170 auto& edgeMap = entityContainer_.idToEdgeInfoMap;
171
172 // we are resizing the containers, so we can add the entities in the indexSet order, the index is obtained from
173 // the infos (also this is a bit more efficient as pushing back all the time)
174
175 auto& vertexContainer = std::get<2>(entityContainer_.entityImps_.back());
176 vertexContainer.resize(vertexMap.size());
177 for (auto& [vertexId, vertexInfo] : vertexMap) {
178 auto vertex = parameterSpaceGrid_->entity(vertexInfo.hostSeed);
179 vertexContainer[vertexInfo.indexInLvlStorage] = VertexEntity{grid_, vertex, vertexInfo};
180 }
181
182 auto& edgeContainer = std::get<1>(entityContainer_.entityImps_.back());
183 edgeContainer.resize(entityContainer_.sizeOfInfos(1, level));
184 for (auto& [edgeId, edgeInfo] : edgeMap) {
185 if (edgeInfo.lvl != level)
186 continue;
187 auto edge = parameterSpaceGrid_->entity(edgeInfo.hostSeed);
188 edgeContainer[edgeInfo.indexInLvlStorage] = EdgeEntity{grid_, edge, edgeInfo};
189 }
190}
