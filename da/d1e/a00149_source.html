10template <int dim, int dimworld, typename ScalarType>
11auto TrimmerImpl<dim, dimworld, ScalarType>::makeElementID(const HostEntity<0>& ele) -> GlobalIdType {
12 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
13 auto hostId = globalIdSetParameterSpace.id(ele);

17template <int dim, int dimworld, typename ScalarType>
18void TrimmerImpl<dim, dimworld, ScalarType>::createAndSaveElementInfo(const std::tuple<int, int, int>& indices,
19 const HostEntity<0>& ele, bool trimmed) {
20 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
21
22 std::optional<GlobalIdType> fatherId{};
23 if (ele.hasFather())
24 fatherId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = globalIdSetParameterSpace.id(ele.father())};
25
26 GlobalIdType elementId = makeElementID(ele);
27
28 const int unTrimmedElementIndex = std::get<0>(indices);
29 const int trimmedElementIndex = std::get<1>(indices);
30 const int newLevel = std::get<2>(indices);
31 EntityInfo<0> elementInfo = {
32 .indexInLvlStorage = trimmedElementIndex + unTrimmedElementIndex,
33 .unTrimmedIndexInLvl = unTrimmedElementIndex,
34 .lvl = newLevel,
35 .stemFromTrim = trimmed,
36 .id = elementId,
37 .hostSeed = ele.seed(),
38 .fatherId = fatherId,
39 };
40
41 entityContainer_.idToElementInfoMap.insert({elementId, elementInfo});
42
43 // If we have a father we have to add us as his son, this can be faster, we can store in decendantIds,
44 // the indexInLvlStorage and lvl, which would provide faster access
45 if (fatherId.has_value()) {
46 entityContainer_.idToElementInfoMap.at(fatherId.value()).decendantIds.push_back(elementId);
47 entityContainer_.template entity<0>(fatherId.value(), newLevel - 1).entityInfo_.decendantIds.push_back(elementId);
48 }
49 std::get<0>(entityContainer_.entityImps_.back()).emplace_back(grid_, ele, elementInfo);
50}

52template <int dim, int dimworld, typename ScalarType>
53void TrimmerImpl<dim, dimworld, ScalarType>::collectElementEdges(int level, const HostEntity<0>& ele,
54 const ElementTrimData& eleTrimData) {
55 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
56 GlobalIdType elementId = makeElementID(ele);
57
58 auto& elementEdgeIndices = entityContainer_.globalEdgesIdOfElementsMap_[elementId];
59 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
60
61 // Define some lambdas (can be moved to seperate functions later)
62 auto addHostEdge = [&](const unsigned int localEdgeIndex) {
63 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
64 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostEdgeId};
65 elementEdgeIndices.emplace_back(edgeId);
66
67 if (entityContainer_.idToEdgeInfoMap.contains(edgeId))
68 return;
69
70 auto edge = ele.template subEntity<1>(localEdgeIndex);
71 EntityInfo<1> edgeInfo{.indexInLvlStorage = indexSet.index(edge), .lvl = level, .stemFromTrim = false, .id = edgeId};
72 entityContainer_.idToEdgeInfoMap.insert({edgeId, edgeInfo});
73
74 auto& edgeContainer = std::get<1>(entityContainer_.entityImps_.back());
75 edgeContainer.emplace_back(grid_, edge, edgeInfo);
76
77 // store vertex ids of edges, for subIndex method of indexSet
78 auto& edgeVertexIndices = entityContainer_.globalVertexIdOfEdgesMap_[edgeId];
79 if (edgeVertexIndices.size() < 2) {
80 const auto& cube = ReferenceElements<ctype, mydimension>::cube();
81 for (auto vertexLocalIndexWRTElement : cube.subEntities(localEdgeIndex, 1, 2)) {
82 auto hostVertexId = globalIdSetParameterSpace.subId(ele, vertexLocalIndexWRTElement, 2);
83 GlobalIdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
84 edgeVertexIndices.emplace_back(vertexId);
85 }
86 }
87 };
88
89 auto addTrimmedHostEdge = [&](int localEdgeIndex, const typename ElementTrimData::EdgeInfo& edgeOfTrimmedElement) {
90 // To get neighborhoud information we save the original hostId
91 auto hostEdgeId = globalIdSetParameterSpace.subId(ele, localEdgeIndex, 1);
92
93 // \todo We have to somehow check if this edge is already present in the idSet
94 // Check for hostId, then for coordinates
95
96 GlobalIdType edgeId = {.entityIdType = GlobalIdType::EntityIdType::newId,
97 .id = globalIdSet_->newFreeIndex(),
98 .hostId = std::make_optional(hostEdgeId)};
99
100 };
101
102 if (eleTrimData.flag() == ElementTrimFlag::full) {
103 for (auto localEdgeIndex : Dune::range(ele.subEntities(1))) {
104 addHostEdge(localEdgeIndex);
105 }
106 } else /* trimmed */ {
107 for (auto edgeOfTrimmedElement : eleTrimData.edges()) {
108 if (edgeOfTrimmedElement.isHost and not edgeOfTrimmedElement.isTrimmed) {
109 // This is a host Edge with full length (do some as above)
110 int localEdgeIndex = Util::edgeIndexMapping[edgeOfTrimmedElement.idx];
111 addHostEdge(localEdgeIndex);
112 } else if (edgeOfTrimmedElement.isHost and edgeOfTrimmedElement.isTrimmed) {
113 // This is a host Edge which is partially trimmed
114 auto localEdgeIndex = Util::edgeIndexMapping[edgeOfTrimmedElement.idx];
115 addTrimmedHostEdge(localEdgeIndex, edgeOfTrimmedElement);
116 }
117 }
118 }
119}

121template <int dim, int dimworld, typename ScalarType>
122void TrimmerImpl<dim, dimworld, ScalarType>::collectElementVertices(int level, const HostEntity<0>& ele,
123 const ElementTrimData& eleTrimData) {
124 GlobalIdType elementId = makeElementID(ele);
125 auto& indexSet = parameterSpaceGrid_->levelGridView(level).indexSet();
126
127 auto& elementVertexIndices = entityContainer_.globalVerticesIdOfElementsMap[elementId];
128 auto& globalIdSetParameterSpace = parameterSpaceGrid_->globalIdSet();
129
130 if (eleTrimData.flag() == ElementTrimFlag::full) {
131 for (int localVertexId = 0; localVertexId < ele.subEntities(2); ++localVertexId) {
132 // setup all vertex indices for given element
133 auto hostVertexId = globalIdSetParameterSpace.subId(ele, localVertexId, 2);
134 IdType vertexId = {.entityIdType = GlobalIdType::EntityIdType::host, .id = hostVertexId};
135 elementVertexIndices.emplace_back(vertexId);
136
137 if (entityContainer_.idToVertexInfoMap.back().contains(vertexId))
138 continue;
139
140 auto vertex = ele.template subEntity<2>(localVertexId);
141 EntityInfo<2> vertexInfo{
142 .indexInLvlStorage = indexSet.index(vertex), .lvl = level, .stemFromTrim = false, .id = vertexId};
143 entityContainer_.idToVertexInfoMap.back().insert({vertexId, vertexInfo});
144
145 auto& vertexContainer = std::get<2>(entityContainer_.entityImps_.back());
146 vertexContainer.emplace_back(grid_, vertex, vertexInfo);
147 }
148 } else /* trimmed */ {
149
150 }
151}
