17auto clipElementRectangle(const auto& element,
18 const auto& patchTrimData) -> std::tuple<ElementTrimFlag, ClippingResult> {
19 using namespace Clipper2Lib;
20
21 auto eleGeo = element.geometry();
22 static constexpr int numberOfCorners = 4;
23
24 PathD eleRect;
25 for (const auto vertexIndex : Dune::range(numberOfCorners)) {
26 auto corner = eleGeo.corner(vertexIndexMapping[vertexIndex]);
27 eleRect.emplace_back(corner[0], corner[1], vertexIndex + 1);
28 }
29
30 const auto& trimmingCurves = patchTrimData.clipperLoops();
31
32 // Element is trimmed
33 ClipperD clipper(8);
34 ClippingResult result(eleRect);
35
36 auto checkParallel = [&](const auto& curve, const int edgeIndex) -> bool {
37 return curve.affine() and std::get<0>(findIntersectionLinearCurveAndLine(
38 curve, eleGeo.corner(vertexIndexMapping[edgeIndex]), edgeDirections[edgeIndex],
39 {curve.domainMidPoint()[0], 0.5})) == IntersectionCurveAndLine::parallel;
40 };
41
42 clipper.SetZCallback([&](const PointD& rectangleFirstPoint, const PointD& rectangleSecondPoint,
43 const PointD& firstTrimmingCurvePoint, const PointD& secondTrimmingCurvePoint,
44 const PointD& intersectionPoint) {
45 assert(rectangleFirstPoint.z > 0 and rectangleFirstPoint.z < 5 &&
46 "Intersection should only occur with the rectangle edges");
47 assert(rectangleSecondPoint.z > 0 and rectangleSecondPoint.z < 5 &&
48 "Intersection should only occur with the rectangle edges");
49
50 const auto edgeIdx = giveEdgeIdx(rectangleFirstPoint.z - 1, rectangleSecondPoint.z - 1);
51
52 auto firstTrimmingCurveIndices = patchTrimData.getIndices(firstTrimmingCurvePoint.z);
53 auto secondTrimmingCurveIndices = patchTrimData.getIndices(secondTrimmingCurvePoint.z);
54
55 const auto [isHost, idx] = isCornerVertex(intersectionPoint, eleRect);
56 if (isHost) {
57 result.addOriginalVertex(idx, secondTrimmingCurvePoint.z);
58 return;
59 }
60
61 auto intersectionIndices = patchTrimData.getIndices(intersectionPoint.z);
62
63 // std::cout << "First Curve: " << firstTrimmingCurvePoint.z << " " << firstTrimmingCurveIndices << std::endl;
64 // std::cout << "Second Curve: " << secondTrimmingCurvePoint.z << " " << secondTrimmingCurveIndices << std::endl;
65 // std::cout << "Intersection: " << intersectionPoint.z << " " << intersectionIndices << std::endl;
66
67 assert(firstTrimmingCurveIndices.loop == secondTrimmingCurveIndices.loop &&
68 "The points of the trimming curves should be on the same loop");
69
70 assert(firstTrimmingCurveIndices.curve == secondTrimmingCurveIndices.curve or
71 secondTrimmingCurveIndices.curve == intersectionIndices.curve or
72 firstTrimmingCurveIndices.curve == intersectionIndices.curve &&
73 "The indices of the trimming curves should be the same or the intersection point should be on one of "
74 "the two second curves");
75 const auto vertexZValue = secondTrimmingCurvePoint.z;
76
77 // Now check that we don't have a parallel intersection
78 if (checkParallel(patchTrimData.getCurve(vertexZValue), edgeIdx)) {
79 return;
80 }
81
82 result.addNewVertex(edgeIdx, intersectionPoint, vertexZValue);
83 });
84
85 PathsD resultClosedPaths{};
86 PathsD resultOpenPaths{};
87
88 clipper.AddClip(trimmingCurves);
89 clipper.AddSubject({eleRect});
90
91 clipper.Execute(ClipType::Intersection, FillRule::NonZero, resultClosedPaths, resultOpenPaths);
92 assert(resultOpenPaths.empty() && "There should be no open paths in the clipping result");
93
94 auto isFullElement = [&](const auto& clippedEdges) {
95 return clippedEdges.size() == 4 and
96 (clippedEdges[0].z < 5 and clippedEdges[1].z < 5 and clippedEdges[2].z < 5 and clippedEdges[3].z < 5);
97 };
98
99 if (resultClosedPaths.empty())
100 return std::make_tuple(ElementTrimFlag::empty, ClippingResult{eleRect});
101
102 if (resultClosedPaths.size() == 1 and isFullElement(resultClosedPaths.front())) {
103 return std::make_tuple(ElementTrimFlag::full, ClippingResult{eleRect});
104 }
105
106 // Add corner vertices that are not yet present in the clip path
107 for (const auto& pt : resultClosedPaths.front()) {
108 if (auto [isHost, idx] = isCornerVertex(pt, eleRect); isHost)
109 result.addOriginalVertex(idx);
110 }
111
112 // Now we have to check for points inside the elements
113 for (auto i : Dune::range(patchTrimData.loops().size())) {
114 for (const auto& pt : patchTrimData.getPointsInPatch(i)) {
115 if (const PointD cp{pt.pt[0], pt.pt[1]}; PointInPolygon(cp, eleRect) == PointInPolygonResult::IsInside) {
116 result.addInsideVertex(cp, pt.curveIdxI, pt.curveIdxJ, i);
117 }
118 }
119 }
120
121 // Add startpoints of trimming curves to the vertices if they are on one of the element edges
122 for (auto cI = 0; cI < patchTrimData.loops().front().curves().size(); ++cI) {
123 const auto& curve = patchTrimData.loops().front().curves()[cI];
124
125 for (int i = 0; i < 2; ++i) {
126 auto pt = curve.corner(i);
127 PointD ptClipper{pt[0], pt[1]};
128
129 auto [isHost, idx] = isCornerVertex(ptClipper, eleRect);
130 if (isHost) {
131 result.addOriginalVertex(idx);
132 continue;
133 }
134
135 for (auto e = 0; e < edgeLookUp.size(); ++e) {
136 if (const auto& edgeIdx = edgeLookUp[e];
137 GeometryKernel::isPointOnLineSegment(pt, eleGeo.corner(edgeIdx.front()), eleGeo.corner(edgeIdx.back())) &&
138 !checkParallel(curve, e)) {
139 result.addNewVertex(e, ptClipper, patchTrimData.getZValue(0, cI, std::numeric_limits<u_int64_t>::infinity()));
140 break;
141 }
142 }
143 }
144 }
145
146 // Check again if element is full after adding addtional vertices
147 if (result.vertices_.size() == 4 and
148 std::ranges::all_of(result.vertices_, [](const ClippingResult::Vertex& v) { return v.isHost(); })) {
149 return std::make_tuple(ElementTrimFlag::full, ClippingResult{eleRect});
150 }
151
152 result.finish(patchTrimData);
153
154 // While developing
155 // result.report();
156
157 return std::make_tuple(ElementTrimFlag::trimmed, result);
158}
