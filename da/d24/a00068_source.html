15inline int giveEdgeIdx(const std::size_t e1, const std::size_t e2) {
16 if ((e1 == 0 and e2 == 1) or (e1 == 1 and e2 == 0))
17 return 0;
18 if ((e1 == 1 and e2 == 2) or (e1 == 2 and e2 == 1))
19 return 1;
20 if ((e1 == 2 and e2 == 3) or (e1 == 3 and e2 == 2))
21 return 2;
22 if ((e1 == 3 and e2 == 0) or (e1 == 0 and e2 == 3))
23 return 3;
24 DUNE_THROW(Dune::GridError, "No corresponding edge Index for Input");
25}

27inline bool isConsecutive(const std::size_t e1, const std::size_t e2) {
28 assert(e2 < 4);
29 if (e1 == 3)
30 return e2 == 0;
31 if (e1 > e2)
32 return false;
33 return e2 == e1 + 1;
34}

171 void finish(const auto& patchTrimData) {
172 // Sort the points in counter clockwise manner such that the first point is in the lower left
173 const auto minVertex = lowerLeftVertex();
174 std::ranges::sort(vertices_, [&](const Vertex& aV, const Vertex& bV) { return comparePoints(aV, bV, minVertex); });
175
176 // Resort that we always start on a hostVertex (if there is one)
177 if (const auto it = std::ranges::find_if(vertices_, [](const Vertex& vV) { return vV.isHost(); });
178 it != vertices_.end())
179 std::ranges::rotate(vertices_, it);
180
181 // Now resort InsideVertices (sometimes they end up on the wrong position)
182 for (const auto i : std::views::iota(0ul, vertices_.size())) {
183 if (vertices_[i].isInside()) {
184 const auto insideVertex = vertices_[i];
185 auto it = vertices_.begin() + static_cast<long>(i);
186 const size_t indexBeforeIt =
187 it == vertices_.begin() ? vertices_.size() - 1 : std::ranges::distance(vertices_.begin(), it) - 1;
188 const size_t indexAfterIt =
189 it == std::prev(vertices_.end()) ? 0 : std::ranges::distance(vertices_.begin(), it) + 1;
190
191 // Check vertex before if they line up correctly
192 const auto vertexBefore = vertices_[indexBeforeIt];
193 const auto vertexAfter = vertices_[indexAfterIt];
194
195 if (vertexBefore.isNew() and vertexAfter.isNew()) {
196 auto indices = patchTrimData.getIndices(vertexBefore.zValue());
197 if (indices.loop == insideVertex.loop() and indices.curve == insideVertex.formerCurve())
198 continue;
199 }
200 // If we end up here sort the vertex to a correct place
201 auto correctVertexIt = findCorrespondingVertexIt(insideVertex, patchTrimData);
202 assert(correctVertexIt != vertices_.end() && "No corresponding newVertex found");
203
204 // Add the insideVertex at correctVertexIt+1
205 const auto correctItForInsideVertex = correctVertexIt + 1;
206 if (std::distance(vertices_.begin(), correctItForInsideVertex) < i) {
207 vertices_.insert(correctItForInsideVertex, insideVertex);
208 vertices_.erase(vertices_.begin() + static_cast<long>(i + 1));
209 } else {
210 std::ranges::rotate(it, it + 1, correctItForInsideVertex);
211 }
212 }
213 }
214 }

243 // Sort the points based on polar angle with respect to the reference point
244 inline static auto comparePoints = [](const Vertex& aV, const Vertex& bV, const Vertex& referenceV) -> bool {
245 auto polarAngle = [](const auto& p, const auto& reference) -> double {
246 return atan2(p.y - reference.y, p.x - reference.x);
247 };
248
249 const auto a = aV.pt;
250 const auto b = bV.pt;
251 const auto reference = referenceV.pt;
252 const double angleA = polarAngle(a, reference);
253 const double angleB = polarAngle(b, reference);
254
255 if (angleA != angleB)
256 return angleA < angleB;
257
258 // If two points have the same angle, sort based on distance to the reference point
259 return (a.x - reference.x) * (a.x - reference.x) + (a.y - reference.y) * (a.y - reference.y) <
260 (b.x - reference.x) * (b.x - reference.x) + (b.y - reference.y) * (b.y - reference.y);
261 };
262
263 auto findCorrespondingVertexIt(const Vertex& insideVertex, const auto& patchTrimData) {
264 return std::ranges::find_if(vertices_, [&](const Vertex& vV) {
265 if (vV.isNew()) {
266 const auto indices = patchTrimData.getIndices(vV.zValue());
267 return indices.loop == insideVertex.loop() and indices.curve == insideVertex.formerCurve();
268 } else
269 return false;
270 });
271 }

273 [[nodiscard]] const Vertex& lowerLeftVertex() const {
274 return *std::ranges::min_element(vertices_, [&](const Vertex& aV, const Vertex& bV) {
275 const auto a = aV.pt;
276 const auto b = bV.pt;
277 return (a.y < b.y) || (a.y == b.y && a.x < b.x);
278 });
279 }
