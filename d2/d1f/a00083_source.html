201 auto subEntities(int i, int c, int cc) const {
202 // auto subEntityName = [](std::size_t codim)-> std::string {

208 if (trimData_.flag() == ElementTrimFlag::full) {
209 auto range = cubeGeometry.subEntities(i, c, cc);
210 return SubEntityRangeImpl(range.begin(), range.end(), true, i, c, cc);
211 }
212 if (trimData_.flag() == ElementTrimFlag::trimmed) {
213 if (cc < c)
214 return SubEntityRangeImpl(dummyArray.begin(), dummyArray.end());
215 else if (i == 0 and c == 0 and
216 cc == 0) // iterate over the entity itself (only one element in storage with value zero)
217 {
218 SubEntityRangeImpl range(elementIndexDummy.begin(), elementIndexDummy.end());
219 return range;
220 } else
221
222 if (i == 0 and c == 0 and cc == 1) // iterate over the edges of the element
223 {
224 SubEntityRangeImpl range(edgeIndices.begin(), edgeIndices.end());
225 return range;
226 } else if (c == 0 and cc == 2) // iterate over the vertices of the element
227 {
228 SubEntityRangeImpl range(verticesIndices.begin(), verticesIndices.end() - 1);
229 return range;
230 }
231
232 else if (c == 1 and cc == 2) // iterate over the vertices of edge
233 {
234 SubEntityRangeImpl range(verticesIndices.begin() + i, verticesIndices.begin() + i + 2);
235 return range;
236 }
237
238 else if (c == cc) // iterate over the vertices of edge
239 {
240 if (c == 2) {
241 SubEntityRangeImpl range(verticesIndices.begin() + i, verticesIndices.begin() + i + 1);
242 return range;
243 } else if (cc == 1) {
244 SubEntityRangeImpl range(edgeIndices.begin() + i, edgeIndices.begin() + i + 1);
245 return range;
246 }
247 }
248 }

250 assert(false);

280 bool contains(std::size_t i) const {
281 if (host_)
282 return cubeGeometry.subEntities(i_, c_, cc_).contains(i);
283 else {
284 if (size_ == 0)
285 return false;
286 else if (size_ == 2) // two vertices at one edge case
287 {
288 const int back = *(end() - 1);
289 const int front = *begin();
290 return i == front or i == back;

429 explicit TrimmedReferenceElement(const TrimDataType& trimData)
430 : trimData_{trimData} {
431 if (trimData_.flag() == ElementTrimFlag::trimmed) {
432 edgeIndices.resize(trimData_.edges().size());
433 for (auto eI : Dune::range(trimData_.edges().size()))
434 edgeIndices[eI] = eI;
435
436 verticesIndices.resize(trimData_.vertices().size() + 1);
437 for (auto vI : Dune::range(trimData_.vertices().size())) {
438 verticesIndices[vI] = vI;
439 }
440 // to mimic a circular vector we add the first vertex at the end
