26inline Type typeForTypeString(const std::string& typeString) {
27 if (typeString == "NurbsCurveGeometry2D")
28 return Type::NurbsCurveGeometry2D;
29 else if (typeString == "NurbsSurfaceGeometry3D")
30 return Type::NurbsSurfaceGeometry3D;
31 else if (typeString == "BrepLoop")
32 return Type::BrepLoopType;
33 else if (typeString == "BrepTrim")
34 return Type::BrepTrimType;
35 else if (typeString == "Brep")
36 return Type::BrepType;
37 else
38 return Type::NoType;

53template <int dim, int worldDim>
54requires(dim < 3) && (worldDim >= dim)
55struct IbraNURBSData : public IbraBase
56{
57 std::array<int, dim> degree{};
58 std::array<int, dim> n_controlPoints{};
59 std::array<std::vector<double>, dim> knots;
60 Dune::DynamicMatrix<double> controlPoints;
61 std::vector<double> weights;
62
63 template <typename ControlPointType>
64 [[nodiscard]] std::vector<std::vector<ControlPointType>> transformControlPoints() const {
65 if constexpr (dim == 1) {
66 std::vector<ControlPointType> cps;
67 for (int i = 0; i < n_controlPoints[0]; ++i) {
68 cps.push_back(controlPointAt<ControlPointType>({i}));
69 }
70 return {cps};
71 } else if constexpr (dim == 2) {
72 std::vector<std::vector<ControlPointType>> vec;
73 std::vector<ControlPointType> cpTemp(n_controlPoints[1]);
74 for (int i = 0; i < n_controlPoints[0]; ++i) {
75 cpTemp.clear();
76 for (int j = 0; j < n_controlPoints[1]; ++j) {
77 cpTemp.push_back(controlPointAt<ControlPointType>({i, j}));
78 }
79 vec.push_back(cpTemp);
80 }
81 return vec;
82 }
83 __builtin_unreachable();
84 }
85
86 [[nodiscard]] std::array<std::vector<double>, dim> compileKnotVectors() const {
87 std::array<std::vector<double>, dim> knotVec;
88
89 // Insert first and last knot repeatedly
90 for (int i = 0; i < dim; ++i) {
91 knotVec[i].insert(knotVec[i].end(), knots[i][0]);
92 knotVec[i].insert(knotVec[i].end(), knots[i].begin(), knots[i].end());
93 knotVec[i].insert(knotVec[i].end(), knots[i].back());
94 }
95 return knotVec;
96 }
97
98private:
99 template <typename ControlPointType>
100 [[nodiscard]] ControlPointType controlPointAt(std::array<int, dim> idx) const {
101 int row = (dim == 1) ? idx[0] : idx[0] * n_controlPoints[1] + idx[1];
102
103 Dune::FieldVector<double, worldDim> p;
104 for (int k = 0; k < worldDim; ++k)
105 p[k] = controlPoints[row][k];
106
107 double w = weights[row];
108
109 return {p, w};
110 }
111};

125struct BrepTrim : IbraBase
126{
127 Curve2D geometry;
128 Utilities::Domain<double> domain = {-1, -1};
129
130 BrepTrim(BrepTrimRepresentation& trimRepresentation, std::vector<Curve2D>& allCurves)
131 : domain{trimRepresentation.domain},
132 IbraBase{static_cast<IbraBase&>(trimRepresentation)} {
133 auto it = std::find_if(allCurves.begin(), allCurves.end(),
134 [trimRepresentation](auto x) { return (x.key == trimRepresentation.geometry); });
135 if (it != allCurves.end())
136 geometry = *it;
137 else
138 DUNE_THROW(Dune::InvalidStateException, "Couldn't find geometry in BrepTrim: " << key);
139 }

141 template <typename TrimmingCurve>
142 auto asCurve() const {
143 using ControlPointType = typename TrimmingCurve::ControlPointType;
144 using ControlPointNetType = typename TrimmingCurve::ControlPointNetType;
145 using PatchData = typename TrimmingCurve::PatchData;
146
147 const auto cp = geometry.transformControlPoints<ControlPointType>()[0];
148 std::array dimSize{static_cast<int>(cp.size())};
149 std::array knotSpans{geometry.compileKnotVectors()};
150 ControlPointNetType controlNet{dimSize, cp};
151
152 return TrimmingCurve(PatchData(knotSpans, controlNet, std::array(geometry.degree)));
153 }
154};

190template <int worldDim>
191struct Brep : IbraBase
192{
193 std::vector<Curve2D> curves2D;
194 using SurfaceType = Surface<worldDim>;
195 std::vector<SurfaceType> surfaces;
196 std::vector<BrepLoop> loops;
197 std::vector<BrepTrim> trims;
198
199 Brep(const BrepRepresentation& representation, std::vector<Curve2D>& allCurves, std::vector<SurfaceType>& allSurfaces,
200 std::vector<Ibra::BrepLoopRepresentation>& allBrepLoopRepresentations,
201 std::vector<Ibra::BrepTrimRepresentation>& allBrepTrimRepresentations)
202 : IbraBase{static_cast<const BrepRepresentation&>(representation)} {
203 // Lambda to check if a given key string is in a list of strings
204 auto objectIsInList = [](const std::vector<std::string>& strRepr) {
205 return [&](const auto& keyString) -> bool { return std::ranges::find(strRepr, keyString) != strRepr.end(); };
206 };
207
208 // Find in all Curves
209 std::ranges::copy_if(allCurves, std::back_inserter(curves2D), objectIsInList(representation.curve_geometries_2d),
210 &Curve2D::key);
211 // and all Surfaces
212 std::ranges::copy_if(allSurfaces, std::back_inserter(surfaces),
213 objectIsInList(representation.surface_geometries_3d), &SurfaceType::key);
214
215 // Find all Trims
216 for (auto& trimRepr : allBrepTrimRepresentations) {
217 if (objectIsInList(representation.trims)(trimRepr.key))
218 trims.emplace_back(trimRepr, curves2D);
219 }
220 // All Loops
221 for (auto& loopRepr : allBrepLoopRepresentations) {
222 if (objectIsInList(representation.loops)(loopRepr.key))
223 loops.emplace_back(loopRepr, trims);
224 }
225
226 // Asserts if all geometries were found
227 assert(curves2D.size() == representation.curve_geometries_2d.size());
228 assert(surfaces.size() == representation.surface_geometries_3d.size());
229 assert(loops.size() == representation.loops.size());
230 assert(trims.size() == representation.trims.size());
231 }
232};

248template <int worldDim>
249void from_json(const json& j, Curve<worldDim>& curve) {
250 getGenerics(j, curve);
251
252 // Degree
253 j.at("degree").get_to(curve.degree[0]);
254
255 // Knots (u)
256 std::ranges::copy(j.at("knots"), std::back_inserter(curve.knots[0]));
257
258 // Poles aka Control Points
259 j.at("nb_poles").get_to(curve.n_controlPoints[0]);
260
261 curve.controlPoints.resize(curve.n_controlPoints[0], worldDim);
262
263 for (int i = 0; i < curve.n_controlPoints[0]; ++i)
264 for (int k = 0; k < worldDim; ++k)
265 curve.controlPoints[i][k] = j.at("poles")[i][k];
266
267 if (j.contains("weights"))
268 std::ranges::copy(j.at("weights"), std::back_inserter(curve.weights));
269 else {
270 curve.weights.resize(curve.n_controlPoints[0]);
271 std::ranges::fill(curve.weights, 1.0);
272 }
273}

275template <int worldDim>
276void from_json(const json& j, Surface<worldDim>& surface) {
277 getGenerics(j, surface);
278
279 // Degree
280 j.at("degree_u").get_to(surface.degree[0]);
281 j.at("degree_v").get_to(surface.degree[1]);
282
283 // Knots (u,v)
284 std::ranges::copy(j.at("knots_u"), std::back_inserter(surface.knots[0]));
285 std::ranges::copy(j.at("knots_v"), std::back_inserter(surface.knots[1]));
286
287 // Poles aka Control Points
288 j.at("nb_poles_u").get_to(surface.n_controlPoints[0]);
289 j.at("nb_poles_v").get_to(surface.n_controlPoints[1]);
290
291 int total_controlPoints = surface.n_controlPoints[0] * surface.n_controlPoints[1];
292 surface.controlPoints.resize(total_controlPoints, worldDim);
293
294 for (int i = 0; i < total_controlPoints; ++i)
295 for (int k = 0; k < worldDim; ++k)
296 surface.controlPoints[i][k] = j.at("poles")[i][k];
297
298 if (j.contains("weights"))
299 std::ranges::copy(j.at("weights"), std::back_inserter(surface.weights));
300 else {
301 surface.weights.resize(total_controlPoints);
302 std::ranges::fill(surface.weights, 1.0);
303 }
304}
