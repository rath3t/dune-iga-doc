73 lFE_(lFE) {}

78 void evaluateFunction(const FieldVector<D, dim>& in, std::vector<FieldVector<R, 1>>& out) const {
79 FieldVector<D, dim> globalIn = offset_;
80 scaling_.umv(in, globalIn);
81
82 preBasis_.evaluateFunction(globalIn, out, lFE_.currentKnotSpan_);
83 }
84
88 void evaluateJacobian(const FieldVector<D, dim>& in, std::vector<FieldMatrix<D, 1, dim>>& out) const {
89 FieldVector<D, dim> globalIn = offset_;
90 scaling_.umv(in, globalIn);
91
92 preBasis_.evaluateJacobian(globalIn, out, lFE_.currentKnotSpan_);
93
94 for (size_t i = 0; i < out.size(); i++)
95 for (int j = 0; j < dim; j++)
96 out[i][0][j] *= scaling_[j][j];
97 }
98
100 inline void partial(const typename std::array<unsigned int, dim>& order, const typename Traits::DomainType& in,
101 std::vector<typename Traits::RangeType>& out) const {
102 FieldVector<D, dim> globalIn = offset_;
103 scaling_.umv(in, globalIn);
104 preBasis_.partial(order, globalIn, out, lFE_.currentKnotSpan_);
105
106 for (size_t d = 0; d < dim; ++d)
107 for (size_t i = 0; i < out.size(); i++)
108 for (std::size_t fac = 0; fac < order[d]; ++fac)
109 out[i][0] *= scaling_[d][d];
110 }
111
119 [[nodiscard]] unsigned int order() const {
120 return *std::max_element(preBasis_.patchData_.degree.begin(), preBasis_.patchData_.degree.end());
121 }
122
125 [[nodiscard]] std::size_t size() const { return lFE_.size(); }

129
130 const NurbsLocalFiniteElement<GV, R>& lFE_;

133 // by an affine transformation. This transformation is stored in offset_ and scaling_.
134 FieldVector<D, dim> offset_;
135 DiagonalMatrix<D, dim> scaling_;

151template <int dim>
152class NurbsLocalCoefficients
153{
154 // Return i as a d-digit number in the (k+1)-nary system
155 std::array<unsigned int, dim> multiindex(unsigned int i) const {
156 std::array<unsigned int, dim> alpha;
157 for (int j = 0; j < dim; j++) {
158 alpha[j] = i % sizes_[j];
159 i = i / sizes_[j];
160 }
161 return alpha;
162 }
163
165 void setup1d(std::vector<unsigned int>& subEntity) {
166 if (sizes_[0] == 1) {
167 subEntity[0] = 0;
168 return;
169 }
170
171 /* edge and vertex numbering
172 0----0----1
173 */
174 unsigned lastIndex = 0;
175 subEntity[lastIndex++] = 0; // corner 0
176 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
177 subEntity[lastIndex++] = 0; // inner dofs of element (0)
178
179 subEntity[lastIndex++] = 1; // corner 1
180
181 assert(size() == lastIndex);
182 }
183
184 void setup2d(std::vector<unsigned int>& subEntity) {
185 unsigned lastIndex = 0;
186
187 // LocalKey: entity number , entity codim, dof indices within each entity
188 /* edge and vertex numbering
189 2----3----3
190 | |
191 | |
192 0 1
193 | |
194 | |
195 0----2----1
196 */
197
198 // lower edge (2)
199 subEntity[lastIndex++] = 0; // corner 0
200 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
201 subEntity[lastIndex++] = 2; // inner dofs of lower edge (2)
202
203 subEntity[lastIndex++] = 1; // corner 1
204
205 // iterate from bottom to top over inner edge dofs
206 for (unsigned e = 0; e < sizes_[1] - 2; ++e) {
207 subEntity[lastIndex++] = 0; // left edge (0)
208 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
209 subEntity[lastIndex++] = 0; // face dofs
210 subEntity[lastIndex++] = 1; // right edge (1)
211 }
212
213 // upper edge (3)
214 subEntity[lastIndex++] = 2; // corner 2
215 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
216 subEntity[lastIndex++] = 3; // inner dofs of upper edge (3)
217
218 subEntity[lastIndex++] = 3; // corner 3
219
220 assert(size() == lastIndex);
221 }
222
223public:
224 void init(const std::array<unsigned, dim>& sizes) {
225 sizes_ = sizes;
226
227 li_.resize(size());
228
229 // Set up array of codimension-per-dof-number
230 std::vector<unsigned int> codim(li_.size());
231
232 for (std::size_t i = 0; i < codim.size(); i++) {
233 codim[i] = 0;
234 // Codimension gets increased by 1 for each coordinate direction
235 // where dof is on boundary
236 std::array<unsigned int, dim> mIdx = multiindex(i);
237 for (int j = 0; j < dim; j++)
238 if (mIdx[j] == 0 or mIdx[j] == sizes[j] - 1)
239 codim[i]++;
240 }
241
242 // Set up index vector (the index of the dof in the set of dofs of a given subentity)
243 // Algorithm: the 'index' has the same ordering as the dof number 'i'.
244 // To make it consecutive we interpret 'i' in the (k+1)-adic system, omit all digits
245 // that correspond to axes where the dof is on the element boundary, and transform the
246 // rest to the (k-1)-adic system.
247 std::vector<unsigned int> index(size());
248
249 for (std::size_t i = 0; i < index.size(); i++) {
250 index[i] = 0;
251
252 std::array<unsigned int, dim> mIdx = multiindex(i);
253
254 for (int j = dim - 1; j >= 0; j--)
255 if (mIdx[j] > 0 and mIdx[j] < sizes[j] - 1)
256 index[i] = (sizes[j] - 1) * index[i] + (mIdx[j] - 1);
257 }
258
259 // Set up entity and dof numbers for each (supported) dimension separately
260 std::vector<unsigned int> subEntity(li_.size());
261
262 if (!subEntity.empty()) {
263 if (dim == 1) {
264 setup1d(subEntity);
265
266 } else if (dim == 2 and sizes_[0] > 1 and sizes_[1] > 1) {
267 setup2d(subEntity);
268 }
269 }
270
271 for (size_t i = 0; i < li_.size(); i++)
272 li_[i] = LocalKey(subEntity[i], codim[i], index[i]);
273 }
274
276 [[nodiscard]] std::size_t size() const {
277 return std::accumulate(sizes_.begin(), sizes_.end(), 1, std::multiplies<>());
278 }
279
281 [[nodiscard]] const LocalKey& localKey(std::size_t i) const { return li_[i]; }
282
283private:
284 // Number of shape functions on this element per coordinate direction
285 std::array<unsigned, dim> sizes_;
286
287 std::vector<LocalKey> li_;
288};

340 NurbsLocalFiniteElement(const NurbsLocalFiniteElement& other)
341 : preBasis_(other.preBasis_),
342 localBasis_(preBasis_, *this) {
343 if (other.isBound) {
344 this->bind(other.elementIdx_);

346 }

356 // @todo save correct element info
357 void bind(const std::array<int, dim>& elementIdx) {
358 const auto& patchData = preBasis_.patchData_;
359 for (size_t i = 0; i < elementIdx.size(); i++) {
360 currentKnotSpan_[i] = Dune::IGANEW::Splines::findSpan(patchData.degree[i],
361 *(preBasis_.uniqueKnotVector_[i].begin() + elementIdx[i]),
362 patchData.knotSpans[i], elementIdx[i]);
363
364 // Compute the geometric transformation from knotspan-local to global coordinates
365 localBasis_.offset_[i] = preBasis_.patchData_.knotSpans[i][currentKnotSpan_[i]];
366 localBasis_.scaling_[i][i] = preBasis_.patchData_.knotSpans[i][currentKnotSpan_[i] + 1] -
367 preBasis_.patchData_.knotSpans[i][currentKnotSpan_[i]];
368 }
369
370 // Set up the LocalCoefficients object
371 std::array<unsigned int, dim> sizes;
372 for (size_t i = 0; i < dim; i++)
373 sizes[i] = oneDimensionalSize(i);
374 localCoefficients_.init(sizes);
375 isBound = true;
376 elementIdx_ = elementIdx;
377 }
378
380 const NurbsLocalBasis<GV, R>& localBasis() const {
381 assert(isBound && "This element is not bound!");
382 return localBasis_;
383 }
384
386 const NurbsLocalCoefficients<dim>& localCoefficients() const {
387 assert(isBound && "This element is not bound!");
388 return localCoefficients_;
389 }
390
392 const NurbsLocalInterpolation<dim, NurbsLocalBasis<GV, R>>& localInterpolation() const {
393 assert(isBound && "This element is not bound!");
394 return localInterpolation_;
395 }
396
398 [[nodiscard]] unsigned size() const {
399 std::size_t r = 1;
400 for (int i = 0; i < dim; i++)
401 r *= oneDimensionalSize(i);
402 return r;
403 }
404
407 [[nodiscard]] GeometryType type() const { return GeometryTypes::cube(dim); }
408
410 [[nodiscard]] unsigned int oneDimensionalSize(int i) const { return preBasis_.patchData_.degree[i] + 1; }
411
412 const NurbsPreBasis<GV, R>& preBasis_;
413
414 NurbsLocalCoefficients<dim> localCoefficients_;
415 NurbsLocalInterpolation<dim, NurbsLocalBasis<GV, R>> localInterpolation_;
416
417 // The knot span we are bound to
418 std::array<int, dim> currentKnotSpan_;
419 NurbsLocalBasis<GV, R> localBasis_;
420 std::array<int, dim> elementIdx_{};
421 bool isBound{false};
422};

562 void createUntrimmedNodeIndices() {
563 std::array<int, dim> localSizes;
564 size_type sizeOfShapeFunctions = 1;
565 for (int i = 0; i < dim; i++) {
566 localSizes[i] = patchData_.degree[i] + 1;
567 sizeOfShapeFunctions *= localSizes[i];
568 }
569 const auto order = patchData_.degree;
570
571 auto spanSize = gridView_.impl().untrimmedElementNumbers();
572 auto extractIndex = std::views::transform([&](const auto& ele) { return gridView_.indexSet().index(ele); });
573 for (auto directIndex : elements(gridView_) | extractIndex) {
574 auto elementIdx = getIJK(directIndex, spanSize);
575
576 std::array<int, dim> currentKnotSpan;
577 for (size_t i = 0; i < elementIdx.size(); i++)
578 currentKnotSpan[i] =
579 IGANEW::Splines::findSpan(patchData_.degree[i], *(uniqueKnotVector_[i].begin() + elementIdx[i]),
580 patchData_.knotSpans[i], elementIdx[i]);
581
582 // Here magic is happening
583 for (size_type i = 0; i < sizeOfShapeFunctions; ++i) {
584 std::array<int, dim> localIJK = getIJK(i, localSizes);
585 std::array<unsigned int, dim> globalIJK;
586 for (int j = 0; j < dim; j++)
587 globalIJK[j] = std::max((int)currentKnotSpan[j] - (int)order[j], 0) + localIJK[j];
588
589 // Make one global flat index from the globalIJK tuple
590 size_type globalIdx = globalIJK[dim - 1];
591
592 for (int j = dim - 2; j >= 0; j--)
593 globalIdx = globalIdx * sizePerDirection(j) + globalIJK[j];
594
595 originalIndices_[directIndex].push_back(globalIdx);
596 }
597 }

788 class NurbsPreBasisFactoryFromDegreeElevation
789 {
790 public:

793 explicit NurbsPreBasisFactoryFromDegreeElevation(const std::array<int, dim>& degreeElevate)
794 : degreeElevate_(degreeElevate) {}
795
796 template <class GridView>
797 auto operator()(const GridView& gridView) const {
798 static_assert(dim == GridView::dimension,
799 "The degree elevation argument should have the same size as the grid dimensions");
800 auto patchData = gridView.impl().patchData();
801 for (int dir = 0; auto elevatesInDirection : degreeElevate_) {
802 if (elevatesInDirection == 0) {
