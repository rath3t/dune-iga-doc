148template <int dim>
149class NurbsLocalCoefficients
150{
151 // Return i as a d-digit number in the (k+1)-nary system
152 std::array<unsigned int, dim> multiindex(unsigned int i) const {
153 std::array<unsigned int, dim> alpha;
154 for (int j = 0; j < dim; j++) {
155 alpha[j] = i % sizes_[j];
156 i = i / sizes_[j];
157 }
158 return alpha;
159 }
160
162 void setup1d(std::vector<unsigned int>& subEntity) {
163 if (sizes_[0] == 1) {
164 subEntity[0] = 0;
165 return;
166 }
167
168 /* edge and vertex numbering
169 0----0----1
170 */
171 unsigned lastIndex = 0;
172 subEntity[lastIndex++] = 0; // corner 0
173 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
174 subEntity[lastIndex++] = 0; // inner dofs of element (0)
175
176 subEntity[lastIndex++] = 1; // corner 1
177
178 assert(size() == lastIndex);
179 }
180
181 void setup2d(std::vector<unsigned int>& subEntity) {
182 unsigned lastIndex = 0;
183
184 // LocalKey: entity number , entity codim, dof indices within each entity
185 /* edge and vertex numbering
186 2----3----3
187 | |
188 | |
189 0 1
190 | |
191 | |
192 0----2----1
193 */
194
195 // lower edge (2)
196 subEntity[lastIndex++] = 0; // corner 0
197 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
198 subEntity[lastIndex++] = 2; // inner dofs of lower edge (2)
199
200 subEntity[lastIndex++] = 1; // corner 1
201
202 // iterate from bottom to top over inner edge dofs
203 for (unsigned e = 0; e < sizes_[1] - 2; ++e) {
204 subEntity[lastIndex++] = 0; // left edge (0)
205 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
206 subEntity[lastIndex++] = 0; // face dofs
207 subEntity[lastIndex++] = 1; // right edge (1)
208 }
209
210 // upper edge (3)
211 subEntity[lastIndex++] = 2; // corner 2
212 for (unsigned i = 0; i < sizes_[0] - 2; ++i)
213 subEntity[lastIndex++] = 3; // inner dofs of upper edge (3)
214
215 subEntity[lastIndex++] = 3; // corner 3
216
217 assert(size() == lastIndex);
218 }
219
220public:
221 void init(const std::array<unsigned, dim>& sizes) {
222 sizes_ = sizes;
223
224 li_.resize(size());
225
226 // Set up array of codimension-per-dof-number
227 std::vector<unsigned int> codim(li_.size());
228
229 for (std::size_t i = 0; i < codim.size(); i++) {
230 codim[i] = 0;
231 // Codimension gets increased by 1 for each coordinate direction
232 // where dof is on boundary
233 std::array<unsigned int, dim> mIdx = multiindex(i);
234 for (int j = 0; j < dim; j++)
235 if (mIdx[j] == 0 or mIdx[j] == sizes[j] - 1)
236 codim[i]++;
237 }
238
239 // Set up index vector (the index of the dof in the set of dofs of a given subentity)
240 // Algorithm: the 'index' has the same ordering as the dof number 'i'.
241 // To make it consecutive we interpret 'i' in the (k+1)-adic system, omit all digits
242 // that correspond to axes where the dof is on the element boundary, and transform the
243 // rest to the (k-1)-adic system.
244 std::vector<unsigned int> index(size());
245
246 for (std::size_t i = 0; i < index.size(); i++) {
247 index[i] = 0;
248
249 std::array<unsigned int, dim> mIdx = multiindex(i);
250
251 for (int j = dim - 1; j >= 0; j--)
252 if (mIdx[j] > 0 and mIdx[j] < sizes[j] - 1)
253 index[i] = (sizes[j] - 1) * index[i] + (mIdx[j] - 1);
254 }
255
256 // Set up entity and dof numbers for each (supported) dimension separately
257 std::vector<unsigned int> subEntity(li_.size());
258
259 if (!subEntity.empty()) {
260 if (dim == 1) {
261 setup1d(subEntity);
262
263 } else if (dim == 2 and sizes_[0] > 1 and sizes_[1] > 1) {
264 setup2d(subEntity);
265 }
266 }
267
268 for (size_t i = 0; i < li_.size(); i++)
269 li_[i] = LocalKey(subEntity[i], codim[i], index[i]);
270 }
271
272 // number of coefficients
273 [[nodiscard]] std::size_t size() const {
274 return std::accumulate(sizes_.begin(), sizes_.end(), 1, std::multiplies<>());
275 }
276
277 // get i'th index
278 [[nodiscard]] const LocalKey& localKey(std::size_t i) const {
279 return li_[i];
280 }
281
282private:
283 // Number of shape functions on this element per coordinate direction
284 std::array<unsigned, dim> sizes_;
285
286 std::vector<LocalKey> li_;
287};

355 // @todo save correct element info
356 void bind(const std::array<int, dim>& elementIdx) {
357 const auto& patchData = preBasis_.patchData_;
358 for (size_t i = 0; i < elementIdx.size(); i++) {
359 currentKnotSpan_[i] = Dune::IGANEW::Splines::findSpan(patchData.degree[i],

361 patchData.knotSpans[i], elementIdx[i]);

364 localBasis_.offset_[i] = preBasis_.patchData_.knotSpans[i][currentKnotSpan_[i]];
365 localBasis_.scaling_[i][i] = preBasis_.patchData_.knotSpans[i][currentKnotSpan_[i] + 1] -
366 preBasis_.patchData_.knotSpans[i][currentKnotSpan_[i]];
367 }

370 std::array<unsigned int, dim> sizes;
371 for (size_t i = 0; i < dim; i++)
372 sizes[i] = oneDimensionalSize(i);
373 localCoefficients_.init(sizes);
374 isBound = true;
375 elementIdx_ = elementIdx;
376 }

397 [[nodiscard]] unsigned size() const {
398 std::size_t r = 1;
399 for (int i = 0; i < dim; i++)
400 r *= oneDimensionalSize(i);
401 return r;
402 }
403

407 return GeometryTypes::cube(dim);
408 }

412 return preBasis_.patchData_.degree[i] + 1;
413 }

416
417 NurbsLocalCoefficients<dim> localCoefficients_;
418 NurbsLocalInterpolation<dim, NurbsLocalBasis<GV, R>> localInterpolation_;
419
420 // The knot span we are bound to
421 std::array<int, dim> currentKnotSpan_;
422 NurbsLocalBasis<GV, R> localBasis_;
423 std::array<int, dim> elementIdx_{};
424 bool isBound{false};
425};

575 void createUntrimmedNodeIndices() {
576 std::array<int, dim> localSizes;
577 size_type sizeOfShapeFunctions = 1;
578 for (int i = 0; i < dim; i++) {
579 localSizes[i] = patchData_.degree[i] + 1;
580 sizeOfShapeFunctions *= localSizes[i];
581 }
582 const auto order = patchData_.degree;
583
584 auto spanSize = gridView_.impl().untrimmedElementNumbers();
585 auto numElements = std::accumulate(spanSize.begin(), spanSize.end(), 1u, std::multiplies());
586
587 // auto extractIndex = std::views::transform([&](const auto& ele) { return getDirectIndex(ele); });
588 for (auto directIndex : Dune::range(numElements)) {
589 auto elementIdx = getIJK(directIndex, spanSize);
590
591 std::array<int, dim> currentKnotSpan;
592 for (size_t i = 0; i < elementIdx.size(); i++)
593 currentKnotSpan[i] =
594 IGANEW::Splines::findSpan(patchData_.degree[i], *(uniqueKnotVector_[i].begin() + elementIdx[i]),
595 patchData_.knotSpans[i], elementIdx[i]);
596
597 // Here magic is happening
598 for (size_type i = 0; i < sizeOfShapeFunctions; ++i) {
599 std::array<int, dim> localIJK = getIJK(i, localSizes);
600 std::array<unsigned int, dim> globalIJK;
601 for (int j = 0; j < dim; j++)
602 globalIJK[j] = std::max((int)currentKnotSpan[j] - (int)order[j], 0) + localIJK[j];
603
604 // Make one global flat index from the globalIJK tuple
605 size_type globalIdx = globalIJK[dim - 1];
606
607 for (int j = dim - 2; j >= 0; j--)
608 globalIdx = globalIdx * sizePerDirection(j) + globalIJK[j];
609
610 originalIndices_[directIndex].push_back(globalIdx);
611 }
612 }

637 void createTrimmedNodeIndices() {
638 if constexpr (not TrimmerType::isAlwaysTrivial) {
639 if (not gridView_.grid().trimmer().trimData_.has_value())

642 unsigned int n_ind_original = cachedSize_;

644 auto spanSize = gridView_.impl().untrimmedElementNumbers();

656 for (unsigned int i = 0; i < n_ind_original; ++i) {
657 if (std::ranges::find(indicesInTrim, i) != indicesInTrim.end())
658 indexMap_.emplace(i, realIndexCounter++);
659 }

661 }

698 void evaluateJacobian(const FieldVector<typename GV::ctype, dim>& in, std::vector<FieldMatrix<R, 1, dim>>& out,
699 const std::array<int, dim>& currentKnotSpan) const {
700 const auto dN = IGANEW::Splines::Nurbs<dim, ScalarType>::basisFunctionDerivatives(
701 in, patchData_.knotSpans, patchData_.degree, extractWeights(patchData_.controlPoints), 1, false,
702 currentKnotSpan);

704 for (int j = 0; j < dim; ++j) {
705 std::array<int, dim> multiIndex{};
706 multiIndex[j] = 1;
707 const auto& dNcurrent = dN.get(multiIndex);
708 for (int i = 0; i < dNcurrent.size(); ++i)
709 out[i][0][j] = dNcurrent.directGet(i);
710 }
