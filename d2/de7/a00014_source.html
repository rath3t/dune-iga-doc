67 const VectorType direction2) {

69 const ScalarType tol = 1e-8;
70
71 const VectorType basePointDifference = basepoint1 - basepoint2;
72 using Dune::dot;
73 const ScalarType dir1squaredLength = dot(direction1, direction1);
74 const ScalarType angle = dot(direction1, direction2);
75 const ScalarType dir2squaredLength = dot(direction2, direction2);
76 const ScalarType dir1BasePdiff = dot(direction1, basePointDifference);
77 const ScalarType dir2BasePdiff = dot(direction2, basePointDifference);
78 const ScalarType D = dir1squaredLength * dir2squaredLength - angle * angle;
79
80 if (std::abs(D) < tol)
81 DUNE_THROW(Dune::MathError, "The two lines are almost parallel.");
82
83 const ScalarType parameter1 = (angle * dir2BasePdiff - dir2squaredLength * dir1BasePdiff) / D;
84 const ScalarType parameter2 = (dir1squaredLength * dir2BasePdiff - angle * dir1BasePdiff) / D;
85
86 VectorType c1 = basepoint1 + parameter1 * direction1;
87 assert(dot(c1 - (basepoint2 + parameter2 * direction2), c1 - (basepoint2 + parameter2 * direction2)) < tol &&
88 "Both calculated points do not coincide");
89 return c1;
