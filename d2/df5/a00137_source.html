23template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
24class TrimmedLocalGeometryImpl<2, coorddim, GridImp, localGeometryTag>
25{
26public:
27 using ctype = typename GridImp::ctype;
28
29 static constexpr int mydimension = 2;
30 using ParameterSpace = typename GridImp::ParameterSpace;
31
32 static constexpr int coorddimension = coorddim;
33 static constexpr int codim = coorddimension - mydimension;
34 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
35 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
36
37 using HostGeometry =
38 typename ParameterSpace::ParameterSpaceTraits::template Codim<0>::HostParameterSpaceGridEntity::Geometry;
39 using TrimData = typename ParameterSpace::ElementTrimData;
40
41 using ParameterSpaceElement = typename GridImp::ParameterSpace::template Codim<0>::ParameterSpaceGridEntity;
42
43 using LocalCoordinate = FieldVector<ctype, mydimension>;
44 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
45 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
46 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
47 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
48 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
49 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
50 using Volume = ctype;
51
52 // type of the LocalView of the patch geometry
53 using GeometryLocalView =
54 typename GeometryKernel::NURBSPatch<mydimension, coorddimension,
55 ctype>::template GeometryLocalView<codim, ParameterSpace>;
56
58 TrimmedLocalGeometryImpl() = delete;
59 explicit TrimmedLocalGeometryImpl(const HostGeometry& hostGeometry, const ParameterSpaceElement& element)
60 : hostGeometry_{hostGeometry},
61 trimData_{element.trimData()},
62 element_(element) {}
63
64 bool operator==(const TrimmedLocalGeometryImpl& b) const {
65 return b.hostGeometry_ == this->hostGeometry_;
66 }
67
70 [[nodiscard]] GeometryType type() const {
71 if (Preferences::getInstance().reportTrimmedElementGeometryTypeAsNone())
72 return GeometryTypes::none(mydimension);
73 return GeometryTypes::cube(mydimension);
74 }
75
76 // return whether we have an affine mapping (true for straight lines)
77 [[nodiscard]] bool affine() const {
78 return true;
79 }
80
81 // return the number of corners of this element. Corners are numbered 0...n-1
82 [[nodiscard]] int corners() const {
83 return trimData_.size(2);
84 }
85
86 GlobalCoordinate center() const {
87 return GeometryKernel::centerOfMass(*this);
88 }
89
90 // access to coordinates of corners. Index is the number of the corner
91 GlobalCoordinate corner(int i) const {
92 auto vData = trimData_.vertex(i);
93 if (vData.isHost)
94 return hostGeometry_.corner(Transformations::mapToDune(2, vData.idx));
95
96 return vData.geometry.value();
97 }
98
101 GlobalCoordinate global(const LocalCoordinate& local) const {
102 return hostGeometry_.global(local);
103 }
104
107 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
108 return hostGeometry_.jacobianTransposed(local);
109 }
110
113 LocalCoordinate local(const GlobalCoordinate& global) const {
114 return hostGeometry_.local(global);
115 }
116
117 // Returns true if the point is in the current element
118 // TODO
119 bool checkInside(const LocalCoordinate& local) const {
120 return true;
121 }
122
123 // TODO
124 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
125 return hostGeometry_.volume();
126 }
127
128 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
129 return hostGeometry_.jacobianInverseTransposed(local);
130 }
131
132 auto getQuadratureRule(const std::optional<int>& p_order = std::nullopt,
133 const QuadratureType::Enum qt = QuadratureType::GaussLegendre) const {
134 return element_.getQuadratureRule(p_order, qt);
135 }
136
137 bool isTrimmed() const {
138 return true;
139 }
140
141private:
142 HostGeometry hostGeometry_;
143 TrimData trimData_;
144 ParameterSpaceElement element_; // This can not be a ptr as the entity could live shorter than the geo, maybe only
145 // store a quadrature rule
146};

148template <int coorddim, class GridImp, LocalGeometryTag localGeometryTag>
149class TrimmedLocalGeometryImpl<1, coorddim, GridImp, localGeometryTag>
150{
151public:
152 using ctype = typename GridImp::ctype;

157 static constexpr int coorddimension = coorddim;
158 static constexpr int codim = coorddimension - mydimension;
159 using PatchGeometry = GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>;
160 using LocalCoordinateInPatch = typename PatchGeometry::LocalCoordinate;
161 using LocalCoordinate = FieldVector<ctype, mydimension>;
162 using GlobalCoordinate = FieldVector<ctype, coorddimension>;
163 using JacobianTransposed = FieldMatrix<ctype, mydimension, coorddimension>;
164 using Hessian = FieldMatrix<ctype, mydimension*(mydimension + 1) / 2, coorddimension>;
165 using Jacobian = FieldMatrix<ctype, coorddimension, mydimension>;
166 using JacobianInverseTransposed = FieldMatrix<ctype, coorddimension, mydimension>;
167 using JacobianInverse = FieldMatrix<ctype, mydimension, coorddimension>;
168 using Volume = ctype;
169

171 using GeometryLocalView =
172 typename GeometryKernel::NURBSPatch<mydimension, coorddimension,

174
176 TrimmedLocalGeometryImpl() = default;
177 explicit TrimmedLocalGeometryImpl(const GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype>& patchGeometry)
178 : patchGeometry{patchGeometry} {}
179
180 bool operator==(const TrimmedLocalGeometryImpl& b) const {
181 return Dune::FloatCmp::eq(b.patchGeometry.corner(0) == this->patchGeometry.corner(0)) and
182 Dune::FloatCmp::eq(b.patchGeometry.corner(1) == this->patchGeometry.corner(1));
183 };
184
187 [[nodiscard]] GeometryType type() const {
188 return GeometryTypes::cube(mydimension);
189 }
190
191 // return whether we have an affine mapping (true for straight lines)
192 [[nodiscard]] bool affine() const {
193 // handy if the trims are straight lines for other algorithms
194 if constexpr (codim == 0)
195 return true;
196 else
197 return patchGeometry.degree()[0] == 1;
198 }
199
200 // return the number of corners of this element. Corners are numbered 0...n-1
201 [[nodiscard]] int corners() const {
202 return patchGeometry.corners();
203 }
204
205 GlobalCoordinate center() const {
206 return patchGeometry.center();
207 }
208
209 // access to coordinates of corners. Index is the number of the corner
210 GlobalCoordinate corner(int i) const {
211 return patchGeometry.corner(i);
212 }
213
216 GlobalCoordinate global(const LocalCoordinate& local) const {
217 return patchGeometry.global(local);
218 }
219
222 JacobianTransposed jacobianTransposed(const LocalCoordinate& local) const {
223 return patchGeometry.jacobianTransposed(local);
224 }
225
228 LocalCoordinate local(const GlobalCoordinate& global) const {
229 return patchGeometry.local(global);
230 }
231
232 // Returns true if the point is in the current element
233 // TODO
234 bool checkInside(const LocalCoordinate& local) const {
235 return true;
236 }
237
238 [[nodiscard]] Volume integrationElement(const LocalCoordinate& local) const {
239 return patchGeometry.volume();
240 }
241
242 // The Jacobian matrix of the mapping from the reference element to this element
243 // TODO not yet implemented
244 [[nodiscard]] JacobianInverseTransposed jacobianInverseTransposed(const LocalCoordinate& local) const {
245 return patchGeometry.jacobianInverseTransposed(local);
246 }
247
248private:
249 GeometryKernel::NURBSPatch<mydimension, coorddimension, ctype> patchGeometry;
250};
