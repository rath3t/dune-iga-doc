30 template <int dim>
31 auto& voigtIndices() {
32 if constexpr (dim == 1) {
33 static std::array<IndexPair, dim*(dim + 1) / 2> voigt1D = {
34 {0, 0}

37 } else if constexpr (dim == 2) {

44 {{0, 0}, {1, 1}, {2, 2}, {1, 2}, {0, 2}, {0, 1}}

92 // Coordinates in a single knot span differ from coordinates on the B-spline patch
93 // by an affine transformation. This transformation is stored in the diagonal entries.
94 // If trimming is disabled the Jacobian in the parameter space of subentities (edges or surfaces) is a
95 // DiagonalMatrixBlock but is treated as a FieldMatrix, since is also just cubes and an axis-aligned geometry, if
96 // trimming is enabled the Jacobian of the parameterspace for subentities is a potentially fully populated
97 // FieldMatrix, Think about an arbitrary curve in the axis-aligned cube parameter grid
98 // ------->
99 // `____________________C_______D_______`
100 // ::````````````````::``````````'|````::
101 // :: :: `| ::
102 // :: ^F:: `| ::
103 // :: | :: `| ::
104 // :: | :: `| ::
105 // :: |E:: `| ::
106 // :: :: .| x ::
107 // :: :: _: ::
108 // ::................::..........|:....::
109 // ::''''''''''''''''::''''''''''/'''''::
110 // :: :: _:B ::
111 // :: :: .| ::
112 // :: :: `::A ::
113 // :: '::::||::::::' ::
114 // :: ::` :: x ::
115 // :: |' :: ::
116 // :: | x :: ::
117 // ::````````|```````::````````````````::
118 // `____________________________________`
119 // at point A the tangent of the curve (the JacobianInParameterSpace) points into the oblique direction (to B)
120 // for the axis-aligned geometry these tangents are from E to F or from C to D, depending on the orientation of
121 // the subentity, which are aligned with the axis
122 using JacobianTransposedInParameterSpace = typename ParameterSpaceGeometry::JacobianTransposed;
123 using GlobalInParameterSpace = typename ParameterSpaceGeometry::GlobalCoordinate;
124

322 Hessian hessian(const LocalCoordinate& local) const {
323 checkState();
324 const auto ouInPatch = globalInParameterSpace(local);
325
326 const auto [p, dfdg, dfdgdg] =
327 patchGeometry_->zeroFirstAndSecondDerivativeOfPositionImpl(ouInPatch, nurbsLocalView_, localControlPointNet);
328
329 const JacobianTransposedInParameterSpace dgdt = jacobianTransposedInParameterSpace(local);
330 static_assert(JacobianTransposedInParameterSpace::rows == mydimension);
331 static_assert(JacobianTransposedInParameterSpace::cols == gridDimension);
332 // dgdt (1x2, curve on surface), (2x2 surface in surface), (2x3 surface in 3D), (3x3 volume in 3D)
333 FieldMatrix<ctype, patchNumberOfSecondDerivatives, numberOfSecondDerivatives> dgdtSquared(0);
334 for (int patchIndex = 0; auto [patchRow, patchCol] : Impl::voigtIndices<gridDimension>()) {
335 for (int myIndex = 0; auto [myRow, myCol] : Impl::voigtIndices<mydimension>()) {
336 if constexpr (not std::is_same_v<JacobianTransposedInParameterSpace, DiagonalMatrix<ctype, gridDimension>>)
337 dgdtSquared[patchIndex][myIndex] =
338 (dgdt[myRow][patchRow] * dgdt[myCol][patchCol] + dgdt[myCol][patchRow] * dgdt[myRow][patchCol]) *
339 ((patchRow == patchCol) ? 0.5 : 1.0);
340 else {
341 const auto dgdtmyRowpatchRow = (myRow == patchRow) ? dgdt[myRow][patchRow] : 0;
342 const auto dgdtmyColpatchCol = (myCol == patchCol) ? dgdt[myCol][patchCol] : 0;
343 const auto dgdtmyColpatchRow = (myCol == patchRow) ? dgdt[myCol][patchRow] : 0;
344 const auto dgdtmyRowpatchCol = (myRow == patchCol) ? dgdt[myRow][patchCol] : 0;
345 dgdtSquared[patchIndex][myIndex] =
346 (dgdtmyRowpatchRow * dgdtmyColpatchCol + dgdtmyColpatchRow * dgdtmyRowpatchCol) *
347 ((patchRow == patchCol) ? 0.5 : 1.0);
348 }
349 ++myIndex;
350 }
351 ++patchIndex;
352 }
353
354 // h = mydimension * (mydimension + 1) / 2 X worlddimension = 1x2 (curve in 2D), 1x3 (curve in 3D), (3x2 surface
355 // in 2D), (3x3 surface in 3D), (6x3 volume in 3D)
356 //
357 // dfdgdg = gridDimension*(gridDimension + 1) / 2 X worlddimension
358 // = 3x2 (curve on surface flat),3x3 (curve on surface in 3D), 6x3 (curve in volume), 6x3 (surface in
359 // volume),(3x3 surface in surface in 3D)
360 //
361 // dgdtSquared = gridDimension * (gridDimension + 1) / 2 X mydimension * (mydimension +
362 // 1) / 2 = 3x1 (curve on surface), 6x1 (curve in volume), 6x3 (surface in volume), 3x3 ( surface in surface),
363 // 6x6 ( volume in volume)
364 Hessian h = transposedView(dgdtSquared) * dfdgdg;
365
366 /* if trimming is enabled the parameter space geometry is potentially non-linear,
367 * the resutling Hessian has another contribution due to chain-rule, namely the second derivative of g */
368 if constexpr (not Trimmer::template isLocalGeometryLinear<codim>) {
369 // const auto dgdtdt = parameterSpaceGeometry->hessian(ouInPatch);
370 //
371 // assert(TrimmerType::template isLocalGeometryLinear<
372 // codim> && "This can not be checked yet. Check if this works with trimming and then remove
373 // assert");
374 // h += transpose(transposedView(dfdg) * dgdtdt);
375 }
376 return h;
377 }
