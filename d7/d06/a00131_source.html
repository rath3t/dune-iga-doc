52 void processVertexPair(const Vertex& vV1, const Vertex& vV2) {
53 if (vV1.isHost() && vV2.isHost()) {
54 processHostToHost(vV1, vV2);
55 } else if (vV1.isHost() && vV2.isNew()) {
56 processHostToNew(vV1, vV2);
57 } else if (vV1.isNew() && vV2.isNew()) {
58 processNewToNew(vV1, vV2);
59 } else if (vV1.isNew() && vV2.isHost()) {
60 processNewToHost(vV1, vV2);
61 } else {
62 if (not processAdditionalCases(vV1, vV2))
63 throwGridError();
64 }
65 }

92 void addNewBoundarySegementIdx(const CurveIndex& index, double t1, double t2) {
93 if (initFlag_) {
94 elementTrimData_.addBoundarySegmentIdxToLastEdge(parameterSpace_->numBoundarySegments_);
95 parameterSpace_->boundarySegmentsArchive_[yaspID].push_back(
96 std::make_tuple(index, t1, t2, parameterSpace_->numBoundarySegments_));
97 ++parameterSpace_->numBoundarySegments_;
98 } else {
99 auto fatherId = idSet.id(Util::coarsestFather(element));
100 auto archiveData = parameterSpace_->boundarySegmentsArchive_.at(fatherId);
101 auto it = findSegmentWithSameCurveIdx(archiveData, index);
102 assert(it != archiveData.end());
103 elementTrimData_.addBoundarySegmentIdxToLastEdge(std::get<3>(*it));
104 }

107 template <typename GV>
108 void setUpBoundarySegmentIndices(GV&& gridView) {
109 const auto& idSet = gridView.grid().globalIdSet();
110 yaspID = idSet.id(element);
111 for (const auto& intersection : Dune::intersections(gridView, element)) {
112 auto idx = Transformations::mapToParameterSpace(1, intersection.indexInInside());
113 if (not intersection.boundary())
114 boundarySegementIndices_[idx] = std::make_pair(false, std::numeric_limits<size_t>::max());
115 else
116 boundarySegementIndices_[idx] = std::make_pair(true, intersection.boundarySegmentIndex());
117 }
118 }

164template <int dim, int dimworld, typename ScalarType>
165auto ParameterSpaceImpl<dim, dimworld, ScalarType>::trimElement(const YASPEntity<0>& element, const auto& gv,
166 const PatchTrimData& patchTrimData, bool initFlag)
167 -> ElementTrimData {
168 auto [flag, result] = Util::clipElementRectangle(element, patchTrimData);
169
170 ElementTrimData elementTrimData(flag, element);
171 if (flag != ElementTrimFlag::trimmed)
172 return elementTrimData;
173
174 LoopHandler<ParameterSpaceImpl> handler(patchTrimData, elementTrimData, initFlag, this, gv, element);
175
176 for (const auto i : std::views::iota(0u, result.vertices_.size())) {
177 auto vV1 = result.vertices_[i];
178 auto vV2 = result.vertices_[(i + 1) % result.vertices_.size()];
179 handler.processVertexPair(vV1, vV2);
180 }
181
182 elementTrimData.finalize();
183 return elementTrimData;
184}
