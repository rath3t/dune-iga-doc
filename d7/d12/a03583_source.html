505 std::vector<ElementTrimData> trimElements(std::optional<int> level_ = std::nullopt) {
506 int level = level_.value_or(maxLevel());
507 bool initFlag = level == 0;
508 if (initFlag)
509 numBoundarySegments_ = untrimmedParameterSpaceGrid_->numBoundarySegments();
510
511 std::vector<ElementTrimData> elementTrimDatas;
512 auto gv = untrimmedParameterSpaceGrid_->levelGridView(level);
513 for (const auto& ele : elements(gv)) {
514 if (trimData_.has_value())
515 elementTrimDatas.emplace_back(trimElement(ele, gv, trimData_.value(), initFlag));
516 else
517 elementTrimDatas.emplace_back(ElementTrimFlag::full, ele);
518 }
519 return elementTrimDatas;
520 }
