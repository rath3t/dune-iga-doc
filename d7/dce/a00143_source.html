202 auto subEntities(int i, int c, int cc) const {
203 // auto subEntityName = [](std::size_t codim)-> std::string {
204 // return codim==0 ? "element" : codim==1 ? "edges" : "vertices" ;

206 // std::cout<<"The range numbers of "<<subEntityName(cc)<<" with codim "<<cc<<" of the "<<i<<"-th

209 if (trimData_.flag() == ElementTrimFlag::full) {
210 auto range = cubeGeometry.subEntities(i, c, cc);
211 return SubEntityRangeImpl(range.begin(), range.end(), true, i, c, cc);
212 }
213 if (trimData_.flag() == ElementTrimFlag::trimmed) {
214 if (cc < c)
215 return SubEntityRangeImpl(dummyArray.begin(), dummyArray.end());
216 else if (i == 0 and c == 0 and
217 cc == 0) // iterate over the entity itself (only one element in storage with value zero)
218 {
219 SubEntityRangeImpl range(elementIndexDummy.begin(), elementIndexDummy.end());
220 return range;
221 } else
222
223 if (i == 0 and c == 0 and cc == 1) // iterate over the edges of the element
224 {
225 SubEntityRangeImpl range(edgeIndices.begin(), edgeIndices.end());
226 return range;
227 } else if (c == 0 and cc == 2) // iterate over the vertices of the element
228 {
229 SubEntityRangeImpl range(verticesIndices.begin(), verticesIndices.end() - 1);
230 return range;
231 }
232
233 else if (c == 1 and cc == 2) // iterate over the vertices of edge
234 {
235 SubEntityRangeImpl range(verticesIndices.begin() + i, verticesIndices.begin() + i + 2);
236 return range;
237 }
238
239 else if (c == cc) // iterate over the vertices of edge
240 {
241 if (c == 2) {
242 SubEntityRangeImpl range(verticesIndices.begin() + i, verticesIndices.begin() + i + 1);
243 return range;
244 } else if (cc == 1) {
245 SubEntityRangeImpl range(edgeIndices.begin() + i, edgeIndices.begin() + i + 1);
246 return range;
247 }
248 }
249 }
250
251 assert(false);

283 bool contains(std::size_t i) const {
284 if (host_)
285 return cubeGeometry.subEntities(i_, c_, cc_).contains(i);
286 else {
287 if (size_ == 0)
288 return false;
289 else if (size_ == 2) // two vertices at one edge case
290 {
291 const int back = *(end() - 1);
292 const int front = *begin();
293 return i == front or i == back;
294 }
295 return size_ == 0 ? false : i >= *begin() and i <= *(end() - 1); // general case
296 }

432 explicit TrimmedReferenceElement(const TrimDataType& trimData)
433 : trimData_{trimData} {
434 if (trimData_.flag() == ElementTrimFlag::trimmed) {
435 edgeIndices.resize(trimData_.edges().size());
436 for (auto eI : Dune::range(trimData_.edges().size()))
437 edgeIndices[eI] = eI;
438
439 verticesIndices.resize(trimData_.vertices().size() + 1);
440 for (auto vI : Dune::range(trimData_.vertices().size())) {
441 verticesIndices[vI] = vI;
442 }
443 // to mimic a circular vector we add the first vertex at the end
444 verticesIndices.back() = 0;
445 }
